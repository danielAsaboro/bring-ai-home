A procedure that cannot say when to stop is dangerous to follow. Recorded workflows rot: layouts change, exports come back empty. The tape should carry its own stop conditions.

The same model proposes three lists with a second schema. The gate is stricter here: a parameter's example value must appear inside its own quote.

**Source checkpoint:** `08c0682` · **Starting checkpoint:** `01890a7`

## What changes

- `src/features/replay/conditions.ts`: `CONDITIONS_JSON_SCHEMA`, the prompt, `parseConditions`, `toParameterName` (snake_case), and `mergeConditions`, which never overwrites what a person already wrote.
- `/v1/propose-conditions` uses the same drafter door with a schema named `conditions`.
- The panel adds **Propose inputs and stop conditions** with checkboxes and reasons. Accepted items merge into the tape's parameter, precondition, and stop condition fields.
- The editor's text fields now follow the saved value after a merge.

To compare with the previous checkpoint: `git diff 01890a7 08c0682`.

## Build and observe

### 1. Propose

```bash
git checkout 08c0682
npm run qvac:service
npm run dev
```

Transcribe your narration and click **Propose inputs and stop conditions**.

### 2. Check the grounding

Each item shows its quote. A parameter like `week_start = September 28th` is only shown when "September 28th" appears inside the quoted words. A stop condition the narration never said is held back.

### 3. Accept and play

Tick the items you want and click **Add N selected items**. Play the tape: the player shows **Stop safely if** with your accepted conditions above the playhead. **Stop safely** still needs you to type a reason.

import { createHash } from "node:crypto";
import { mkdtemp, rm, writeFile } from "node:fs/promises";
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";
import { tmpdir } from "node:os";
import path from "node:path";

import { buildConditionsHistory, CONDITIONS_JSON_SCHEMA, parseConditions } from "../src/features/replay/conditions.ts";
import { toNarration, type NarrationSegment } from "../src/features/replay/narration.ts";
import { buildStepsHistory, parseStepProposals, STEPS_JSON_SCHEMA, type ChatMessage } from "../src/features/replay/proposals.ts";
import { parseScreenNote, SCREEN_NOTE_PROMPT } from "../src/features/replay/screen-notes.ts";
import {
  parseProposeConditionsRequest,
  parseProposeStepsRequest,
  parseReadScreenshotRequest,
  parseTranscribeRequest,
  type ModelState,
} from "../src/features/replay/service-contract.ts";

// Each model sits behind its own small door. Tests inject fakes; main.ts injects QVAC.
export type Transcriber = {
  modelState(): ModelState;
  transcribe(wavPath: string): Promise<NarrationSegment[]>;
};

// A multimodal model that looks at one image file and answers in plain text.
export type Viewer = {
  modelState(): ModelState;
  describe(imagePath: string, prompt: string): Promise<string>;
};

// Text generation constrained by a JSON schema. It returns raw text; the server validates it.
export type Drafter = {
  modelState(): ModelState;
  complete(history: ChatMessage[], schema: { name: string; schema: Record<string, unknown> }): Promise<string>;
};

export type ServiceModels = { transcriber?: Transcriber; viewer?: Viewer; drafter?: Drafter };

// Write request bytes to a private temp file for the length of one model call.
async function withTempFile<T>(name: string, bytes: Buffer, run: (filePath: string) => Promise<T>) {
  const directory = await mkdtemp(path.join(tmpdir(), "replay-"));
  try {
    const filePath = path.join(directory, name);
    await writeFile(filePath, bytes);
    return await run(filePath);
  } finally {
    await rm(directory, { recursive: true, force: true });
  }
}

const LOOPBACK_ORIGIN = /^http:\/\/(localhost|127\.0\.0\.1|\[::1\])(:\d+)?$/;

// Plain fields, not parameter properties: Node's type stripping cannot run those.
export class RequestError extends Error {
  status: number;
  code: string;
  constructor(status: number, code: string, message: string) {
    super(message);
    this.status = status;
    this.code = code;
  }
}

function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  const headers: Record<string, string> = { "content-type": "application/json" };
  if (origin) {
    headers["access-control-allow-origin"] = origin;
    headers["vary"] = "Origin";
  }
  res.writeHead(status, headers);
  res.end(JSON.stringify(body));
}

async function readJson(req: IncomingMessage, limit: number): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > limit) throw new RequestError(413, "too_large", "Request body is too large");
    chunks.push(chunk as Buffer);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString("utf8"));
  } catch {
    throw new RequestError(400, "invalid_json", "Body must be JSON");
  }
}

type Route = (req: IncomingMessage) => Promise<unknown>;

export function createReplayServer(models: ServiceModels) {
  const routes = new Map<string, Route>();

  if (models.transcriber) {
    const transcriber = models.transcriber;
    routes.set("/v1/transcribe", async (req) => {
      const parsed = parseTranscribeRequest(await readJson(req, 50_000_000));
      if (!parsed.ok) throw new RequestError(400, "invalid_request", parsed.message);
      const audio = Buffer.from(parsed.value.audioBase64, "base64");
      const sourceDigest = createHash("sha256").update(audio).digest("hex");
      // The recording lives on disk only for the length of this request.
      return withTempFile("narration.wav", audio, async (wavPath) => {
        const started = Date.now();
        const segments = await transcriber.transcribe(wavPath);
        let narration;
        try {
          narration = toNarration(segments);
        } catch (cause) {
          throw new RequestError(422, "no_speech", (cause as Error).message);
        }
        return { sourceDigest, model: transcriber.modelState().id, transcribeMs: Date.now() - started, narration };
      });
    });
  }

  if (models.viewer) {
    const viewer = models.viewer;
    routes.set("/v1/read-screenshot", async (req) => {
      const parsed = parseReadScreenshotRequest(await readJson(req, 25_000_000));
      if (!parsed.ok) throw new RequestError(400, "invalid_request", parsed.message);
      const image = Buffer.from(parsed.value.imageBase64, "base64");
      const sourceDigest = createHash("sha256").update(image).digest("hex");
      const ext = parsed.value.fileName.toLowerCase().split(".").pop();
      const raw = await withTempFile(`screen.${ext}`, image, (p) => viewer.describe(p, SCREEN_NOTE_PROMPT));
      try {
        return { sourceDigest, model: viewer.modelState().id, note: parseScreenNote(raw) };
      } catch (cause) {
        throw new RequestError(422, "empty_note", (cause as Error).message);
      }
    });
  }

  if (models.drafter) {
    const drafter = models.drafter;
    routes.set("/v1/propose-steps", async (req) => {
      const parsed = parseProposeStepsRequest(await readJson(req, 100_000));
      if (!parsed.ok) throw new RequestError(400, "invalid_request", parsed.message);
      const raw = await drafter.complete(buildStepsHistory(parsed.value), { name: "steps", schema: STEPS_JSON_SCHEMA });
      try {
        return { model: drafter.modelState().id, raw, result: parseStepProposals(raw, parsed.value.transcript) };
      } catch (cause) {
        throw new RequestError(422, "unusable_output", (cause as Error).message);
      }
    });
    routes.set("/v1/propose-conditions", async (req) => {
      const parsed = parseProposeConditionsRequest(await readJson(req, 100_000));
      if (!parsed.ok) throw new RequestError(400, "invalid_request", parsed.message);
      const raw = await drafter.complete(buildConditionsHistory(parsed.value.transcript), {
        name: "conditions",
        schema: CONDITIONS_JSON_SCHEMA,
      });
      try {
        return { model: drafter.modelState().id, raw, result: parseConditions(raw, parsed.value.transcript) };
      } catch (cause) {
        throw new RequestError(422, "unusable_output", (cause as Error).message);
      }
    });
  }

  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (origin !== undefined && !LOOPBACK_ORIGIN.test(origin)) {
      send(res, 403, { error: { code: "origin_rejected", message: "Only loopback origins may call this service" } });
      return;
    }
    if (req.method === "OPTIONS") {
      res.writeHead(204, {
        "access-control-allow-origin": origin ?? "",
        "access-control-allow-methods": "GET, POST",
        "access-control-allow-headers": "content-type",
      });
      res.end();
      return;
    }
    if (req.method === "GET" && req.url === "/health") {
      const states = Object.values(models).map((m) => m.modelState());
      send(res, 200, { service: "ready", models: states }, origin);
      return;
    }
    const route = req.method === "POST" && req.url ? routes.get(req.url) : undefined;
    if (!route) {
      send(res, 404, { error: { code: "not_found", message: "Unknown route" } }, origin);
      return;
    }
    try {
      send(res, 200, await route(req), origin);
    } catch (cause) {
      if (cause instanceof RequestError) {
        send(res, cause.status, { error: { code: cause.code, message: cause.message } }, origin);
        return;
      }
      const message = cause instanceof Error ? cause.message : String(cause);
      send(res, 500, { error: { code: "model_failed", message } }, origin);
    }
  });
}

// Parameters, preconditions, and stop conditions proposed from narration.
// Same rule as frames: the model proposes, this gate decides what can be shown, a person accepts.
import { normalizeForMatch, type ChatMessage } from "./proposals.ts";
import type { Parameter, Tape } from "./tape.ts";

const quoted = { type: "string" } as const;

export const CONDITIONS_JSON_SCHEMA = {
  type: "object",
  properties: {
    parameters: {
      type: "array",
      maxItems: 6,
      items: {
        type: "object",
        properties: { name: { type: "string" }, example: { type: "string" }, quote: quoted },
        required: ["name", "example", "quote"],
      },
    },
    preconditions: {
      type: "array",
      maxItems: 6,
      items: { type: "object", properties: { text: { type: "string" }, quote: quoted }, required: ["text", "quote"] },
    },
    stopConditions: {
      type: "array",
      maxItems: 6,
      items: { type: "object", properties: { text: { type: "string" }, quote: quoted }, required: ["text", "quote"] },
    },
  },
  required: ["parameters", "preconditions", "stopConditions"],
} as const;

export function buildConditionsHistory(transcript: string): ChatMessage[] {
  return [
    {
      role: "system",
      content:
        "You read a spoken demonstration of a repeatable task. List: parameters (values that change on each run, with a snake_case name and the example value that was said), " +
        "preconditions (what must be true before starting), and stop conditions (when the person said to stop or ask for help). " +
        "For every item, copy the exact supporting words from the narration into quote. Leave a list empty if nothing was said. Answer only with JSON.",
    },
    { role: "user", content: `Narration:\n${transcript}` },
  ];
}

export type ProposedParameter = Parameter & { quote: string };
export type ProposedCondition = { text: string; quote: string };
export type ConditionsResult = {
  parameters: ProposedParameter[];
  preconditions: ProposedCondition[];
  stopConditions: ProposedCondition[];
  rejected: { item: string; reason: string }[];
};

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

const str = (v: unknown) => (typeof v === "string" ? v.replace(/\s+/g, " ").trim() : "");

export function toParameterName(raw: string) {
  return raw
    .trim()
    .replace(/([a-z])([A-Z])/g, "$1_$2")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "_")
    .replace(/^_+|_+$/g, "")
    .slice(0, 40);
}

export function parseConditions(raw: string, transcript: string): ConditionsResult {
  let json: unknown;
  try {
    json = JSON.parse(raw);
  } catch {
    throw new Error("The model did not return JSON");
  }
  if (!isRecord(json)) throw new Error("The model's JSON is not an object");
  const haystack = normalizeForMatch(transcript);
  const grounded = (quote: string) => normalizeForMatch(quote).length >= 8 && haystack.includes(normalizeForMatch(quote));
  const result: ConditionsResult = { parameters: [], preconditions: [], stopConditions: [], rejected: [] };

  const names = new Set<string>();
  for (const item of Array.isArray(json.parameters) ? json.parameters.slice(0, 6) : []) {
    const r = isRecord(item) ? item : {};
    const name = toParameterName(str(r.name));
    const example = str(r.example);
    const quote = str(r.quote);
    const label = name || "(unnamed parameter)";
    if (!/^[a-z][a-z0-9_]*$/.test(name)) result.rejected.push({ item: label, reason: "Name is not usable as a parameter" });
    else if (names.has(name)) result.rejected.push({ item: label, reason: "Duplicate parameter" });
    else if (!grounded(quote)) result.rejected.push({ item: label, reason: "Quote does not appear in the narration" });
    else if (!example || !normalizeForMatch(quote).includes(normalizeForMatch(example))) {
      result.rejected.push({ item: label, reason: "Example value is not in the quoted words" });
    } else {
      names.add(name);
      result.parameters.push({ name, example, quote });
    }
  }

  for (const key of ["preconditions", "stopConditions"] as const) {
    const list = json[key];
    for (const item of Array.isArray(list) ? list.slice(0, 6) : []) {
      const r = isRecord(item) ? item : {};
      const text = str(r.text);
      const quote = str(r.quote);
      if (!text || text.length > 160) result.rejected.push({ item: text || "(empty)", reason: "Text is empty or too long" });
      else if (!grounded(quote)) result.rejected.push({ item: text, reason: "Quote does not appear in the narration" });
      else result[key].push({ text, quote });
    }
  }
  return result;
}

// Deterministic merge of what a person accepted. Existing entries win; nothing is overwritten.
export function mergeConditions(
  tape: Tape,
  accepted: { parameters: Parameter[]; preconditions: string[]; stopConditions: string[] },
) {
  const known = new Set(tape.parameters.map((p) => p.name));
  const uniq = (current: string[], extra: string[]) => {
    const seen = new Set(current.map(normalizeForMatch));
    return [...current, ...extra.filter((t) => !seen.has(normalizeForMatch(t)))];
  };
  return {
    parameters: [...tape.parameters, ...accepted.parameters.filter((p) => !known.has(p.name)).map(({ name, example }) => ({ name, example }))],
    preconditions: uniq(tape.preconditions, accepted.preconditions),
    stopConditions: uniq(tape.stopConditions, accepted.stopConditions),
  };
}

  onAddFrames: (steps: StepProposal[]) => void;
  onAddConditions?: (accepted: AcceptedConditions) => void;
}) {
  const [recording, setRecording] = useState<File | null>(null);
  const [narration, setNarration] = useState<TranscribeResponse | null>(null);
  const [status, setStatus] = useState<Status>({ kind: "idle" });
  const [screenshot, setScreenshot] = useState<File | null>(null);
  const [proposed, setProposed] = useState<ProposeStepsResponse | null>(null);
  const [chosen, setChosen] = useState<Set<string>>(new Set());
  const [conditions, setConditions] = useState<{ model: string; result: ConditionsResult } | null>(null);
  const [chosenConditions, setChosenConditions] = useState<Set<string>>(new Set());
  const [notes, setNotes] = useState<(ReadScreenshotResponse & { fileName: string })[]>([]);
  const btn = "rounded border border-stone-700 px-3 py-1 text-sm disabled:opacity-40";

  async function transcribe() {
    if (!recording) return;
    setStatus({ kind: "working", label: "Transcribing on this machine…" });
    const result = await transcribeRecording(recording);
    if (result.ok) {
      setNarration(result.value);
      setStatus({ kind: "idle" });
    } else {
      setStatus({ kind: "error", code: result.code, message: result.message });
    }
  }

  async function readScreen() {
    if (!screenshot) return;
    setStatus({ kind: "working", label: "Reading the screenshot on this machine…" });
    const result = await readScreenshot(screenshot);
    if (result.ok) {
      setNotes((n) => [...n, { ...result.value, fileName: screenshot.name }]);
      setStatus({ kind: "idle" });
    } else {
      setStatus({ kind: "error", code: result.code, message: result.message });
    }
  }

  async function propose() {
    if (!narration) return;
    setStatus({ kind: "working", label: "Proposing frames on this machine…" });
    const result = await proposeSteps({
      transcript: narration.narration.transcript,
      screenNotes: notes.map((n) => n.note.text),
    });
    if (result.ok) {
      setProposed(result.value);
      setChosen(new Set());
      setStatus({ kind: "idle" });
    } else {
      setStatus({ kind: "error", code: result.code, message: result.message });
    }
  }

  async function proposeInputs() {
    if (!narration) return;
    setStatus({ kind: "working", label: "Looking for inputs and stop conditions on this machine…" });
    const result = await proposeConditions(narration.narration.transcript);
    if (result.ok) {
      setConditions(result.value);
      setChosenConditions(new Set());
      setStatus({ kind: "idle" });
    } else {
      setStatus({ kind: "error", code: result.code, message: result.message });
    }
  }

  function addChosenConditions() {
    if (!conditions) return;
    const r = conditions.result;
    onAddConditions({
      parameters: r.parameters.filter((p) => chosenConditions.has(`param:${p.name}`)),
      preconditions: r.preconditions.filter((c) => chosenConditions.has(`pre:${c.text}`)).map((c) => c.text),
      stopConditions: r.stopConditions.filter((c) => chosenConditions.has(`stop:${c.text}`)).map((c) => c.text),
    });
    setConditions(null);
    setChosenConditions(new Set());
  }

  function toggle(set: (fn: (s: Set<string>) => Set<string>) => void, key: string, on: boolean) {
    set((c) => {
      const next = new Set(c);
      if (on) next.add(key);
      else next.delete(key);
      return next;
    });
  }

  function addChosen() {
    if (!proposed) return;
    onAddFrames(proposed.result.proposals.filter((p) => chosen.has(p.id)));
    setProposed(null);
    setChosen(new Set());
  }

  return (

      <h2 className="font-semibold">Draft from a demonstration</h2>

            className="ml-2 text-sm"

        </p>
      )}
      {conditions && (
        <div className="space-y-2">
          <p className="text-xs text-stone-400">Proposed by {conditions.model} · choose what belongs on the tape</p>
          <ul aria-label="Proposed conditions" className="space-y-1 text-sm">
            {[
              ...conditions.result.parameters.map((p) => ({ key: `param:${p.name}`, label: `Parameter ${p.name} = ${p.example}`, quote: p.quote })),
              ...conditions.result.preconditions.map((c) => ({ key: `pre:${c.text}`, label: `Before starting: ${c.text}`, quote: c.quote })),
              ...conditions.result.stopConditions.map((c) => ({ key: `stop:${c.text}`, label: `Stop if: ${c.text}`, quote: c.quote })),
            ].map((item) => (
              <li key={item.key}>
                <label className="flex gap-2">
                  <input

                    onChange={(e) => toggle(setChosenConditions, item.key, e.target.checked)}

"use client";

import { useEffect, useMemo, useState } from "react";

import { mergeConditions } from "./conditions";
import { DemonstrationPanel } from "./demonstration-panel";
import { createLocalStore, type LibraryState, type LibraryStore } from "./library-store";
import {
  addStep,
  advanceRun,
  blankTape,
  moveStep,
  removeStep,
  startRun,
  summarizeRun,
  updateStep,
  updateTape,
  type Run,
  type RunAction,
  type Tape,
} from "./tape";

type Mode = "edit" | "play" | "history";

const now = () => new Date().toISOString();

function lines(text: string) {
  return text.split("\n").map((l) => l.trim()).filter(Boolean);
}

export function ReplayApp({ store }: { store?: LibraryStore }) {
  const [library] = useState(() => store ?? createLocalStore(globalThis.localStorage));
  const [state, setState] = useState<LibraryState>(() => library.load());
  const [selectedId, setSelectedId] = useState<string>(() => state.tapes[0]?.id ?? "");
  const [mode, setMode] = useState<Mode>("edit");
  const [runId, setRunId] = useState<string | null>(null);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => library.save(state), [library, state]);

  const tape = state.tapes.find((t) => t.id === selectedId) ?? null;
  const run = state.runs.find((r) => r.id === runId) ?? null;

  function commitTape(next: Tape) {
    setState((s) => ({ ...s, tapes: s.tapes.map((t) => (t.id === next.id ? next : t)) }));
  }

  function attempt(fn: () => void) {
    try {
      setError(null);
      fn();
    } catch (cause) {
      setError(cause instanceof Error ? cause.message : String(cause));
    }
  }

  function play(t: Tape) {
    attempt(() => {
      const r = startRun(t, now());
      setState((s) => ({ ...s, runs: [r, ...s.runs] }));
      setRunId(r.id);
      setMode("play");
    });
  }

  function act(action: RunAction) {
    if (!run) return;
    attempt(() => {
      const next = advanceRun(run, action, now());
      setState((s) => ({ ...s, runs: s.runs.map((r) => (r.id === next.id ? next : r)) }));
    });
  }

  function newTape() {
    const t = blankTape("Untitled tape", now());
    setState((s) => ({ ...s, tapes: [t, ...s.tapes] }));
    setSelectedId(t.id);
    setMode("edit");
  }

  function resetDemo() {
    const fresh = library.reset();
    setState(fresh);
    setSelectedId(fresh.tapes[0]?.id ?? "");
    setRunId(null);
    setMode("edit");
  }

  return (
    <div className="min-h-screen bg-stone-950 text-stone-100">
      <header className="flex flex-wrap items-center justify-between gap-3 border-b border-stone-800 px-4 py-3">
        <div>
          <h1 className="text-2xl font-bold tracking-tight">Replay</h1>
          <p className="text-sm text-stone-400">A tape library for repeatable work.</p>
        </div>
        <button type="button" onClick={resetDemo} className="rounded border border-stone-700 px-3 py-1 text-sm">
          Reset demo data
        </button>
      </header>
      <div className="grid gap-4 p-4 md:grid-cols-[18rem_1fr]">
        <nav aria-label="Tape shelf" className="space-y-2">
          <button type="button" onClick={newTape} className="w-full rounded bg-amber-500 px-3 py-2 font-semibold text-stone-950">
            New tape
          </button>
          <ul className="space-y-2">
            {state.tapes.map((t) => (
              <li key={t.id}>
                <button
                  type="button"
                  aria-pressed={t.id === selectedId}
                  onClick={() => {
                    setSelectedId(t.id);
                    setMode("edit");
                  }}
                  className={`w-full rounded border px-3 py-2 text-left ${
                    t.id === selectedId ? "border-amber-500 bg-stone-900" : "border-stone-800"
                  }`}
                >
                  <span className="block font-medium">{t.title}</span>
                  <span className="text-xs text-stone-400">
                    {t.steps.length} frames · rev {t.revision}
                  </span>
                </button>
              </li>
            ))}
          </ul>
        </nav>
        <main className="min-w-0 space-y-4">
          {error && (
            <p role="alert" className="rounded border border-red-700 bg-red-950 px-3 py-2 text-sm">
              {error}
            </p>
          )}
          {tape ? (
            <>
              <div role="tablist" aria-label="Tape views" className="flex gap-2">
                {(["edit", "play", "history"] as const).map((m) => (
                  <button
                    key={m}
                    type="button"
                    role="tab"
                    aria-selected={mode === m}
                    onClick={() => (m === "play" && (!run || run.tapeId !== tape.id) ? play(tape) : setMode(m))}
                    className={`rounded px-3 py-1 text-sm capitalize ${mode === m ? "bg-stone-100 text-stone-950" : "bg-stone-800"}`}
                  >
                    {m}
                  </button>
                ))}

                <Player tape={tape} run={run} onAction={act} onReplay={() => play(tape)} />
              )}
              {mode === "history" && <History tape={tape} runs={state.runs.filter((r) => r.tapeId === tape.id)} />}
            </>
          ) : (

          )}
        </main>
      </div>
    </div>
  );

function TapeEditor({

        />

          label="Stop conditions"
          value={tape.stopConditions.join("\n")}

        />
      </div>
    </section>
  );
}

type ListFieldProps = { label: string; value: string; onCommit: (text: string) => void };

// Remount when the saved value changes (for example after accepting proposals), so the draft follows it.
function ListField(props: ListFieldProps) {
  return <ListFieldDraft key={props.value} {...props} />;
}

function ListFieldDraft({ label, value, onCommit }: ListFieldProps) {
  const [draft, setDraft] = useState(value);
  return (
    <label className="block text-sm">
      {label}
      <textarea
        className="h-24 w-full rounded border border-stone-700 bg-stone-900 px-2 py-1"
        value={draft}
        onChange={(e) => setDraft(e.target.value)}
        onBlur={() => {
          if (draft !== value) onCommit(draft);
        }}
      />
    </label>
  );
}

function Player({ tape, run, onAction, onReplay }: { tape: Tape; run: Run; onAction: (a: RunAction) => void; onReplay: () => void }) {

  const [reason, setReason] = useState("");
  const steps = useMemo(() => new Map(tape.steps.map((s) => [s.id, s])), [tape.steps]);
  const current = run.frames[run.playhead];
  const step = steps.get(current.stepId);
  const ended = run.status === "finished" || run.status === "stopped";
  const btn = "rounded border border-stone-700 px-3 py-1 text-sm disabled:opacity-40";

  return (
    <section aria-label="Player" className="space-y-4">
      <p className="text-sm text-stone-400">
        Replay guides you. It never performs a step for you. Status: <strong>{run.status}</strong>
      </p>
      {tape.stopConditions.length > 0 && (
        <div className="rounded border border-amber-700 p-2 text-sm">

            ))}
          </ul>
        </div>
      )}
      <ol aria-label="Tape frames" className="flex gap-1 overflow-x-auto pb-1">
        {run.frames.map((f, i) => (
          <li
            key={f.stepId}
            aria-current={i === run.playhead && !ended ? "step" : undefined}
            title={steps.get(f.stepId)?.title}
            className={`min-w-10 rounded px-2 py-1 text-center font-mono text-xs ${
              i === run.playhead && !ended
                ? "bg-amber-500 text-stone-950"
                : i < run.playhead || run.status === "finished"
                  ? f.outcome === "done"
                    ? "bg-emerald-800"
                    : f.outcome === "failed"
                      ? "bg-red-800"
                      : "bg-stone-700"
                  : "bg-stone-900"
            }`}
          >
            {String(i + 1).padStart(2, "0")}
          </li>
        ))}
      </ol>
      {!ended && step && (
        <div className="rounded border border-stone-700 p-3">
          <p className="text-xs text-stone-400">
            Frame {run.playhead + 1} of {run.frames.length}
            {current.outcome === "failed" && " · marked failed"}
          </p>
          <h2 className="text-lg font-semibold">{step.title}</h2>
          <p className="text-sm text-stone-300">{step.detail}</p>
        </div>
      )}
      {!ended ? (
        <div className="flex flex-wrap gap-2">
          <button type="button" className={btn} disabled={run.status === "paused" || current.outcome === "failed"} onClick={() => onAction({ type: "done" })}>
            Done
          </button>
          <button type="button" className={btn} disabled={run.status === "paused"} onClick={() => onAction({ type: "skip" })}>
            Skip
          </button>
          <button type="button" className={btn} disabled={run.status === "paused" || current.outcome === "failed"} onClick={() => onAction({ type: "fail" })}>
            Mark failed
          </button>
          <button type="button" className={btn} disabled={current.outcome !== "failed" || run.status === "paused"} onClick={() => onAction({ type: "retry" })}>
            Retry
          </button>
          {run.status === "paused" ? (
            <button type="button" className={btn} onClick={() => onAction({ type: "resume" })}>
              Resume
            </button>
          ) : (
            <button type="button" className={btn} onClick={() => onAction({ type: "pause" })}>
              Pause
            </button>
          )}
        </div>
      ) : (
        <div className="space-y-2">
          <p role="status">
            {run.status === "finished" ? "Run finished." : `Run stopped: ${run.stopReason}`}{" "}
            {(() => {
              const s = summarizeRun(run);
              return `${s.done} done, ${s.skipped} skipped, ${s.failed} failed.`;
            })()}
          </p>
          <button type="button" className={btn} onClick={onReplay}>
            Play again
          </button>
        </div>
      )}
      {!ended && (
        <div className="grid gap-2 md:grid-cols-2">
          <form
            className="flex gap-2"
            onSubmit={(e) => {
              e.preventDefault();
              onAction({ type: "annotate", note });
              setNote("");
            }}
          >
            <input aria-label="Frame note" placeholder={current.note || "Note for this frame"} className="flex-1 rounded border border-stone-700 bg-stone-900 px-2" value={note} onChange={(e) => setNote(e.target.value)} />
            <button type="submit" className={btn}>
              Save note
            </button>
          </form>
          <form
            className="flex gap-2"
            onSubmit={(e) => {
              e.preventDefault();
              onAction({ type: "stop", reason });
            }}
          >
            <input aria-label="Stop reason" placeholder="Why stop?" className="flex-1 rounded border border-stone-700 bg-stone-900 px-2" value={reason} onChange={(e) => setReason(e.target.value)} />
            <button type="submit" className={btn}>
              Stop safely
            </button>
          </form>
        </div>
      )}
    </section>
  );
}

function History({ tape, runs }: { tape: Tape; runs: Run[] }) {

                    </li>
                  ))}
              </ul>
            )}
          </li>
        );
      })}
    </ul>
  );
}

export type ClientResult<T> = { ok: true; value: T } | { ok: false; code: string; message: string };

// Every call can fail in three different ways, and the UI says which one happened.
export async function postJson<T>(
  path: string,
  body: unknown,
  parse: (json: unknown) => Parsed<T>,
  fetchImpl: typeof fetch = fetch,
): Promise<ClientResult<T>> {
  let response: Response;
  try {
    response = await fetchImpl(`${SERVICE_URL}${path}`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(body),
    });
  } catch {
    return {
      ok: false,
      code: "service_unreachable",
      message: "The local QVAC service is not running. Start it with npm run qvac:service. Manual editing still works.",
    };
  }
  let json: unknown;
  try {
    json = await response.json();
  } catch {
    return { ok: false, code: "invalid_response", message: "The service answered with something that is not JSON" };
  }
  if (!response.ok) {
    const error = parseServiceError(json);
    return { ok: false, code: error?.code ?? "service_error", message: error?.message ?? `HTTP ${response.status}` };
  }
  const parsed = parse(json);
  return parsed.ok ? parsed : { ok: false, code: "invalid_response", message: parsed.message };
}

export async function fileToBase64(file: Blob) {
  const bytes = new Uint8Array(await file.arrayBuffer());
  let binary = "";
  for (let i = 0; i < bytes.length; i += 0x8000) {
    binary += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(binary);
}

// JSON from another process can be anything, so both sides validate at runtime.
import type { ConditionsResult } from "./conditions.ts";
import type { Narration } from "./narration.ts";

export const SERVICE_PORT = 4319;
export const SERVICE_URL = `http://127.0.0.1:${SERVICE_PORT}`;

export type Parsed<T> = { ok: true; value: T } | { ok: false; message: string };

export type ModelState = { kind: string; id: string; state: "not_loaded" | "loading" | "ready" | "error" };

export type ServiceError = { error: { code: string; message: string } };

export type TranscribeRequest = { fileName: string; audioBase64: string };
export type TranscribeResponse = {
  sourceDigest: string;
  model: string;
  transcribeMs: number;
  narration: Narration;
};

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function nonEmptyString(value: unknown, max: number): value is string {
  return typeof value === "string" && value.trim().length > 0 && value.length <= max;
}

export function parseTranscribeRequest(json: unknown): Parsed<TranscribeRequest> {
  if (!isRecord(json)) return { ok: false, message: "Body must be an object" };
  if (!nonEmptyString(json.fileName, 200)) return { ok: false, message: "fileName is required" };
  if (!/\.wav$/i.test(json.fileName)) return { ok: false, message: "Only .wav recordings are supported" };
  if (!nonEmptyString(json.audioBase64, 40_000_000)) return { ok: false, message: "audioBase64 is required" };
  return { ok: true, value: { fileName: json.fileName, audioBase64: json.audioBase64 } };
}

export function parseTranscribeResponse(json: unknown): Parsed<TranscribeResponse> {
  if (!isRecord(json) || !isRecord(json.narration)) return { ok: false, message: "Missing narration" };
  const n = json.narration;
  if (typeof n.transcript !== "string" || !Array.isArray(n.segments)) {
    return { ok: false, message: "Narration is malformed" };
  }
  const segmentsOk = n.segments.every(
    (s) => isRecord(s) && typeof s.text === "string" && typeof s.startMs === "number" && typeof s.endMs === "number",
  );
  if (!segmentsOk || typeof json.model !== "string" || typeof json.sourceDigest !== "string") {
    return { ok: false, message: "Narration is malformed" };
  }
  return { ok: true, value: json as TranscribeResponse };
}

export function parseServiceError(json: unknown): ServiceError["error"] | null {
  if (isRecord(json) && isRecord(json.error) && typeof json.error.code === "string" && typeof json.error.message === "string") {
    return { code: json.error.code, message: json.error.message };
  }
  return null;
}

export function parseProposeStepsRequest(json: unknown): Parsed<ProposeStepsRequest> {
