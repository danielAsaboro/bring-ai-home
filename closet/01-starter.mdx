Most people own more outfits than they remember. The shirt that only works with one pair of trousers, the jacket that is always in the wash on the morning you need it, the weekend bag that ends up with three pairs of shoes. Closet is a private fashion feed built from clothes you actually own.

This chapter builds the whole product without AI. That is on purpose. Every later chapter adds one local capability, and the app must keep working when that capability is missing.

Use Node 22.22.2 or newer. Open `http://localhost:3000`.

### 2. Read the feed

The Feed tab shows four outfit posts, newest first, and a row of stories counting garments that are worn, washing, or in repair. Every garment and image is invented. The drawings in `public/garments` are simple shapes, safe to share and easy to judge by eye.

### 3. Move a garment through its states

Open **Garments**. The grey tee is `worn`. Its buttons only offer `washing`, `clean`, and `repair`. Open `model.ts` and find `TRANSITIONS`. A clean shirt cannot jump straight into the wash; it has to be worn first. Repair is reachable from anywhere.

Mark the grey tee `washing` and reload the page. It stays `washing`. `closet-store.ts` saved it. That file is the only one that knows about `localStorage`; swap it and the rest of the app does not change.

### 4. Assemble a look

Open **Assemble**, give the look a title, tick two garments, and press **Post look**. It appears at the top of the feed. `createLook` refuses a look with fewer than two garments, and refuses ids that are not in your closet. Keep that rule in mind; a later chapter relies on it when a model proposes garment ids.

### 5. Pack the carousel

Open **Packing**. The Lisbon weekend has a limit of six. Only clean garments appear under **Clean and ready**. Mark a packed garment `worn` in the Garments tab and it leaves the carousel, because you would not pack it.

### 6. Reset

**Reset demo data** clears saved storage and restores the fictional wardrobe.

### 7. Write down the rule for this course

> QVAC may describe, label, rank, and propose. Only Closet's own rules decide what is valid, and only a person adds a garment, posts a look, or packs a bag.

We will test this rule in every chapter.

```typescript
// Closet's domain model. Every rule about what a garment, look, or trip may
// contain lives here, so the UI and (later) any local model can only propose.

export const CATEGORIES = ["top", "bottom", "dress", "outerwear", "shoes", "accessory"] as const;
export type Category = (typeof CATEGORIES)[number];

export const GARMENT_STATES = ["clean", "worn", "washing", "repair"] as const;
export type GarmentState = (typeof GARMENT_STATES)[number];

export type Garment = {
  id: string;
  name: string;
  category: Category;
  colour: string;
  material: string;
  notes: string;
  state: GarmentState;
  image: string | null;
};

export type Look = {
  id: string;
  title: string;
  occasion: string;
  garmentIds: string[];
  createdAt: string;
};

export type Trip = {
  name: string;
  limit: number;
  garmentIds: string[];
};

export type Closet = {
  garments: Garment[];
  looks: Look[];
  trip: Trip;
};

// Which state a garment may move to next. Repair is reachable from anywhere.
const TRANSITIONS: Record<GarmentState, GarmentState[]> = {
  clean: ["worn", "repair"],
  worn: ["washing", "clean", "repair"],
  washing: ["clean", "repair"],
  repair: ["clean"],
};

export function nextStates(state: GarmentState): GarmentState[] {
  return TRANSITIONS[state];
}

export function isCategory(value: unknown): value is Category {
  return typeof value === "string" && (CATEGORIES as readonly string[]).includes(value);
}

export function setGarmentState(closet: Closet, id: string, state: GarmentState): Closet {
  const garment = closet.garments.find((g) => g.id === id);
  if (!garment) throw new Error(`Unknown garment ${id}`);
  if (!TRANSITIONS[garment.state].includes(state)) {
    throw new Error(`${garment.name} cannot go from ${garment.state} to ${state}`);
  }
  // A garment that is no longer clean leaves the packing carousel.
  const trip =
    state === "clean"
      ? closet.trip
      : { ...closet.trip, garmentIds: closet.trip.garmentIds.filter((g) => g !== id) };
  return {
    ...closet,
    trip,
    garments: closet.garments.map((g) => (g.id === id ? { ...g, state } : g)),
  };
}

export type GarmentDraft = Omit<Garment, "id" | "state"> & { state?: GarmentState };

function slug(text: string) {
  return text.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/(^-|-$)/g, "") || "item";
}

function uniqueId(base: string, taken: Set<string>) {
  let id = base;
  for (let n = 2; taken.has(id); n += 1) id = `${base}-${n}`;
  return id;
}

export function addGarment(closet: Closet, draft: GarmentDraft): Closet {
  const name = draft.name.trim();
  if (!name) throw new Error("A garment needs a name");
  if (!isCategory(draft.category)) throw new Error("Choose a category");
  const id = uniqueId(slug(name), new Set(closet.garments.map((g) => g.id)));
  const garment: Garment = {
    id,
    name,
    category: draft.category,
    colour: draft.colour.trim(),
    material: draft.material.trim(),
    notes: draft.notes.trim(),
    state: draft.state ?? "clean",
    image: draft.image,
  };
  return { ...closet, garments: [...closet.garments, garment] };
}

export function updateGarment(
  closet: Closet,
  id: string,
  patch: Partial<Pick<Garment, "name" | "category" | "colour" | "material" | "notes">>,
): Closet {
  if (patch.name !== undefined && !patch.name.trim()) throw new Error("A garment needs a name");
  if (patch.category !== undefined && !isCategory(patch.category)) throw new Error("Choose a category");
  return {
    ...closet,
    garments: closet.garments.map((g) => (g.id === id ? { ...g, ...patch } : g)),
  };
}

export function createLook(
  closet: Closet,
  input: { title: string; occasion: string; garmentIds: string[] },
  now: Date = new Date(),
): Closet {
  const title = input.title.trim();
  if (!title) throw new Error("A look needs a title");
  const ids = [...new Set(input.garmentIds)];
  if (ids.length < 2) throw new Error("A look needs at least two garments");
  const known = new Set(closet.garments.map((g) => g.id));
  const unknown = ids.filter((id) => !known.has(id));
  if (unknown.length) throw new Error(`Unknown garments: ${unknown.join(", ")}`);
  const id = uniqueId(`look-${slug(title)}`, new Set(closet.looks.map((l) => l.id)));
  const look: Look = {
    id,
    title,
    occasion: input.occasion.trim(),
    garmentIds: ids,
    createdAt: now.toISOString(),
  };
  return { ...closet, looks: [look, ...closet.looks] };
}

export function packGarment(closet: Closet, id: string): Closet {
  const garment = closet.garments.find((g) => g.id === id);
  if (!garment) throw new Error(`Unknown garment ${id}`);
  if (closet.trip.garmentIds.includes(id)) return closet;
  if (garment.state !== "clean") throw new Error(`${garment.name} is not clean`);
  if (closet.trip.garmentIds.length >= closet.trip.limit) {
    throw new Error(`The carousel is full (${closet.trip.limit} items)`);
  }
  return { ...closet, trip: { ...closet.trip, garmentIds: [...closet.trip.garmentIds, id] } };
}

export function unpackGarment(closet: Closet, id: string): Closet {
  return {
    ...closet,
    trip: { ...closet.trip, garmentIds: closet.trip.garmentIds.filter((g) => g !== id) },
  };
}

export function setTripLimit(closet: Closet, limit: number): Closet {
  const safe = Math.max(1, Math.min(30, Math.floor(limit)));
  return {
    ...closet,
    trip: { ...closet.trip, limit: safe, garmentIds: closet.trip.garmentIds.slice(0, safe) },
  };
}
```

```typescript
import { CATEGORIES, GARMENT_STATES, type Closet } from "./model";
import { sampleCloset } from "./sample-closet";

export const STORAGE_KEY = "closet:v1";

type KeyValue = Pick<Storage, "getItem" | "setItem" | "removeItem">;

// The one place Closet touches browser storage. Swap the adapter, keep the app.
export function createClosetStore(storage: KeyValue | null) {
  return {
    load(): Closet {
      if (!storage) return sampleCloset();
      try {
        const raw = storage.getItem(STORAGE_KEY);
        if (!raw) return sampleCloset();
        const parsed = JSON.parse(raw) as unknown;
        return isCloset(parsed) ? parsed : sampleCloset();
      } catch {
        return sampleCloset();
      }
    },
    save(closet: Closet) {
      try {
        storage?.setItem(STORAGE_KEY, JSON.stringify(closet));
      } catch {
        // Storage can be full or blocked; the closet stays usable in memory.
      }
    },
    reset(): Closet {
      try {
        storage?.removeItem(STORAGE_KEY);
      } catch {
        // ignore
      }
      return sampleCloset();
    },
  };
}

function isCloset(value: unknown): value is Closet {
  const v = value as Closet | null;
  return (
    !!v &&
    Array.isArray(v.garments) &&
    v.garments.every(
      (g) =>
        typeof g.id === "string" &&
        typeof g.name === "string" &&
        (CATEGORIES as readonly string[]).includes(g.category) &&
        (GARMENT_STATES as readonly string[]).includes(g.state),
    ) &&
    Array.isArray(v.looks) &&
    !!v.trip &&
    Array.isArray(v.trip.garmentIds) &&
    typeof v.trip.limit === "number"
  );
}
```

```tsx
function browserStorage() {
  try {
    return typeof window === "undefined" ? null : window.localStorage;
  } catch {
    return null;

export function ClosetApp() {
  const store = useMemo(() => createClosetStore(browserStorage()), []);
  const [closet, setCloset] = useState<Closet>(sampleCloset);
  const [loaded, setLoaded] = useState(false);
  const [tab, setTab] = useState<Tab>("feed");
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    // Read saved wardrobe after hydration so server and client markup match.
    // eslint-disable-next-line react-hooks/set-state-in-effect
    setCloset(store.load());
    setLoaded(true);
  }, [store]);

  useEffect(() => {
    if (loaded) store.save(closet);
  }, [closet, loaded, store]);

  // Run every change synchronously so domain errors surface in the UI.
  function act(change: (c: Closet) => Closet) {
    try {
      setCloset(change(closet));
      setError(null);
    } catch (e) {
      setError((e as Error).message);
    }
  }

  const byId = new Map(closet.garments.map((g) => [g.id, g]));

        <h1 className="text-2xl font-bold tracking-tight">Closet</h1>
        <button
          type="button"
          className="rounded border border-stone-300 px-3 py-1 text-sm"
          onClick={() => act(() => store.reset())}
        >
          Reset demo data
        </button>
      </header>

      <nav role="tablist" aria-label="Closet sections" className="mb-4 flex gap-1 overflow-x-auto">
        {TABS.map((t) => (
          <button
            key={t.id}
            role="tab"
            type="button"
            aria-selected={tab === t.id}
            onClick={() => setTab(t.id)}
            className={`rounded-full px-4 py-1 text-sm ${tab === t.id ? "bg-stone-900 text-white" : "bg-stone-100"}`}
          >
            {t.label}
          </button>
        ))}
      </nav>

      {error && (
        <p role="alert" className="mb-3 rounded bg-rose-50 px-3 py-2 text-sm text-rose-900">

  act,
  onDone,
}: {
  closet: Closet;
  act: (c: (c: Closet) => Closet) => void;
  onDone: () => void;

  const [title, setTitle] = useState("");
  const [occasion, setOccasion] = useState("");
  const [picked, setPicked] = useState<string[]>([]);

          e.preventDefault();
          let ok = false;

            const next = createLook(c, { title, occasion, garmentIds: picked });
            ok = true;
            return next;
          });
          if (ok) onDone();
        }}
      >
        <label className="flex flex-col">
          Look title
          <input value={title} onChange={(e) => setTitle(e.target.value)} className="rounded border px-2 py-1" />
        </label>
        <label className="flex flex-col">
          Occasion
          <input value={occasion} onChange={(e) => setOccasion(e.target.value)} className="rounded border px-2 py-1" />
        </label>
        <fieldset className="grid grid-cols-2 gap-2 sm:grid-cols-3">
          <legend className="mb-1">Garments</legend>
          {closet.garments.map((g) => (
            <label key={g.id} className="flex items-center gap-2 rounded border border-stone-200 p-1">
              <input
                type="checkbox"
                checked={picked.includes(g.id)}
                onChange={(e) =>
                  setPicked((p) => (e.target.checked ? [...p, g.id] : p.filter((x) => x !== g.id)))
                }
              />
              {g.name}
            </label>
          ))}
        </fieldset>
        <button type="submit" className="rounded bg-stone-900 px-3 py-1 text-white">
          Post look
        </button>
      </form>
    </section>
  );
}

function Packing({
  closet,
  byId,
  act,
}: {
  closet: Closet;
  byId: Map<string, Garment>;
  act: (c: (c: Closet) => Closet) => void;
}) {
  const packed = closet.trip.garmentIds.map((id) => byId.get(id)).filter((g): g is Garment => !!g);
  const available = closet.garments.filter((g) => g.state === "clean" && !closet.trip.garmentIds.includes(g.id));
  return (
    <section aria-label="Packing" className="text-sm">
      <h2 className="font-semibold">
        {closet.trip.name} · {packed.length}/{closet.trip.limit} packed
      </h2>
      <label className="my-2 flex items-center gap-2">
        Packing limit
        <input
          type="number"
          min={1}
          max={30}
          value={closet.trip.limit}
          onChange={(e) => act((c) => setTripLimit(c, Number(e.target.value)))}
          className="w-16 rounded border px-1"
        />
      </label>
      <ul aria-label="Packing carousel" className="mb-4 flex snap-x gap-3 overflow-x-auto pb-2">
        {packed.map((g) => (
          <li key={g.id} className="flex shrink-0 snap-start flex-col items-center gap-1">
            <Thumb garment={g} size={96} />
            <button type="button" aria-label={`Unpack ${g.name}`} onClick={() => act((c) => unpackGarment(c, g.id))} className="underline">
              Unpack
            </button>
          </li>
        ))}
      </ul>
      <h3 className="font-semibold">Clean and ready</h3>
      <ul className="mt-1 flex flex-wrap gap-2">
        {available.map((g) => (
          <li key={g.id}>
            <button type="button" aria-label={`Pack ${g.name}`} onClick={() => act((c) => packGarment(c, g.id))} className="rounded border border-stone-300 px-2 py-1">
              + {g.name}
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}
```

```typescript
import { describe, expect, it } from "vitest";

import { createClosetStore, STORAGE_KEY } from "./closet-store";
import { addGarment, createLook, packGarment, setGarmentState, setTripLimit } from "./model";
import { sampleCloset } from "./sample-closet";

function memoryStorage() {
  const data = new Map<string, string>();
  return {
    getItem: (k: string) => data.get(k) ?? null,
    setItem: (k: string, v: string) => void data.set(k, v),
    removeItem: (k: string) => void data.delete(k),
    data,
  };
}

describe("garment states", () => {
  it("follows clean → worn → washing → clean", () => {
    let c = sampleCloset();
    c = setGarmentState(c, "white-oxford", "worn");
    c = setGarmentState(c, "white-oxford", "washing");
    c = setGarmentState(c, "white-oxford", "clean");
    expect(c.garments.find((g) => g.id === "white-oxford")?.state).toBe("clean");
  });

  it("refuses a jump the wardrobe does not allow", () => {
    expect(() => setGarmentState(sampleCloset(), "white-oxford", "washing")).toThrow("cannot go");
  });

  it("drops a garment from the carousel when it stops being clean", () => {
    const c = setGarmentState(sampleCloset(), "white-oxford", "worn");
    expect(c.trip.garmentIds).not.toContain("white-oxford");
  });
});

describe("looks and packing", () => {
  it("posts a look only from garments you own", () => {
    const c = createLook(sampleCloset(), { title: "Dinner", occasion: "date", garmentIds: ["navy-blazer", "dark-jeans"] });
    expect(c.looks[0].title).toBe("Dinner");
    expect(() =>
      createLook(sampleCloset(), { title: "X", occasion: "", garmentIds: ["navy-blazer", "ghost"] }),
    ).toThrow("Unknown garments: ghost");
  });

  it("packs only clean garments and respects the limit", () => {
    let c = setTripLimit(sampleCloset(), 3);
    c = packGarment(c, "red-scarf");
    expect(() => packGarment(c, "brown-boots")).toThrow("full");
    expect(() => packGarment(setTripLimit(c, 9), "grey-tee")).toThrow("not clean");
  });

  it("gives new garments unique ids", () => {
    const draft = { name: "Navy blazer", category: "outerwear" as const, colour: "", material: "", notes: "", image: null };
    const c = addGarment(sampleCloset(), draft);
    expect(c.garments.at(-1)?.id).toBe("navy-blazer-2");
  });
});

describe("closet store adapter", () => {
  it("round-trips through storage and resets to the sample closet", () => {
    const storage = memoryStorage();
    const store = createClosetStore(storage);
    const changed = setGarmentState(store.load(), "red-scarf", "worn");
    store.save(changed);
    expect(store.load().garments.find((g) => g.id === "red-scarf")?.state).toBe("worn");
    expect(store.reset().garments.find((g) => g.id === "red-scarf")?.state).toBe("clean");
    expect(storage.data.has(STORAGE_KEY)).toBe(false);
  });

  it("ignores corrupt saved data instead of crashing", () => {
    const storage = memoryStorage();
    storage.setItem(STORAGE_KEY, "{not json");
    expect(createClosetStore(storage).load().garments).toHaveLength(12);
  });
});
```
