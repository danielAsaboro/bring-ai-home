## Build and observe

### 1. Break the pacing

Set **Prototype trial** to 240 seconds and add an empty round. The panel flags the heavy round, the empty round, and the long card at once, without the service.

### 2. Add a duplicate beat

In **The problem**, add a card titled `Rushed mornings again`. Press **Check for duplicate beats**. If the embeddings put it close to **Mornings are rushed**, the panel names both cards and their similarity.

### 3. Decide

**Discard the later card** removes it. **Keep as is** hides the flag for this session. Nothing is removed without one of those clicks.

function Match({ deck, onExit }: { deck: Deck; onExit: () => void }) {
  const cards = playOrder(deck);
  const [index, setIndex] = useState(0);
  const card = cards[Math.min(index, cards.length - 1)];
  const round = deck.rounds.find((r) => r.id === card?.roundId);

  useEffect(() => {
    function onKey(event: KeyboardEvent) {
      if (event.key === "ArrowRight") setIndex((i) => Math.min(i + 1, cards.length - 1));
      if (event.key === "ArrowLeft") setIndex((i) => Math.max(i - 1, 0));
      if (event.key === "Escape") onExit();
    }
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [cards.length, onExit]);

  if (!card) return null;
  return (
    <section aria-label="Presentation" className="flex min-h-screen flex-col items-center justify-center gap-6 bg-slate-900 p-8 text-white">
      <p className="text-sm uppercase tracking-widest text-slate-400">
        {round?.title} · card {index + 1} of {cards.length}
      </p>
      <h2 className={card.treatment === "headline" ? "text-5xl font-bold" : "text-3xl"}>{card.title}</h2>
      <p className="max-w-2xl text-center text-xl text-slate-200">{card.body}</p>
      {card.source && <p className="text-sm text-slate-400">Source: {card.source}</p>}
      <div className="flex gap-3">
        <button className={button} onClick={() => setIndex((i) => Math.max(i - 1, 0))} disabled={index === 0}>
          Previous
        </button>
        <button className={button} onClick={() => setIndex((i) => Math.min(i + 1, cards.length - 1))} disabled={index >= cards.length - 1}>
          Next
        </button>
        <button className={button} onClick={onExit}>
          Leave the match
        </button>
      </div>
    </section>
  );
}

"use client";

import { useState } from "react";

import { AssistError, type AssistClient } from "./assist-client";
import type { Deck } from "./deck";
import { cardsInPlayOrder, duplicateBeats, pacingFlags, type HealthFlag } from "./deck-health";
import { cardText } from "./evidence-pool";

const button = "rounded bg-indigo-800 px-3 py-1 text-sm font-medium text-white hover:bg-indigo-900 disabled:opacity-50";
const ghost = "rounded border border-slate-400 px-2 py-0.5 text-xs hover:bg-slate-100";

export function HealthPanel({ deck, client, onDiscard }: { deck: Deck; client: AssistClient; onDiscard: (cardId: string) => void }) {
  const [duplicates, setDuplicates] = useState<HealthFlag[]>([]);
  const [dismissed, setDismissed] = useState<Set<string>>(new Set());
  const [error, setError] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const present = new Set(deck.cards.map((c) => c.id));
  const flags = [...pacingFlags(deck), ...duplicates.filter((f) => f.cardIds.every((id) => present.has(id)))].filter((f) => !dismissed.has(f.id));

  async function checkDuplicates() {
    setBusy(true);
    setError(null);
    try {
      const cards = cardsInPlayOrder(deck);
      if (cards.length < 2) return setDuplicates([]);
      const { vectors } = await client.embed(cards.map(cardText));
      setDuplicates(duplicateBeats(cards, vectors));
    } catch (e) {
      setError(e instanceof AssistError ? e.message : (e as Error).message);
    } finally {
      setBusy(false);
    }
  }

  return (
    <section aria-label="Deck health" className="space-y-2 rounded-lg border border-slate-300 bg-white p-3">
      <h3 className="text-sm font-semibold">Duplicate beats and pacing</h3>
      <button className={button} disabled={busy} onClick={() => void checkDuplicates()}>
        Check for duplicate beats
      </button>
      {error && <p role="alert" className="text-sm text-red-800">{error}</p>}
      {flags.length === 0 && <p className="text-sm">No flags.</p>}
      <ul className="space-y-1">
        {flags.map((f) => (
          <li key={f.id} className="flex flex-wrap items-center gap-2 rounded bg-rose-50 p-2 text-sm">
            <span className="flex-1">{f.message}</span>
            {f.kind === "duplicate-beat" && (
              <button className={ghost} onClick={() => onDiscard(f.cardIds[1])}>
                Discard the later card
              </button>
            )}
            <button className={ghost} onClick={() => setDismissed((s) => new Set(s).add(f.id))}>
              Keep as is
            </button>
          </li>
        ))}
      </ul>
    </section>
  );
}

import { pacing, playOrder, type Card, type Deck } from "./deck";
import { cosine } from "./notes-index";

// Pacing rules are plain arithmetic. Duplicate beats use embedding similarity,
// but the app sets the threshold and a person decides what to cut.

export type HealthFlag = { id: string; kind: "heavy-round" | "empty-round" | "long-card" | "duplicate-beat"; message: string; cardIds: string[] };

export const HEAVY_ROUND_SHARE = 0.5;
export const LONG_CARD_SECONDS = 90;
export const DUPLICATE_MIN_SCORE = 0.9;

export function pacingFlags(deck: Deck): HealthFlag[] {
  const flags: HealthFlag[] = [];
  const { rounds } = pacing(deck);
  for (const r of rounds) {
    if (r.cards === 0) flags.push({ id: `empty-${r.roundId}`, kind: "empty-round", message: `Round “${r.title}” has no cards.`, cardIds: [] });
    else if (rounds.length > 1 && r.share > HEAVY_ROUND_SHARE)
      flags.push({ id: `heavy-${r.roundId}`, kind: "heavy-round", message: `Round “${r.title}” takes ${Math.round(r.share * 100)}% of the match.`, cardIds: [] });
  }
  for (const c of deck.cards) {
    if (c.seconds > LONG_CARD_SECONDS) flags.push({ id: `long-${c.id}`, kind: "long-card", message: `“${c.title}” runs ${c.seconds}s; one card should carry one beat.`, cardIds: [c.id] });
  }
  return flags;
}

export function duplicateBeats(cards: Card[], vectors: number[][], minScore = DUPLICATE_MIN_SCORE): HealthFlag[] {
  if (vectors.length !== cards.length) throw new Error(`Expected ${cards.length} vectors, received ${vectors.length}`);
  const flags: HealthFlag[] = [];
  for (let i = 0; i < cards.length; i++) {
    for (let j = i + 1; j < cards.length; j++) {
      const score = cosine(vectors[i], vectors[j]);
      if (score >= minScore) {
        flags.push({
          id: `dup-${cards[i].id}-${cards[j].id}`,
          kind: "duplicate-beat",
          message: `“${cards[i].title}” and “${cards[j].title}” make the same point (similarity ${score.toFixed(2)}).`,
          cardIds: [cards[i].id, cards[j].id],
        });
      }
    }
  }
  return flags;
}

export function cardsInPlayOrder(deck: Deck): Card[] {
  return playOrder(deck);
}
