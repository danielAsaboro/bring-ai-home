We will let a local LLM draft a short message from notes you already saved. The draft is editable, can be copied or discarded, and Circle never sends it.

Sometimes you know what you want to say but not how to start. A local model can draft a few warm lines from the notes you saved.

The draft is optional and hidden behind a disclosure. It is labelled as not sent and shows which notes it was based on. There is no send button. You copy it, change it, or throw it away.

## Build and observe

### 1. Ask for a draft

Open Dev Raman, expand **Draft a follow-up (optional)**, type `good luck with the talk`, and select **Draft**.

### 2. Check it against the notes

The label shows the model and how many saved notes it used. The prompt tells the model not to add facts, plans, or dates. A small model can still drift. You are the check: read it before you copy it.

### 3. Edit, copy, or discard

Change the text. Select **Copy**, then send it yourself from your own messaging app if it still feels right. Or select **Discard**.

```javascript
import { mkdir, mkdtemp, rm, writeFile } from "node:fs/promises";
import { createServer } from "node:http";
import { tmpdir } from "node:os";
import path from "node:path";
import { fileURLToPath } from "node:url";

import { rankContext } from "../src/features/circle/context.ts";
import {
  buildExtractionPrompt,
  parseExtraction,
  PROPOSAL_SCHEMA,
} from "../src/features/circle/extraction.ts";
import { buildFollowUpPrompt, cleanDraft } from "../src/features/circle/followup.ts";
import { buildRecapDraft } from "./recap.ts";
import { createService, MAX_BODY_BYTES, SERVICE_PORT } from "./service.ts";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const qvacDirectory = path.join(appDirectory, ".qvac");
const cacheDirectory = path.join(qvacDirectory, "models");
const configPath = path.join(qvacDirectory, "config.json");
const HOST = "127.0.0.1";
const ALLOWED_ORIGINS = new Set(["http://localhost:3000", "http://127.0.0.1:3000"]);

await mkdir(cacheDirectory, { recursive: true });
await writeFile(configPath, `${JSON.stringify({ cacheDirectory }, null, 2)}\n`, "utf8");
process.env.QVAC_CONFIG_PATH = configPath;
const sdk = await import("@qvac/sdk");

// Models load on first use and stay loaded until the service stops.
const loaded = new Map();
function model(name, options) {
  if (!loaded.has(name)) {
    loaded.set(
      name,
      sdk.loadModel(options).catch((error) => {
        loaded.delete(name);
        throw error;
      }),
    );
  }
  return loaded.get(name);
}

const capabilities = {
  async transcribe(audio, fileName) {
    const modelId = await model("WHISPER_TINY", {
      modelSrc: sdk.WHISPER_TINY,
      modelConfig: { language: "en", translate: false, temperature: 0.0 },
    });
    const dir = await mkdtemp(path.join(tmpdir(), "circle-recap-"));
    const audioPath = path.join(dir, path.basename(fileName));
    try {
      await writeFile(audioPath, audio);
      const segments = await sdk.transcribe({ modelId, audioChunk: audioPath, metadata: true });
      return buildRecapDraft(segments, "WHISPER_TINY");
    } finally {
      await rm(dir, { recursive: true, force: true });
    }
  },
};

capabilities.extract = async ({ transcript, people, today }) => {
  const modelId = await model("LLAMA_3_2_1B_INST_Q4_0", {
    modelSrc: sdk.LLAMA_3_2_1B_INST_Q4_0,
    modelConfig: { ctx_size: 4096 },
  });
  const run = sdk.completion({
    modelId,
    history: [{ role: "user", content: buildExtractionPrompt(transcript, people, today) }],
    stream: false,
    responseFormat: {
      type: "json_schema",
      json_schema: { name: "circle_proposals", schema: PROPOSAL_SCHEMA },
    },
  });
  const final = await run.final;
  return parseExtraction(final.contentText, {
    transcript,
    people,
    today,
    model: "LLAMA_3_2_1B_INST_Q4_0",
  });
};

capabilities.retrieve = async ({ query, passages }) => {
  const modelId = await model("GTE_LARGE_FP16", {
    modelSrc: sdk.GTE_LARGE_FP16,
    modelType: "embeddings",
  });
  const { embedding } = await sdk.embed({
    modelId,
    text: [query, ...passages.map((p) => p.text)],
  });
  const [queryVector, ...vectors] = embedding;
  return {
    model: "GTE_LARGE_FP16",
    matches: rankContext(queryVector, passages, vectors, { limit: 3, minScore: 0.3 }),
  };
};

capabilities.draft = async ({ personName, intent, context }) => {
  const modelId = await model("LLAMA_3_2_1B_INST_Q4_0", {
    modelSrc: sdk.LLAMA_3_2_1B_INST_Q4_0,
    modelConfig: { ctx_size: 4096 },
  });
  const run = sdk.completion({
    modelId,
    history: [{ role: "user", content: buildFollowUpPrompt(personName, intent, context) }],
    stream: false,
  });
  const final = await run.final;
  return cleanDraft(final.contentText, "LLAMA_3_2_1B_INST_Q4_0", context);
};

const handle = createService(capabilities);

const server = createServer(async (req, res) => {
  const origin = req.headers.origin;
  if (origin && ALLOWED_ORIGINS.has(origin)) {
    res.setHeader("Access-Control-Allow-Origin", origin);
    res.setHeader("Access-Control-Allow-Headers", "content-type");
  }
  if (req.method === "OPTIONS") return res.writeHead(204).end();
  if (origin && !ALLOWED_ORIGINS.has(origin)) {
    return res.writeHead(403).end(JSON.stringify({ error: "origin-not-allowed" }));
  }
  let size = 0;
  const chunks = [];
  for await (const chunk of req) {
    size += chunk.length;
    if (size > MAX_BODY_BYTES) {
      return res.writeHead(413).end(JSON.stringify({ error: "too-large" }));
    }
    chunks.push(chunk);
  }
  let body;
  try {
    body = chunks.length ? JSON.parse(Buffer.concat(chunks).toString("utf8")) : undefined;
  } catch {
    return res.writeHead(400).end(JSON.stringify({ error: "invalid-json" }));
  }
  const url = new URL(req.url ?? "/", `http://${HOST}`);
  const result = await handle(req.method ?? "GET", url.pathname, body);
  res.writeHead(result.status, { "content-type": "application/json" });
  res.end(JSON.stringify(result.body));
});

server.listen(SERVICE_PORT, HOST, () => {
  process.stdout.write(`Circle QVAC service on http://${HOST}:${SERVICE_PORT}\n`);
});

async function shutdown() {
  server.close();
  for (const pending of loaded.values()) {
    const modelId = await pending.catch(() => undefined);
    if (modelId) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  }
  await sdk.close().catch(() => undefined);
  process.exit(0);
}
process.on("SIGINT", shutdown);
process.on("SIGTERM", shutdown);
```

```typescript
import type { ContextMatch, Passage } from "../src/features/circle/context";
import type { FollowUpDraft } from "../src/features/circle/followup";
import type { ExtractionResult, KnownPerson } from "../src/features/circle/extraction";
import type { RecapDraft } from "./recap";

// Everything the native runtime can do for Circle. A missing entry means
// "not configured", never "pretend".
export type Capabilities = {
  transcribe?: (audio: Uint8Array, fileName: string) => Promise<RecapDraft>;
  extract?: (input: {
    transcript: string;
    people: KnownPerson[];
    today: string;
  }) => Promise<ExtractionResult>;
  retrieve?: (input: {
    query: string;
    passages: Passage[];
  }) => Promise<{ model: string; matches: ContextMatch[] }>;
  draft?: (input: {
    personName: string;
    intent: string;
    context: Passage[];
  }) => Promise<FollowUpDraft>;
};

export type ServiceResponse = { status: number; body: unknown };

export const SERVICE_PORT = 4327;
export const MAX_BODY_BYTES = 12 * 1024 * 1024;

class BadRequest extends Error {}

function people(body: Record<string, unknown>): KnownPerson[] {
  const list = body.people;
  if (!Array.isArray(list) || list.length === 0 || list.length > 500) {
    throw new BadRequest("Missing people");
  }
  return list.map((p) => {
    const r = record(p);
    return { id: text(r, "id", 200), name: text(r, "name", 200) };
  });
}

function passages(body: Record<string, unknown>, key: string): Passage[] {
  const list = body[key];
  if (!Array.isArray(list) || list.length > 200) throw new BadRequest(`Missing ${key}`);
  return list.map((p) => {
    const r = record(p);
    return { id: text(r, "id", 200), text: text(r, "text", 2000) };
  });
}

function isoDay(body: Record<string, unknown>, key: string): string {
  const value = text(body, key, 10);
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) throw new BadRequest(`${key} must be YYYY-MM-DD`);
  return value;
}

function record(body: unknown): Record<string, unknown> {
  if (!body || typeof body !== "object" || Array.isArray(body)) {
    throw new BadRequest("Expected a JSON object");
  }
  return body as Record<string, unknown>;
}

function text(body: Record<string, unknown>, key: string, max = 20_000): string {
  const value = body[key];
  if (typeof value !== "string" || !value.trim()) {
    throw new BadRequest(`Missing ${key}`);
  }
  if (value.length > max) throw new BadRequest(`${key} is too long`);
  return value;
}

type Route = {
  capability: keyof Capabilities;
  run: (caps: Required<Capabilities>, body: Record<string, unknown>) => Promise<unknown>;
};

const routes: Record<string, Route> = {
  "/recap/transcribe": {
    capability: "transcribe",
    run: (caps, body) => {
      const audio = Buffer.from(text(body, "audioBase64", MAX_BODY_BYTES), "base64");
      if (audio.length === 0) throw new BadRequest("Empty recording");
      return caps.transcribe(new Uint8Array(audio), text(body, "fileName", 200));
    },
  },
  "/recap/extract": {
    capability: "extract",
    run: (caps, body) =>
      caps.extract({
        transcript: text(body, "transcript"),
        people: people(body),
        today: isoDay(body, "today"),
      }),
  },
  "/context/retrieve": {
    capability: "retrieve",
    run: (caps, body) => {
      const list = passages(body, "passages");
      if (list.length === 0) return Promise.resolve({ model: "none", matches: [] });
      return caps.retrieve({ query: text(body, "query", 500), passages: list });
    },
  },
  "/followup/draft": {
    capability: "draft",
    run: (caps, body) =>
      caps.draft({
        personName: text(body, "personName", 200),
        intent: text(body, "intent", 500),
        context: passages(body, "context"),
      }),
  },
};

export function createService(capabilities: Capabilities) {
  return async function handle(
    method: string,
    path: string,
    body: unknown,
  ): Promise<ServiceResponse> {
    if (method === "GET" && path === "/health") {
      return {
        status: 200,
        body: {
          ok: true,
          service: "circle-qvac",
          capabilities: Object.fromEntries(
            Object.keys(routes).map((r) => [
              routes[r].capability,
              typeof capabilities[routes[r].capability] === "function",
            ]),
          ),
        },
      };
    }
    const route = routes[path];
    if (!route) return { status: 404, body: { error: "not-found" } };
    if (method !== "POST") return { status: 405, body: { error: "method-not-allowed" } };
    if (typeof capabilities[route.capability] !== "function") {
      return {
        status: 503,
        body: { error: "not-configured", capability: route.capability },
      };
    }
    try {
      const result = await route.run(capabilities as Required<Capabilities>, record(body));
      return { status: 200, body: result };
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      if (error instanceof BadRequest) return { status: 400, body: { error: message } };
      return { status: 502, body: { error: "model-failed", message } };
    }
  };
}
```

```tsx
  state: CircleState;
```

```tsx
  const [intent, setIntent] = useState("");

              person.name,
              intent,
              passagesFor(person, state.reminders).slice(0, 5),

            setText(result.text);
          } catch (err) {
            setNote(err instanceof Error ? err.message : String(err));
          }
        }}
      >
        <input
          aria-label="What do you want to say?"
          className="flex-1 rounded border p-1"
          value={intent}
          onChange={(e) => setIntent(e.target.value)}

        </button>
      </form>

              rows={4}
              value={text}
              onChange={(e) => setText(e.target.value)}
            />
          </label>

              type="button"

              Discard
            </button>
          </div>
        </div>
      )}
    </details>
  );
}
```

```typescript
import type { Passage } from "./context";

export type FollowUpDraft = {
  status: "draft";
  model: string;
  text: string;
  basedOn: string[];
};

export const MAX_DRAFT_CHARS = 600;

export function buildFollowUpPrompt(
  personName: string,
  intent: string,
  context: Passage[],
): string {
  return [
    `Write a short, warm message I could send to ${personName}.`,
    `What I want to say: ${intent}`,
    "Use only these notes I saved. Do not add facts, plans, or dates that are not here.",
    ...context.map((c) => `- ${c.text}`),
    "Two to four sentences. Plain text. No subject line, no signature.",
  ].join("\n");
}

// The app, not the model, decides what a usable draft looks like.
export function cleanDraft(
  raw: string,
  model: string,
  context: Passage[],
): FollowUpDraft {
  const text = raw
    .replace(/^\s*subject:.*$/gim, "")
    .trim()
    .replace(/^["“]+|["”]+$/g, "")
    .trim();
  if (!text) throw new Error("The model returned an empty draft");
  return {
    status: "draft",
    model,
    text: text.length > MAX_DRAFT_CHARS ? `${text.slice(0, MAX_DRAFT_CHARS).trimEnd()}…` : text,
    basedOn: context.map((c) => c.id),
  };
}
```

```typescript
import type { ContextMatch, Passage } from "./context";
import type { FollowUpDraft } from "./followup";
import type { ExtractionResult, KnownPerson } from "./extraction";

// The browser never imports @qvac/sdk. It talks to the loopback service.
export const DEFAULT_SERVICE_URL = "http://127.0.0.1:4327";

export type ServiceHealth = {
  ok: boolean;
  capabilities: Record<string, boolean>;
};

export type ProposedRecap = {
  status: "proposed";
  model: string;
  transcript: string;
  durationMs: number;
};

export class ServiceError extends Error {
  constructor(
    message: string,
    readonly code: string,
  ) {
    super(message);
  }
}

type FetchLike = (input: string, init?: RequestInit) => Promise<Response>;

export function createServiceClient(
  baseUrl: string = DEFAULT_SERVICE_URL,
  fetchImpl: FetchLike = (input, init) => fetch(input, init),
) {
  async function call(path: string, init?: RequestInit): Promise<unknown> {
    let res: Response;
    try {
      res = await fetchImpl(`${baseUrl}${path}`, init);
    } catch {
      throw new ServiceError("The local AI service is not running", "unreachable");
    }
    const body = (await res.json().catch(() => ({}))) as Record<string, unknown>;
    if (!res.ok) {
      const code = typeof body.error === "string" ? body.error : `http-${res.status}`;
      const message = typeof body.message === "string" ? body.message : code;
      throw new ServiceError(message, code);
    }
    return body;
  }

  function post(path: string, payload: unknown) {
    return call(path, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(payload),
    });
  }

  return {
    async health(): Promise<ServiceHealth> {
      const body = (await call("/health")) as ServiceHealth;
      return { ok: body.ok === true, capabilities: body.capabilities ?? {} };
    },
    async transcribe(audioBase64: string, fileName: string): Promise<ProposedRecap> {
      const body = (await post("/recap/transcribe", { audioBase64, fileName })) as ProposedRecap;
      if (body.status !== "proposed" || typeof body.transcript !== "string") {
        throw new ServiceError("Unexpected transcription response", "bad-response");
      }
      return body;
    },
    async extract(
      transcript: string,
      people: KnownPerson[],
      today: string,
    ): Promise<ExtractionResult> {
      const body = (await post("/recap/extract", { transcript, people, today })) as ExtractionResult;
      if (body.status !== "proposed" || !Array.isArray(body.proposals)) {
        throw new ServiceError("Unexpected extraction response", "bad-response");
      }
      return body;
    },
    async retrieve(
      query: string,
      passages: Passage[],
    ): Promise<{ model: string; matches: ContextMatch[] }> {
      const body = (await post("/context/retrieve", { query, passages })) as {
        model: string;
        matches: ContextMatch[];
      };
      if (!Array.isArray(body.matches)) {
        throw new ServiceError("Unexpected retrieval response", "bad-response");
      }
      return body;
    },
    async draftFollowUp(
      personName: string,
      intent: string,
      context: Passage[],
    ): Promise<FollowUpDraft> {
      const body = (await post("/followup/draft", { personName, intent, context })) as FollowUpDraft;
      if (body.status !== "draft" || typeof body.text !== "string") {
        throw new ServiceError("Unexpected draft response", "bad-response");
      }
      return body;
    },
    post,
  };
}

export type ServiceClient = ReturnType<typeof createServiceClient>;
```
