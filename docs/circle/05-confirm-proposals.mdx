We will show extracted items as unticked proposals with their evidence.
A person can edit, tick, or ignore each one.
Only ticked items are saved, and plain code saves them.

This is the moment Circle is built for.
You speak a messy recap.
Circle shows what it found.
You keep only what is right.

Every proposal starts unticked.
The save button counts what you ticked and stays disabled at zero.
The model never writes to the book.

| File | Change |
| --- | --- |
| `src/features/circle/RecapPanel.tsx` | Upload or type a recap, find proposals, edit and tick, then save |

## Build and observe

### 1. Run both processes

```bash
npm run qvac:service
```

In another terminal:

```bash
npm run dev
```

### 2. Speak, then read

Choose your recording in **Voice recap**, or type a recap into the box.
Select **Find proposals**.
Each row shows the type, the person, editable text, a date for promises and dates, and `You said: “…”` with the evidence.

### 3. Confirm only what is right

Tick the promise.
Fix its wording if needed.
Leave a wrong item unticked.
Select **Save 1 confirmed**.
Open the person's card: the promise is now a gentle reminder.
The unticked item is nowhere.

### 4. Open what was set aside

Expand **set aside by the app** to see the items the checks in chapter 04 refused, with their reasons.

### 5. Turn the service off

Stop `qvac:service` and select **Find proposals** again.
You see `The local AI service is not running`.
Manual logging still works.

function PersonCard({
  person,
  state,
  onChange,
}: {
  person: Person;
  state: CircleState;
  onChange: (next: CircleState) => void;
}) {
  const [kind, setKind] = useState<EntryKind>("interaction");
  const [date, setDate] = useState(today());

            aria-label="Entry kind"

            aria-label="Entry date"

          value={text}

          Save to timeline

          e.preventDefault();

            () =>

        <input
          aria-label="Reminder"
          className="flex-1 rounded border p-1"
          value={reminderText}

          aria-label="Reminder date"

"use client";

import { useState } from "react";

import type { Proposal, Rejection } from "./extraction";
import type { ServiceClient } from "./service-client";
import { applyConfirmed } from "./state";
import type { CircleState } from "./types";

type Row = Proposal & { checked: boolean };

async function fileToBase64(file: File): Promise<string> {
  const bytes = new Uint8Array(await file.arrayBuffer());
  let binary = "";
  bytes.forEach((b) => (binary += String.fromCharCode(b)));
  return btoa(binary);
}

export function RecapPanel({
  state,
  onChange,
  client,
  today,
}: {
  state: CircleState;
  onChange: (next: CircleState) => void;
  client: Pick<ServiceClient, "transcribe" | "extract">;
  today: string;
}) {
  const [transcript, setTranscript] = useState("");
  const [rows, setRows] = useState<Row[]>([]);
  const [rejected, setRejected] = useState<Rejection[]>([]);
  const [model, setModel] = useState<string | null>(null);
  const [busy, setBusy] = useState<string | null>(null);
  const [message, setMessage] = useState<string | null>(null);

  async function run(label: string, task: () => Promise<void>) {
    setBusy(label);
    setMessage(null);
    try {
      await task();
    } catch (e) {
      setMessage(e instanceof Error ? e.message : String(e));
    } finally {
      setBusy(null);
    }
  }

  const checked = rows.filter((r) => r.checked);

  return (
    <section aria-label="Voice recap" className="my-4 rounded-lg border border-dashed border-amber-300 p-4">
      <h2 className="font-serif text-xl">Voice recap</h2>
      <p className="text-sm text-stone-600">
        Runs on this computer. Everything below is a proposal until you tick it and save.
      </p>
      <label className="my-2 block text-sm">
        Recording
        <input
          type="file"
          accept="audio/*"
          className="ml-2"
          onChange={(e) => {
            const file = e.target.files?.[0];
            if (!file) return;
            void run("Transcribing locally…", async () => {
              const draft = await client.transcribe(await fileToBase64(file), file.name);
              setTranscript(draft.transcript);
            });
          }}
        />
      </label>
      <textarea
        aria-label="Recap transcript"
        className="w-full rounded border p-2"
        rows={3}
        value={transcript}
        onChange={(e) => setTranscript(e.target.value)}
      />
      <button
        type="button"
        className="rounded bg-amber-200 px-3 py-1 disabled:opacity-50"
        disabled={!transcript.trim() || busy !== null}
        onClick={() =>
          run("Finding proposals…", async () => {
            const result = await client.extract(
              transcript,
              state.people.map((p) => ({ id: p.id, name: p.name })),
              today,
            );
            setRows(result.proposals.map((p) => ({ ...p, checked: false })));
            setRejected(result.rejected);
            setModel(result.model);
          })
        }
      >
        Find proposals
      </button>
      {busy && <p role="status">{busy}</p>}
      {message && <p role="alert">{message}</p>}

      {model && (
        <div className="mt-3">
          <h3 className="font-serif text-lg">Proposals from {model}</h3>
          {rows.length === 0 && <p>Nothing the app could verify.</p>}
          <ul className="space-y-2">
            {rows.map((row) => (
              <li key={row.id} className="rounded bg-amber-50 p-2 text-sm">
                <label className="flex items-center gap-2">
                  <input
                    type="checkbox"
                    checked={row.checked}
                    aria-label={`Keep ${row.type} for ${row.personName}`}
                    onChange={() =>
                      setRows(rows.map((r) => (r.id === row.id ? { ...r, checked: !r.checked } : r)))
                    }
                  />
                  <strong>{row.type}</strong> · {row.personName}
                </label>
                <input
                  aria-label={`Edit ${row.type} for ${row.personName}`}
                  className="mt-1 w-full rounded border p-1"
                  value={row.text}
                  onChange={(e) =>
                    setRows(rows.map((r) => (r.id === row.id ? { ...r, text: e.target.value } : r)))
                  }
                />
                {row.type !== "memory" && (
                  <input
                    aria-label={`Date for ${row.type} for ${row.personName}`}
                    type="date"
                    value={row.date ?? ""}
                    onChange={(e) =>
                      setRows(
                        rows.map((r) =>
                          r.id === row.id ? { ...r, date: e.target.value || undefined } : r,
                        ),
                      )
                    }
                  />
                )}
                <p className="text-stone-500">You said: “{row.evidence}”</p>
              </li>
            ))}
          </ul>
          {rejected.length > 0 && (
            <details className="text-sm">
              <summary>{rejected.length} set aside by the app</summary>
              <ul>
                {rejected.map((r, i) => (
                  <li key={i}>{r.reason}</li>
                ))}
              </ul>
            </details>
          )}
          <button
            type="button"
            className="mt-2 rounded bg-amber-300 px-3 py-1 disabled:opacity-50"
            disabled={checked.length === 0}
            onClick={() => {
              try {
                onChange(applyConfirmed(state, checked, today));
                setRows([]);
                setRejected([]);
                setModel(null);
                setTranscript("");
                setMessage(`Saved ${checked.length} confirmed item${checked.length === 1 ? "" : "s"}.`);
              } catch (e) {
                setMessage(e instanceof Error ? e.message : String(e));
              }
            }}
          >
            Save {checked.length} confirmed
          </button>
        </div>
      )}
    </section>
  );
}

import type { CircleState, EntryKind, Person, Reminder } from "./types";

let counter = 0;
export function newId(prefix: string): string {
  counter += 1;
  return `${prefix}-${Date.now().toString(36)}-${counter}`;
}

export function isIsoDate(value: string): boolean {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(value)) return false;
  const date = new Date(`${value}T00:00:00Z`);
  return !Number.isNaN(date.getTime()) && date.toISOString().startsWith(value);
}

export function letterOf(name: string): string {
  const first = name.normalize("NFD").replace(/[̀-ͯ]/g, "").charAt(0);
  return /[A-Za-z]/.test(first) ? first.toUpperCase() : "#";
}

export function peopleByLetter(people: Person[], letter: string): Person[] {
  return people
    .filter((person) => letterOf(person.name) === letter)
    .sort((a, b) => a.name.localeCompare(b.name));
}

function updatePerson(
  state: CircleState,
  personId: string,
  update: (person: Person) => Person,
): CircleState {
  if (!state.people.some((p) => p.id === personId)) {
    throw new Error(`Unknown person: ${personId}`);
  }
  return {
    ...state,
    people: state.people.map((p) => (p.id === personId ? update(p) : p)),
  };
}

export function logEntry(
  state: CircleState,
  personId: string,
  entry: { kind: EntryKind; date: string; text: string },
): CircleState {
  const text = entry.text.trim();
  if (!text) throw new Error("Write something before saving");
  if (!isIsoDate(entry.date)) throw new Error("Use a real date");
  return updatePerson(state, personId, (person) => ({
    ...person,
    timeline: [
      { id: newId("t"), kind: entry.kind, date: entry.date, text },
      ...person.timeline,
    ].sort((a, b) => b.date.localeCompare(a.date)),
  }));
}

export function addReminder(
  state: CircleState,
  reminder: Omit<Reminder, "id" | "done">,
): CircleState {
  const text = reminder.text.trim();
  if (!text) throw new Error("Write the reminder first");
  if (!isIsoDate(reminder.due)) throw new Error("Use a real date");
  if (!state.people.some((p) => p.id === reminder.personId)) {
    throw new Error(`Unknown person: ${reminder.personId}`);
  }
  return {
    ...state,
    reminders: [
      ...state.reminders,
      { ...reminder, text, id: newId("r"), done: false },
    ].sort((a, b) => a.due.localeCompare(b.due)),
  };
}

export function completeReminder(state: CircleState, id: string): CircleState {
  return {
    ...state,
    reminders: state.reminders.map((r) => (r.id === id ? { ...r, done: true } : r)),
  };
}

export function toggleCircle(
  state: CircleState,
  personId: string,
  circle: string,
): CircleState {
  return updatePerson(state, personId, (person) => ({
    ...person,
    circles: person.circles.includes(circle)
      ? person.circles.filter((c) => c !== circle)
      : [...person.circles, circle],
  }));
}

export function openReminders(state: CircleState, personId?: string): Reminder[] {
  return state.reminders.filter(
    (r) => !r.done && (personId === undefined || r.personId === personId),
  );
}

export type ConfirmedProposal = {
  type: "memory" | "promise" | "date";
  personId: string;
  text: string;
  date?: string;
};

// Only proposals a person ticked reach this function. It is plain code:
// the model never writes to the book.
export function applyConfirmed(
  state: CircleState,
  confirmed: ConfirmedProposal[],
  recapDate: string,
): CircleState {
  return confirmed.reduce((next, item) => {
    if (item.type === "memory") {
      return logEntry(next, item.personId, { kind: "memory", date: recapDate, text: item.text });
    }
    if (item.type === "promise") {
      return addReminder(next, {
        personId: item.personId,
        due: item.date ?? recapDate,
        text: `Promised: ${item.text.trim()}`,
      });
    }
    if (!item.date) throw new Error("An important date needs a date");
    return addReminder(
      logEntry(next, item.personId, {
        kind: "memory",
        date: recapDate,
        text: `Important date ${item.date}: ${item.text.trim()}`,
      }),
      { personId: item.personId, due: item.date, text: item.text },
    );
  }, state);
}
