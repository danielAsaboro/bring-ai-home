## Build and observe

### 1. Warm the cache

With the network on, run each feature once from the app, or run the proof once.
This downloads every model into `.qvac/models`.

### 2. Turn the network off

Switch off Wi-Fi and unplug any cable.
Restart the service:

```bash
npm run qvac:service
```

### 3. Run the proof

```bash
npm run qvac:offline-proof -- ~/Downloads/dev-recap.wav
```

The script first checks whether it can resolve `registry.npmjs.org`.
If it can, the verdict is `not-proven: the network was reachable during this run`.
With the network off and warm models, each route should answer `200`, and the verdict is `offline: every route answered with the network off`.

### 4. Use the app offline

Open `http://localhost:3000`.
Speak a recap, confirm a proposal, gather context, and draft a follow-up.
Your book, your voice, and the people in it stay on this computer.

<Accordion title="qvac-service/offline-proof.mjs">

```javascript
import { lookup } from "node:dns/promises";

import { SERVICE_PORT } from "./service.ts";
import { summarizeProof } from "./offline.ts";

// Run with Wi-Fi off and `npm run qvac:service` already running.
const base = process.env.CIRCLE_SERVICE_URL ?? `http://127.0.0.1:${SERVICE_PORT}`;
const today = new Date().toISOString().slice(0, 10);
const recap =
  "Coffee with Dev today. He is nervous about the meetup. I said I would send him my slide template by Friday.";

let networkReachable = true;
try {
  await lookup("registry.npmjs.org");
} catch {
  networkReachable = false;
}

// A cold model load can take a while; a step that never answers is reported, not hidden.
const timeoutMs = Number(process.env.CIRCLE_PROOF_TIMEOUT_MS ?? 180000);

async function step(route, payload) {
  const startedAt = Date.now();
  try {
    const res = await fetch(`${base}${route}`, {
      signal: AbortSignal.timeout(timeoutMs),
      ...(payload
        ? { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(payload) }
        : {}),
    });
    const body = await res.json().catch(() => ({}));
    return { route, status: res.status, ms: Date.now() - startedAt, model: body.model, body };
  } catch (error) {
    const reason = error instanceof Error ? error.name : String(error);
    return { route, status: 0, ms: Date.now() - startedAt, body: { error: reason } };
  }
}

const people = [{ id: "dev-raman", name: "Dev Raman" }];
const notes = [{ id: "t5", text: "2026-10-10 interaction: Practised the first five minutes of his talk." }];
const steps = [
  await step("/health"),
  await step("/recap/extract", { transcript: recap, people, today }),
  await step("/context/retrieve", { query: "his talk", passages: notes }),
  await step("/followup/draft", { personName: "Dev Raman", intent: "good luck", context: notes }),
];
if (process.argv[2]) {
  const { readFile } = await import("node:fs/promises");
  const audio = await readFile(process.argv[2]);
  steps.push(
    await step("/recap/transcribe", { audioBase64: audio.toString("base64"), fileName: "recap.wav" }),
  );
}
const receipt = summarizeProof(networkReachable, steps);
process.stdout.write(`${JSON.stringify(receipt, null, 2)}\n`);
if (!receipt.verdict.startsWith("offline")) process.exitCode = 1;
```

```typescript
import { readdirSync, readFileSync, statSync } from "node:fs";
import path from "node:path";

import { describe, expect, it } from "vitest";

import { DEFAULT_SERVICE_URL } from "../src/features/circle/service-client";
import { isLoopback, summarizeProof } from "./offline";

function files(dir: string): string[] {
  return readdirSync(dir).flatMap((name) => {
    const full = path.join(dir, name);
    return statSync(full).isDirectory() ? files(full) : [full];
  });
}

describe("offline proof", () => {
  it("points the browser at loopback only", () => {
    expect(isLoopback(DEFAULT_SERVICE_URL)).toBe(true);
    expect(isLoopback("https://api.example.com")).toBe(false);
  });

  it("ships no remote URLs in app or service code", () => {
    const roots = [path.resolve(__dirname, "../src"), __dirname];
    const remote = roots
      .flatMap(files)
      .filter((f) => /\.(ts|tsx|mjs)$/.test(f) && !f.endsWith(".test.ts"))
      .flatMap((f) =>
        (readFileSync(f, "utf8").match(/https?:\/\/[^\s"'`)]+/g) ?? [])
          .filter((u) => !u.includes("${") && !isLoopback(u))
          .map((u) => `${path.basename(f)}: ${u}`),
      );
    expect(remote).toEqual([]);
  });

  it("refuses to call a run offline while the network is up", () => {
    const steps = [{ route: "/health", status: 200, ms: 3 }];
    expect(summarizeProof(true, steps).verdict).toMatch(/^not-proven/);
    expect(summarizeProof(false, steps).verdict).toMatch(/^offline/);
    expect(summarizeProof(false, [{ route: "/recap/extract", status: 502, ms: 9 }]).verdict).toBe(
      "failed: /recap/extract 502",
    );
  });
});
```

```typescript
export type ProofStep = {
  route: string;
  status: number;
  ms: number;
  model?: string;
};

export function isLoopback(url: string): boolean {
  const host = new URL(url).hostname;
  return host === "127.0.0.1" || host === "localhost" || host === "[::1]";
}

// A proof only counts when the network was really off and every route answered.
export function summarizeProof(networkReachable: boolean, steps: ProofStep[]) {
  const failed = steps.filter((s) => s.status !== 200);
  const verdict = networkReachable
    ? "not-proven: the network was reachable during this run"
    : failed.length > 0
      ? `failed: ${failed.map((s) => `${s.route} ${s.status}`).join(", ")}`
      : "offline: every route answered with the network off";
  return { networkReachable, verdict, steps };
}
```
