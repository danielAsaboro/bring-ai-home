Generated deck text is often generic or unsupported.
Here the model may only propose.
The app checks every proposal against the note sections it was given, and a person decides what enters the deck.

## What you change

| File | Change |
| --- | --- |
| `src/features/deck/claim-proposals.ts` | JSON schema, prompt with bracketed source ids, and `parseClaimProposals`: drops cards with no valid citation, generic titles, duplicates, and over-long text, and records why |

| `qvac-service/serve.mjs` | Loads `LLAMA_3_2_1B_INST_Q4_0`; retrieves first, skips the model when nothing matched, then completes with `json_schema` and reads `(await run.final).contentText` |
| `src/features/deck/ProposalPanel.tsx` | Topic, target round, cited proposals, **Accept** and **Reject**, and a list of what the app removed |
| `src/features/deck/assist-client.ts`, `DeckApp.tsx` | `propose()` and the accept path that adds a claim card with its source |

## Build and observe

### 1. Ask for proposals

With the service running, type `Why buyers should stock a kettle that shows temperature` and choose **The problem**.
Press **Propose cards**.

Each proposal shows the note sections it cites, quoted.
Under them, **proposal(s) removed by the app** lists anything dropped and why, for example `cites no retrieved note section`.

### 2. Accept one

Press **Accept** on one card.
It joins the target round as a claim card whose **Source** names the note and section.
Rejected or ignored proposals never touch the deck.

### 3. Why the schema is not enough

`json_schema` constrains shape, not truth.
A citation can be well-formed and still point at an id that was never retrieved.
`parseClaimProposals` checks every id against the passages sent in the prompt.
The test uses a hand-written model answer with an invented id, an empty citation list, and a generic `Conclusion` title to show each rule.

```javascript
import path from "node:path";
import { fileURLToPath } from "node:url";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const qvacDirectory = path.join(appDirectory, ".qvac");
const cacheDirectory = path.join(qvacDirectory, "models");
const configPath = path.join(qvacDirectory, "config.json");

const { createDeckServer, SERVICE_PORT } = await import("./server.ts");

await mkdir(cacheDirectory, { recursive: true });
await writeFile(configPath, `${JSON.stringify({ cacheDirectory }, null, 2)}\n`, "utf8");
process.env.QVAC_CONFIG_PATH = configPath;

const loaded = [];
let sdk = null;
try {
  sdk = await import("@qvac/sdk");
} catch (error) {
  process.stderr.write(`QVAC SDK unavailable: ${error instanceof Error ? error.message : String(error)}\n`);
}

// Load one model, or report why not and leave that capability off.
async function tryLoad(label, options) {
  if (!sdk) return null;
  try {
    const modelId = await sdk.loadModel(options);
    loaded.push(modelId);
    return modelId;
  } catch (error) {
    process.stderr.write(`${label} unavailable: ${error instanceof Error ? error.message : String(error)}\n`);
    return null;
  }
}

// Listen first. Each capability switches on when its model has loaded, so the
// deck gets a clear 503 instead of a hanging request while models download.
const capabilities = { retrieve: null, propose: null };
const server = createDeckServer(capabilities);
server.listen(SERVICE_PORT, "127.0.0.1", () => {
  process.stderr.write(`Deck QVAC service on http://127.0.0.1:${SERVICE_PORT}\n`);
});

const index = await readFile(path.join(qvacDirectory, "notes-index.json"), "utf8")
  .then((text) => JSON.parse(text))
  .catch(() => {
    process.stderr.write("No notes index yet. Run npm run qvac:ingest-notes first.\n");
    return null;
  });
const embedId = await tryLoad("Embeddings", { modelSrc: sdk?.GTE_LARGE_FP16, modelType: "embeddings" });
const embedOne = async (text) => (await sdk.embed({ modelId: embedId, text })).embedding;

if (index && embedId) {
  const { searchNotes } = await import("../src/features/deck/notes-index.ts");
  capabilities.retrieve = async (query) => ({
    passages: searchNotes(index, await embedOne(query), { limit: 3, minScore: 0.3 }),
    model: "GTE_LARGE_FP16",
  });
  process.stderr.write("Note retrieval ready\n");
}

const llmId = await tryLoad("Text model", { modelSrc: sdk?.LLAMA_3_2_1B_INST_Q4_0, modelType: "llm" });
async function completeJson(history, name, schema) {
  const run = sdk.completion({
    modelId: llmId,
    history,
    stream: true,
    responseFormat: { type: "json_schema", json_schema: { name, schema } },
  });
  return (await run.final).contentText;
}
if (index && embedId && llmId) {
  const { searchNotes } = await import("../src/features/deck/notes-index.ts");
  const { buildProposalHistory, PROPOSAL_JSON_SCHEMA } = await import("../src/features/deck/claim-proposals.ts");
  capabilities.propose = async (request) => {
    const passages = searchNotes(index, await embedOne(request.topic), { limit: 4, minScore: 0.3 });
    // No retrieved sections means nothing to cite, so the model is not asked.
    if (passages.length === 0) return { text: "", passages, model: "LLAMA_3_2_1B_INST_Q4_0" };
    const text = await completeJson(buildProposalHistory(request, passages), "claim_cards", PROPOSAL_JSON_SCHEMA);
    return { text, passages, model: "LLAMA_3_2_1B_INST_Q4_0 + GTE_LARGE_FP16" };
  };
  process.stderr.write("Card proposals ready\n");
}

async function shutdown() {
  server.close();
  for (const modelId of loaded) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  if (sdk) await sdk.close().catch(() => undefined);
  process.exit(0);
}
process.on("SIGINT", shutdown);
process.on("SIGTERM", shutdown);
```

```typescript
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";

import {
  parseClaimProposals,
  parseProposeRequest,
  type ProposeRequest,
} from "../src/features/deck/claim-proposals.ts";
import type { ScoredChunk } from "../src/features/deck/notes-index.ts";

export const SERVICE_PORT = 4321;
const MAX_BODY_BYTES = 256 * 1024;

export type NotesRetriever = (query: string) => Promise<{ passages: ScoredChunk[]; model: string }>;

// Retrieves sections, then asks the LLM. Returns raw model text for the app to validate.
export type ClaimProposer = (request: ProposeRequest) => Promise<{ text: string; passages: ScoredChunk[]; model: string }>;

export type Capabilities = {
  retrieve: NotesRetriever | null;
  propose: ClaimProposer | null;
};

export class Unavailable extends Error {}
export class BadRequest extends Error {}

export function isAllowedOrigin(origin: string | undefined): boolean {
  if (!origin) return true; // curl and other local tools send no Origin
  try {
    const url = new URL(origin);
    return url.protocol === "http:" && (url.hostname === "localhost" || url.hostname === "127.0.0.1");
  } catch {
    return false;
  }
}

export function parseQuery(value: unknown): string {
  const v = value as { query?: unknown } | null;
  if (!v || typeof v.query !== "string" || !v.query.trim() || v.query.length > 300) {
    throw new BadRequest("Request must contain a query of at most 300 characters");
  }
  return v.query.trim();
}

function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  res.writeHead(status, {
    "content-type": "application/json",
    ...(origin ? { "access-control-allow-origin": origin, vary: "Origin" } : {}),
    "access-control-allow-headers": "content-type",
    "access-control-allow-methods": "GET, POST, OPTIONS",
  });
  res.end(status === 204 ? undefined : JSON.stringify(body));
}

async function readJson(req: IncomingMessage): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > MAX_BODY_BYTES) throw new BadRequest("Request is too large");
    chunks.push(chunk as Buffer);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString("utf8"));
  } catch {
    throw new BadRequest("Request body is not JSON");
  }
}

type Route = (body: unknown) => Promise<unknown>;

// The service proposes. It never writes to a deck. The browser decides.
export function createDeckServer(capabilities: Capabilities) {
  const routes: Record<string, Route> = {
    "/retrieve": async (body) => {
      const query = parseQuery(body);
      if (!capabilities.retrieve) throw new Unavailable("Note retrieval is not available");
      return { query, ...(await capabilities.retrieve(query)) };
    },
    "/propose": async (body) => {
      let request: ProposeRequest;
      try {
        request = parseProposeRequest(body);
      } catch (error) {
        throw new BadRequest((error as Error).message);
      }
      if (!capabilities.propose) throw new Unavailable("Card proposals are not available");
      const { text, passages, model } = await capabilities.propose(request);
      if (passages.length === 0) return { model, passages, proposals: [], dropped: [] };
      return { model, passages, ...parseClaimProposals(text, passages) };
    },
  };

  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (!isAllowedOrigin(origin)) return send(res, 403, { error: "Origin not allowed" });
    if (req.method === "OPTIONS") return send(res, 204, null, origin);
    if (req.method === "GET" && req.url === "/health") {
      const available = Object.fromEntries(Object.entries(capabilities).map(([k, v]) => [k, v !== null]));
      return send(res, 200, { ok: true, capabilities: available }, origin);
    }
    const route = req.method === "POST" && req.url ? routes[req.url] : undefined;
    if (!route) return send(res, 404, { error: "Not found" }, origin);
    try {
      return send(res, 200, await route(await readJson(req)), origin);
    } catch (error) {
      if (error instanceof BadRequest) return send(res, 400, { error: error.message }, origin);
      if (error instanceof Unavailable) return send(res, 503, { error: error.message }, origin);
      return send(res, 502, { error: (error as Error).message }, origin);
    }
  });
}
```

```tsx
function Match({ deck, onExit }: { deck: Deck; onExit: () => void }) {
  const cards = playOrder(deck);
  const [index, setIndex] = useState(0);
  const card = cards[Math.min(index, cards.length - 1)];
  const round = deck.rounds.find((r) => r.id === card?.roundId);
```

```tsx
import { AssistError, type AssistClient } from "./assist-client";
import { sourceLabel, type ClaimProposal } from "./claim-proposals";
import type { Deck } from "./deck";

const field = "rounded border border-slate-400 bg-white px-2 py-1 text-sm text-slate-900";
const button = "rounded bg-indigo-800 px-3 py-1 text-sm font-medium text-white hover:bg-indigo-900 disabled:opacity-50";

export type AcceptedClaim = { title: string; body: string; source: string; roundId: string };

// The model proposes, the app has already validated citations, and only the
// Accept button adds a card.
export function ProposalPanel({ deck, client, onAccept }: { deck: Deck; client: AssistClient; onAccept: (card: AcceptedClaim) => void }) {
  const [topic, setTopic] = useState("");
  const [roundId, setRoundId] = useState(deck.rounds[0]?.id ?? "");
  const [proposals, setProposals] = useState<ClaimProposal[]>([]);
  const [dropped, setDropped] = useState<{ title: string; reason: string }[]>([]);
  const [error, setError] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const targetRound = deck.rounds.some((r) => r.id === roundId) ? roundId : deck.rounds[0]?.id;

  async function propose() {
    setBusy(true);
    setError(null);
    try {
      const result = await client.propose(topic, deck.audience);
      setProposals(result.proposals);
      setDropped(result.dropped);
      if (result.passages.length === 0) setError("No note section matched, so nothing was proposed.");
    } catch (e) {
      setError(e instanceof AssistError ? e.message : "Proposal failed");
    } finally {
      setBusy(false);
    }
  }

  return (
    <section aria-label="Proposed claim cards" className="space-y-2 rounded-lg border border-slate-300 bg-white p-3">
      <h3 className="text-sm font-semibold">Propose cited claim cards</h3>
      <form
        className="flex flex-wrap gap-1"
        onSubmit={(e) => {
          e.preventDefault();
          if (topic.trim()) void propose();
        }}
      >
        <input className={`${field} min-w-0 flex-1`} aria-label="Card topic" placeholder="What should this round argue?" value={topic} onChange={(e) => setTopic(e.target.value)} />

          {deck.rounds.map((r) => (
            <option key={r.id} value={r.id}>
              {r.title}
            </option>
          ))}
        </select>
        <button className={button} disabled={busy}>
          Propose cards
        </button>
      </form>
      {error && <p role="alert" className="text-sm text-red-800">{error}</p>}
      <ul className="space-y-2">
        {proposals.map((p) => (
          <li key={p.id} className="rounded border border-dashed border-indigo-400 p-2 text-sm">
            <p className="font-medium">{p.title}</p>
            <p>{p.body}</p>
            {p.citations.map((c) => (
              <blockquote key={c.id} className="mt-1 border-l-2 border-slate-300 pl-2 text-xs text-slate-600">
                <cite className="block not-italic">{c.noteTitle} · {c.section}</cite>
                {c.text}
              </blockquote>
            ))}
            <span className="mt-1 flex gap-1">
              <button
                className={ghost}
                onClick={() => {

              <li key={i}>
                {d.title}: {d.reason}
              </li>
            ))}
          </ul>
        </details>
      )}
    </section>
  );
}
```

```typescript
import type { ClaimProposal } from "./claim-proposals";
import type { ScoredChunk } from "./notes-index";

export const SERVICE_URL = "http://127.0.0.1:4321";

export type AssistErrorKind = "service-stopped" | "unavailable" | "rejected" | "failed";

export class AssistError extends Error {
  constructor(
    readonly kind: AssistErrorKind,
    message: string,
  ) {
    super(message);
  }
}

export type RetrieveResult = { query: string; model: string; passages: ScoredChunk[] };

export type ProposeResult = {
  model: string;
  passages: ScoredChunk[];
  proposals: ClaimProposal[];
  dropped: { title: string; reason: string }[];
};

export type AssistClient = {
  retrieve(query: string): Promise<RetrieveResult>;
  propose(topic: string, audience: string): Promise<ProposeResult>;
};

// Every failure becomes one of four kinds, so the UI can say what went wrong
// and the deck stays exactly as it was.
export function createAssistClient(baseUrl = SERVICE_URL, fetchImpl: typeof fetch = (...a) => fetch(...a)): AssistClient {
  async function post<T>(path: string, body: unknown): Promise<T> {
    let res: Response;
    try {
      res = await fetchImpl(`${baseUrl}${path}`, {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify(body),
      });
    } catch {
      throw new AssistError("service-stopped", "The local QVAC service is not running");
    }
    const payload = (await res.json().catch(() => ({}))) as { error?: string };
    if (res.status === 503) throw new AssistError("unavailable", payload.error ?? "The model is not loaded yet");
    if (res.status === 400) throw new AssistError("rejected", payload.error ?? "The request was rejected");
    if (!res.ok) throw new AssistError("failed", payload.error ?? `The service answered ${res.status}`);
    return payload as T;
  }

  return {
    retrieve: (query) => post<RetrieveResult>("/retrieve", { query }),
    propose: (topic, audience) => post<ProposeResult>("/propose", { topic, audience }),
  };
}
```

```typescript
import type { ScoredChunk } from "./notes-index";

// QVAC may propose claim cards from retrieved note sections. A proposal enters
// the deck only when it cites a retrieved section and a person accepts it.

export type ClaimProposal = {
  id: string;
  title: string;
  body: string;
  citations: { id: string; noteTitle: string; section: string; text: string }[];
};

export type ProposalResult = { proposals: ClaimProposal[]; dropped: { title: string; reason: string }[] };

export type ProposeRequest = { topic: string; audience: string };

export const PROPOSAL_JSON_SCHEMA = {
  type: "object",
  properties: {
    cards: {
      type: "array",
      maxItems: 4,
      items: {
        type: "object",
        properties: {
          title: { type: "string" },
          body: { type: "string" },
          citations: { type: "array", items: { type: "string" } },
        },
        required: ["title", "body", "citations"],
      },
    },
  },
  required: ["cards"],
} as const;

export function parseProposeRequest(value: unknown): ProposeRequest {
  const v = value as { topic?: unknown; audience?: unknown } | null;
  if (!v || typeof v.topic !== "string" || !v.topic.trim() || v.topic.length > 300) {
    throw new Error("Request must contain a topic of at most 300 characters");
  }
  return { topic: v.topic.trim(), audience: typeof v.audience === "string" ? v.audience.slice(0, 200) : "" };
}

export function buildProposalHistory(request: ProposeRequest, passages: ScoredChunk[]) {
  const sources = passages.map((p) => `[${p.id}] ${p.noteTitle} — ${p.section}: ${p.text}`).join("\n");
  return [
    {
      role: "system",
      content:
        "You propose presentation claim cards. Use only the sources. Each card has a short title, one sentence body, and citations listing the bracketed source ids it relies on. Do not invent numbers.",
    },
    {
      role: "user",
      content: `Audience: ${request.audience || "general"}\nTopic: ${request.topic}\n\nSources:\n${sources}\n\nReturn JSON with up to 3 cards.`,
    },
  ];
}

const GENERIC = /^(introduction|conclusion|overview|summary|thank you|questions)$/i;

// The app decides: shape, length, citations, and genericness are checked here.
export function parseClaimProposals(text: string, passages: ScoredChunk[]): ProposalResult {
  let data: unknown;
  try {
    data = JSON.parse(text);
  } catch {
    throw new Error("The model did not return JSON");
  }
  const cards = (data as { cards?: unknown })?.cards;
  if (!Array.isArray(cards)) throw new Error("The model response has no cards array");
  const byId = new Map(passages.map((p) => [p.id, p]));
  const result: ProposalResult = { proposals: [], dropped: [] };
  const seen = new Set<string>();
  cards.forEach((raw, i) => {
    const c = raw as { title?: unknown; body?: unknown; citations?: unknown };
    const title = typeof c.title === "string" ? c.title.trim() : "";
    const body = typeof c.body === "string" ? c.body.trim() : "";
    if (!title || !body) return result.dropped.push({ title: title || `card ${i + 1}`, reason: "missing title or body" });
    if (title.length > 80 || body.length > 240) return result.dropped.push({ title, reason: "too long for one card" });
    if (GENERIC.test(title)) return result.dropped.push({ title, reason: "generic title" });
    if (seen.has(title.toLowerCase())) return result.dropped.push({ title, reason: "duplicate title" });
    const ids = Array.isArray(c.citations) ? c.citations.filter((x): x is string => typeof x === "string") : [];
    const cited = [...new Set(ids)].map((id) => byId.get(id.replace(/^\[|\]$/g, ""))).filter((p) => p !== undefined);
    if (cited.length === 0) return result.dropped.push({ title, reason: "cites no retrieved note section" });
    seen.add(title.toLowerCase());
    result.proposals.push({
      id: `proposal-${i}-${cited.map((p) => p.id).join("+")}`,
      title,
      body,
      citations: cited.map(({ id, noteTitle, section, text }) => ({ id, noteTitle, section, text })),
    });
  });
  return result;
}

export function sourceLabel(proposal: ClaimProposal): string {
  return proposal.citations.map((c) => `${c.noteTitle} · ${c.section}`).join("; ");
}
```
