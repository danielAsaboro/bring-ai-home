The facts behind a launch deck live in private notes: a strategy memo, a diary study, a trial log.
They are exactly what you should not paste into a public deck generator.

This chapter adds QVAC for the first time, in a standalone script.
It splits each note into sections, embeds them with `GTE_LARGE_FP16` on your machine, and writes a small index.
Nothing in the app uses it yet.

## Build and observe

### 1. Install the pinned SDK

```bash
npm install
```

On Linux, two quirks: keep `b4a` as a direct dependency (the SDK expects to resolve it), and make the bundled runtime executable once:

```bash
chmod +x node_modules/bare-runtime-linux-x64/bin/bare
```

### 2. Look at the sections

Open `public/test-files/deck/diary-study-notes.md`.
Each `##` heading becomes one chunk with a stable id such as `diary-study-notes#unattended-boiling`.
That id is what every later citation points to.

### 3. Ingest the notes

```bash
npm run qvac:ingest-notes
```

The first run downloads `GTE_LARGE_FP16` into `.qvac/models`.
It then prints the model, the vector dimensions, `notes: 3`, `chunks: 9`, the elapsed time, and the index path.
Record those numbers from your own machine.

### 4. Know which boundary failed

If the command sits at download, that is a network problem, not a code problem.
If it fails with a vector count message, the embedder returned the wrong batch size and `buildNotesIndex` refused to write a misaligned index.

```javascript
import { mkdir, readFile, writeFile } from "node:fs/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const qvacDirectory = path.join(appDirectory, ".qvac");
const cacheDirectory = path.join(qvacDirectory, "models");
const configPath = path.join(qvacDirectory, "config.json");
const indexPath = path.join(qvacDirectory, "notes-index.json");
const notesDirectory = path.join(appDirectory, "public/test-files/deck");

let sdk;
let modelId;
try {
  const { SAMPLE_NOTES, buildNotesIndex, chunkNote } = await import("../src/features/deck/notes-index.ts");
  const chunks = [];
  for (const note of SAMPLE_NOTES) {
    chunks.push(...chunkNote(note, await readFile(path.join(notesDirectory, note.file), "utf8")));
  }

  await mkdir(cacheDirectory, { recursive: true });
  await writeFile(configPath, `${JSON.stringify({ cacheDirectory }, null, 2)}\n`, "utf8");
  process.env.QVAC_CONFIG_PATH = configPath;
  sdk = await import("@qvac/sdk");
  modelId = await sdk.loadModel({
    modelSrc: sdk.GTE_LARGE_FP16,
    modelType: "embeddings",
    onProgress: (p) => process.stderr.write(`Downloading ${p.percentage.toFixed(0)}%\n`),
  });
  const startedAt = Date.now();
  const index = await buildNotesIndex(chunks, "GTE_LARGE_FP16", async (texts) => {
    const { embedding } = await sdk.embed({ modelId, text: texts });
    return embedding;
  });
  await writeFile(indexPath, `${JSON.stringify(index)}\n`, "utf8");
  process.stdout.write(
    `${JSON.stringify(
      { model: index.model, dimensions: index.dimensions, notes: SAMPLE_NOTES.length, chunks: index.entries.length, elapsedMs: Date.now() - startedAt, indexPath },
      null,
      2,
    )}\n`,
  );
} catch (error) {
  process.stderr.write(`QVAC notes ingest failed: ${error instanceof Error ? error.message : String(error)}\n`);
  process.exitCode = 1;
} finally {
  if (sdk && modelId) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  if (sdk) await sdk.close().catch(() => undefined);
}
```

import { readFileSync } from "node:fs";
import path from "node:path";

import { describe, expect, it } from "vitest";

import { buildNotesIndex, chunkNote, SAMPLE_NOTES, searchNotes } from "./notes-index";

const dir = path.join(process.cwd(), "public/test-files/deck");
const chunks = SAMPLE_NOTES.flatMap((note) => chunkNote(note, readFileSync(path.join(dir, note.file), "utf8")));

// FIXTURE vectors: hand-written, three dimensions, not model output.
// Dimension 0 = retail, 1 = diary study, 2 = prototype.
function fixtureEmbed(texts: string[]): Promise<number[][]> {
  return Promise.resolve(
    texts.map((t) => [t.includes("strategy memo") ? 1 : 0, t.includes("diary study") ? 1 : 0, t.includes("Prototype") ? 1 : 0]),
  );
}

describe("notes index", () => {
  it("chunks each sample note by section with stable ids", () => {
    expect(chunks).toHaveLength(9);
    expect(chunks[0]).toMatchObject({ id: "lumen-strategy-memo#retail-position", section: "Retail position" });
    expect(chunks.map((c) => c.id)).toContain("diary-study-notes#unattended-boiling");
  });

  it("returns only chunks above the score floor, best first", async () => {
    const index = await buildNotesIndex(chunks, "fixture", fixtureEmbed);
    const hits = searchNotes(index, [0, 1, 0.1], { limit: 3, minScore: 0.5 });
    expect(hits.map((h) => h.noteId)).toEqual(["diary-study-notes", "diary-study-notes", "diary-study-notes"]);
    expect(searchNotes(index, [0, 0, 0], { minScore: 0.1 })).toEqual([]);
  });

  it("rejects a mismatched query vector and a short embedding batch", async () => {
    const index = await buildNotesIndex(chunks, "fixture", fixtureEmbed);
    expect(() => searchNotes(index, [1, 0])).toThrow("dimensions");
    await expect(buildNotesIndex(chunks, "fixture", async () => [[1, 0, 0]])).rejects.toThrow("Expected 9 vectors");
  });
});

```typescript
// Private notes become small cited chunks. Each chunk keeps the note and the
// section it came from, so every later proposal can point back to it.

export type Note = { id: string; title: string; file: string };

export type NoteChunk = {
  id: string; // `${noteId}#${sectionSlug}`
  noteId: string;
  noteTitle: string;
  section: string;
  text: string;
};

export type NotesIndex = {
  model: string;
  dimensions: number;
  entries: (NoteChunk & { vector: number[] })[];
};

export type ScoredChunk = NoteChunk & { score: number };

export const SAMPLE_NOTES: Note[] = [
  { id: "lumen-strategy-memo", title: "Lumen Kettle strategy memo", file: "lumen-strategy-memo.md" },
  { id: "diary-study-notes", title: "Kitchen diary study notes", file: "diary-study-notes.md" },
  { id: "prototype-trial-log", title: "Prototype trial log", file: "prototype-trial-log.md" },
];

function slug(text: string): string {
  return text.toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
}

// One chunk per `## ` section. Text before the first section is ignored.
export function chunkNote(note: Note, markdown: string): NoteChunk[] {
  const chunks: NoteChunk[] = [];
  let section: string | null = null;
  let lines: string[] = [];
  const flush = () => {
    const text = lines.join(" ").replace(/\s+/g, " ").trim();
    if (section && text) {
      chunks.push({ id: `${note.id}#${slug(section)}`, noteId: note.id, noteTitle: note.title, section, text });
    }
    lines = [];
  };
  for (const line of markdown.split(/\r?\n/)) {
    const heading = /^##\s+(.+)$/.exec(line);
    if (heading) {
      flush();
      section = heading[1].trim();
    } else if (section) {
      lines.push(line);
    }
  }
  flush();
  return chunks;
}

export function cosine(a: readonly number[], b: readonly number[]): number {
  if (a.length !== b.length || a.length === 0) throw new Error(`Vector size mismatch: ${a.length} vs ${b.length}`);
  let dot = 0;
  let na = 0;
  let nb = 0;
  for (let i = 0; i < a.length; i++) {
    dot += a[i] * b[i];
    na += a[i] * a[i];
    nb += b[i] * b[i];
  }
  return na && nb ? dot / Math.sqrt(na * nb) : 0;
}

// Embedding text includes the note and section, so a chunk keeps its context.
export function embeddingText(chunk: NoteChunk): string {
  return `${chunk.noteTitle} — ${chunk.section}: ${chunk.text}`;
}

export async function buildNotesIndex(
  chunks: NoteChunk[],
  model: string,
  embedBatch: (texts: string[]) => Promise<number[][]>,
): Promise<NotesIndex> {
  if (chunks.length === 0) throw new Error("No note sections to index");
  const vectors = await embedBatch(chunks.map(embeddingText));
  if (vectors.length !== chunks.length) {
    throw new Error(`Expected ${chunks.length} vectors, received ${vectors.length}`);
  }
  const dimensions = vectors[0].length;
  if (!vectors.every((v) => v.length === dimensions && v.every(Number.isFinite))) {
    throw new Error("Embedding vectors are inconsistent");
  }
  return { model, dimensions, entries: chunks.map((chunk, i) => ({ ...chunk, vector: vectors[i] })) };
}

export function searchNotes(
  index: NotesIndex,
  query: readonly number[],
  { limit = 3, minScore = 0.3 }: { limit?: number; minScore?: number } = {},
): ScoredChunk[] {
  if (query.length !== index.dimensions) {
    throw new Error(`Query has ${query.length} dimensions; the index has ${index.dimensions}`);
  }
  return index.entries
    .map(({ vector, ...chunk }) => ({ ...chunk, score: cosine(query, vector) }))
    .filter((c) => c.score >= minScore)
    .sort((a, b) => b.score - a.score)
    .slice(0, limit);
}
```
