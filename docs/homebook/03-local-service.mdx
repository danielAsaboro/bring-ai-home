The browser cannot load native models.
A small Node service on the same computer can.
This chapter adds that service and a **Local assistant** panel on each asset.

The rule for the whole course: the service proposes, the binder decides, and a person confirms.
The service never writes to the binder.

**Source checkpoint:** `fb2b93d` · **Starting checkpoint:** `d126ea1`

## What you change

| File | Change |
| --- | --- |
| `qvac-service/server.ts` | Loopback HTTP service: origin check, size limit, `/health`, `POST /label`, and clear 400/403/503/502 errors |
| `qvac-service/serve.mjs` | Starts listening first, then loads models in the background. Each capability turns on when its model loads |
| `src/features/homebook/assist-client.ts` | Browser client. Classifies a stopped service, a missing model, and a rejected request |
| `src/features/homebook/AssistPanel.tsx` | Upload a rating plate, review proposed facts, and press **Use model** to accept one |
| `src/features/homebook/Homebook.tsx` | Show the panel on each asset plate |
| `tsconfig.json` | Allow `.ts` import extensions so Node can load app modules from the service |

## Build and observe

### 1. Start both processes

```bash
npm run qvac:service
npm run dev
```

The service prints `Homebook QVAC service on http://127.0.0.1:4319`.
It binds to `127.0.0.1`, not `0.0.0.0`, so other machines on your network cannot reach it.

### 2. Check health before the model is ready

```bash
curl -s http://127.0.0.1:4319/health
```

While OCR loads, the service answers `{"ok":true,"capabilities":{"label":false}}`.
Ask it to read the label now and it answers 503 `Label reading is not available`.
We first wrote `serve.mjs` to load the model before listening.
In a network that stalls the download, the service never started listening, and the browser could only say it was offline.
Listening first gives the correct error.

### 3. Check the origin rule

```bash
curl -s -H 'Origin: https://example.com' http://127.0.0.1:4319/health
```

It answers 403.
Only `http://localhost` and `http://127.0.0.1` pages may call the service.

### 4. Review a plate in the binder

Register a second water heater by hand with only a name and a room.
In its **Local assistant**, choose `public/test-files/homebook/water-heater-label.png`.
Each proposed value shows the text it came from and its confidence.
Nothing changes until you press **Use model** or **Use serial**.

Stop the service and try again.
The panel says `The local QVAC service is not running` and the binder is unchanged.

### 5. A bug at the boundary

The first `serve.mjs` run failed with `ERR_MODULE_NOT_FOUND ... label-facts`.
Node loads TypeScript files directly but needs the real file extension.
`server.ts` now imports `label-facts.ts`, and `tsconfig.json` sets `allowImportingTsExtensions`.

```javascript
import { mkdir, writeFile } from "node:fs/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const qvacDirectory = path.join(appDirectory, ".qvac");
const cacheDirectory = path.join(qvacDirectory, "models");
const configPath = path.join(qvacDirectory, "config.json");

const { createHomebookServer, SERVICE_PORT } = await import("./server.ts");

await mkdir(cacheDirectory, { recursive: true });
await writeFile(configPath, `${JSON.stringify({ cacheDirectory }, null, 2)}\n`, "utf8");
process.env.QVAC_CONFIG_PATH = configPath;

const loaded = [];
let sdk = null;
try {
  sdk = await import("@qvac/sdk");
} catch (error) {
  process.stderr.write(`QVAC SDK unavailable: ${error instanceof Error ? error.message : String(error)}\n`);
}

// Load one model, or report why not and leave that capability off.
async function tryLoad(label, options) {
  if (!sdk) return null;
  try {
    const modelId = await sdk.loadModel(options);
    loaded.push(modelId);
    return modelId;
  } catch (error) {
    process.stderr.write(`${label} unavailable: ${error instanceof Error ? error.message : String(error)}\n`);
    return null;
  }
}

// Listen first. Each capability switches on when its model has loaded, so the
// binder gets a clear 503 instead of a hanging request while models download.
const capabilities = { label: null };
const server = createHomebookServer(capabilities);
server.listen(SERVICE_PORT, "127.0.0.1", () => {
  process.stderr.write(`Homebook QVAC service on http://127.0.0.1:${SERVICE_PORT}\n`);
});

const ocrId = await tryLoad("Label OCR", {
  modelSrc: sdk?.OCR_LATIN,
  modelType: sdk?.MODEL_TYPES.ggmlOcr,
  modelConfig: { langList: ["en"], magRatio: 1.5, contrastRetry: false, lowConfidenceThreshold: 0.5 },
});
if (ocrId) {
  capabilities.label = async (image) => {
    const run = sdk.ocr({ modelId: ocrId, image, options: { paragraph: false } });
    return { blocks: await run.blocks, model: "OCR_LATIN" };
  };
  process.stderr.write("Label OCR ready\n");
}

async function shutdown() {
  server.close();
  for (const modelId of loaded) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  if (sdk) await sdk.close().catch(() => undefined);
  process.exit(0);
}
process.on("SIGINT", shutdown);
process.on("SIGTERM", shutdown);
```

```typescript
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";

import type { LabelFact, OcrBlock } from "../src/features/homebook/label-facts.ts";
import { proposeLabelFacts } from "../src/features/homebook/label-facts.ts";

export const SERVICE_PORT = 4319;
const MAX_BODY_BYTES = 6 * 1024 * 1024;
const IMAGE_TYPES = new Set(["image/png", "image/jpeg"]);

export type LabelReader = (image: Buffer) => Promise<{ blocks: OcrBlock[]; model: string }>;

export type Capabilities = {
  label: LabelReader | null;
};

export type ImageRequest = { fileName: string; mimeType: string; image: Buffer };

export function isAllowedOrigin(origin: string | undefined): boolean {
  if (!origin) return true; // curl and other local tools send no Origin
  try {
    const url = new URL(origin);
    return url.protocol === "http:" && (url.hostname === "localhost" || url.hostname === "127.0.0.1");
  } catch {
    return false;
  }
}

export function parseImageRequest(value: unknown): ImageRequest {
  const v = value as { fileName?: unknown; mimeType?: unknown; imageBase64?: unknown } | null;
  if (!v || typeof v.fileName !== "string" || typeof v.imageBase64 !== "string" || typeof v.mimeType !== "string") {
    throw new Error("Request must contain fileName, mimeType, and imageBase64");
  }
  if (!IMAGE_TYPES.has(v.mimeType)) throw new Error("Only PNG and JPEG images are accepted");
  const image = Buffer.from(v.imageBase64, "base64");
  if (image.length === 0) throw new Error("The image is empty");
  return { fileName: v.fileName, mimeType: v.mimeType, image };
}

export function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  res.writeHead(status, {
    "content-type": "application/json",
    ...(origin ? { "access-control-allow-origin": origin, vary: "Origin" } : {}),
    "access-control-allow-headers": "content-type",
    "access-control-allow-methods": "GET, POST, OPTIONS",
  });
  res.end(status === 204 ? undefined : JSON.stringify(body));
}

export async function readJson(req: IncomingMessage): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > MAX_BODY_BYTES) throw new Error("Request is too large");
    chunks.push(chunk as Buffer);
  }
  return JSON.parse(Buffer.concat(chunks).toString("utf8"));
}

type Route = (body: unknown) => Promise<unknown>;

export class Unavailable extends Error {}

// The service proposes. It never writes to the binder. The browser decides.
export function createHomebookServer(capabilities: Capabilities) {
  const routes: Record<string, Route> = {
    "/label": async (body) => {
      const request = parseImageRequest(body);
      if (!capabilities.label) throw new Unavailable("Label reading is not available");
      const { blocks, model } = await capabilities.label(request.image);
      const facts: LabelFact[] = proposeLabelFacts(blocks);
      return { kind: "proposed-label-facts", fileName: request.fileName, model, blocks, facts };
    },
  };

  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (!isAllowedOrigin(origin)) return send(res, 403, { error: "Origin not allowed" });
    if (req.method === "OPTIONS") return send(res, 204, null, origin);
    if (req.method === "GET" && req.url === "/health") {
      const available = Object.fromEntries(Object.entries(capabilities).map(([k, v]) => [k, v !== null]));
      return send(res, 200, { ok: true, capabilities: available }, origin);
    }
    const route = req.method === "POST" && req.url ? routes[req.url] : undefined;
    if (!route) return send(res, 404, { error: "Not found" }, origin);
    let body: unknown;
    try {
      body = await readJson(req);
    } catch (error) {
      return send(res, 400, { error: (error as Error).message }, origin);
    }
    try {
      return send(res, 200, await route(body), origin);
    } catch (error) {
      if (error instanceof Unavailable) return send(res, 503, { error: error.message }, origin);
      const message = (error as Error).message;
      const status = /^(Request|Only|The image)/.test(message) ? 400 : 502;
      return send(res, status, { error: message }, origin);
    }
  });
}
```

```tsx
"use client";

import { useState, type ChangeEvent } from "react";

import { AssistError, serviceClient, type AssistClient } from "./assist-client";
import type { Asset } from "./binder";
import type { LabelFact, LabelField } from "./label-facts";

const FIELD_NAMES: Record<LabelField, string> = {
  brand: "Brand",
  model: "Model",
  serial: "Serial",
  installedOn: "Installed",
};

type Status = { state: "idle" } | { state: "working"; what: string } | { state: "error"; message: string };

export function AssistPanel({
  asset,
  client = serviceClient,
  onAccept,
}: {
  asset: Asset;
  client?: AssistClient;
  onAccept: (field: LabelField, value: string) => void;
}) {
  const [status, setStatus] = useState<Status>({ state: "idle" });
  const [facts, setFacts] = useState<{ assetId: string; model: string; items: LabelFact[] } | null>(null);

  async function run<T>(what: string, job: () => Promise<T>): Promise<T | null> {
    setStatus({ state: "working", what });
    try {
      const result = await job();
      setStatus({ state: "idle" });
      return result;
    } catch (error) {
      const message = error instanceof AssistError ? error.message : "Something went wrong. The binder is unchanged.";
      setStatus({ state: "error", message });
      return null;
    }
  }

  async function onLabel(event: ChangeEvent<HTMLInputElement>) {
    const file = event.target.files?.[0];
    event.target.value = "";
    if (!file) return;
    const result = await run("Reading the label locally…", () => client.readLabel(file));
    if (result) setFacts({ assetId: asset.id, model: result.model, items: result.facts });
  }

  const visibleFacts = facts?.assetId === asset.id ? facts : null;

  return (
    <section aria-label="Local assistant" className="space-y-3 rounded border border-emerald-800/40 bg-emerald-50/50 p-3 text-sm">
      <h3 className="font-serif text-lg">Local assistant</h3>
      <p className="text-xs text-stone-600">Runs on this computer through QVAC. It proposes; you decide what goes in the binder.</p>

      <label className="block">
        <span className="font-medium">Read a rating plate</span>
        <input type="file" accept="image/png,image/jpeg" aria-label="Rating plate photo" onChange={onLabel} className="mt-1 block text-xs" />
      </label>

      {status.state === "working" ? <p role="status">{status.what}</p> : null}
      {status.state === "error" ? <p role="alert" className="text-red-900">{status.message}</p> : null}

      {visibleFacts ? (
        <div>
          <p className="text-xs text-stone-600">Proposed by {visibleFacts.model}. Check each value against the plate.</p>
          {visibleFacts.items.length === 0 ? <p>No label fields were recognised. Enter them by hand.</p> : null}
          <ul aria-label="Proposed label facts" className="divide-y">
            {visibleFacts.items.map((fact) => (
              <li key={fact.field} className="flex flex-wrap items-center justify-between gap-2 py-1">
                <span>
                  <strong>{FIELD_NAMES[fact.field]}:</strong> <code>{fact.value}</code>{" "}
                  <span className="text-xs text-stone-600">
                    from “{fact.sourceText}”{fact.confidence !== null ? ` · ${Math.round(fact.confidence * 100)}%` : ""}
                  </span>
                </span>
                <button
                  type="button"
                  className="rounded border border-emerald-800 px-2 py-0.5"
                  onClick={() => {
                    onAccept(fact.field, fact.value);
                    setFacts({ ...visibleFacts, items: visibleFacts.items.filter((f) => f.field !== fact.field) });
                  }}
                >
                  Use {FIELD_NAMES[fact.field].toLowerCase()}
                </button>
              </li>
            ))}
          </ul>
        </div>
      ) : null}
    </section>
  );
}
```

```tsx
"use client";

import { useState, useSyncExternalStore, type FormEvent } from "react";

import type { AssistClient } from "./assist-client";
import { AssistPanel } from "./AssistPanel";
import {
  ASSET_CATEGORIES,
  completeTask,
  dueState,
  fileDocument,
  historyFor,
  recordProject,
  registerAsset,
  type AssetCategory,
  type Binder,
  type DocumentKind,
  updateAsset,
} from "./binder";
import { browserStore, type BinderStore } from "./storage";

const noopSubscribe = () => () => undefined;

function todayIso() {
  return new Date().toISOString().slice(0, 10);
}

export function Homebook({ store, today, client }: { store?: BinderStore; today?: string; client?: AssistClient }) {
  const mounted = useSyncExternalStore(noopSubscribe, () => true, () => false);
  if (!mounted) return <p className="p-6">Opening the binder…</p>;
  return <BinderView store={store ?? browserStore()} today={today ?? todayIso()} client={client} />;
}

const field = "rounded border border-stone-400 bg-white px-2 py-1 text-sm";
const button = "rounded bg-emerald-800 px-3 py-1 text-sm font-medium text-white hover:bg-emerald-900";

function BinderView({ store, today, client }: { store: BinderStore; today: string; client?: AssistClient }) {
  const [binder, setBinder] = useState<Binder>(() => store.load());
  const [roomId, setRoomId] = useState<string>("all");
  const [selectedId, setSelectedId] = useState<string | null>(binder.assets[0]?.id ?? null);
  const [error, setError] = useState<string | null>(null);

  function commit(next: Binder) {
    setBinder(next);
    store.save(next);
    setError(null);
  }

  function attempt(change: () => Binder) {
    try {
      commit(change());
    } catch (e) {
      setError((e as Error).message);
    }
  }

  const assets = binder.assets.filter((a) => roomId === "all" || a.roomId === roomId);
  const selected = binder.assets.find((a) => a.id === selectedId) ?? null;
  const roomName = (id: string) => binder.rooms.find((r) => r.id === id)?.name ?? id;

  function onRegister(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const form = new FormData(event.currentTarget);
    const text = (k: string) => String(form.get(k) ?? "");
    const id = `asset-${binder.assets.length + 1}-${Date.now().toString(36)}`;
    attempt(() =>
      registerAsset(
        binder,
        {
          name: text("name"),
          roomId: text("roomId"),
          category: text("category") as AssetCategory,
          brand: text("brand"),
          model: text("model"),
          serial: text("serial"),
          installedOn: text("installedOn"),
          notes: "",
        },
        id,
      ),
    );
    if (text("name").trim()) {
      setSelectedId(id);
      event.currentTarget.reset();
    }
  }

  function onFile(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    if (!selected) return;
    const form = new FormData(event.currentTarget);
    attempt(() =>
      fileDocument(binder, {
        assetId: selected.id,
        kind: String(form.get("kind")) as DocumentKind,
        title: String(form.get("title") ?? ""),
        file: null,
        expiresOn: String(form.get("expiresOn") ?? "") || null,
      }),
    );
    event.currentTarget.reset();
  }

  function onProject(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const form = new FormData(event.currentTarget);
    attempt(() =>
      recordProject(binder, {
        roomId: roomId === "all" ? binder.rooms[0].id : roomId,
        title: String(form.get("title") ?? ""),
        before: String(form.get("before") ?? ""),
        after: String(form.get("after") ?? ""),
        finishedOn: null,
      }),
    );
    event.currentTarget.reset();
  }

  return (
    <div className="mx-auto max-w-5xl space-y-6 p-4 text-stone-900 sm:p-6">
      <header className="flex flex-wrap items-end justify-between gap-3 border-b-4 border-double border-stone-500 pb-3">
        <div>
          <h1 className="font-serif text-3xl">Homebook</h1>
```

```typescript
import type { LabelFact } from "./label-facts";

export const SERVICE_URL = "http://127.0.0.1:4319";

export type AssistErrorKind = "service-offline" | "model-unavailable" | "rejected" | "failed";
export class AssistError extends Error {
  constructor(public kind: AssistErrorKind, message: string) {
    super(message);
  }
}

export async function postJson<T>(path: string, body: unknown, fetcher: typeof fetch = fetch): Promise<T> {
  let res: Response;
  try {
    res = await fetcher(`${SERVICE_URL}${path}`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(body),
    });
  } catch {
    throw new AssistError("service-offline", "The local QVAC service is not running. Start it with npm run qvac:service.");
  }
  const payload = (await res.json().catch(() => ({}))) as { error?: string } & T;
  if (res.ok) return payload;
  const message = payload.error ?? `Service answered ${res.status}`;
  if (res.status === 503) throw new AssistError("model-unavailable", message);
  if (res.status === 400) throw new AssistError("rejected", message);
  throw new AssistError("failed", message);
}

async function toBase64(file: Blob): Promise<string> {
  const bytes = new Uint8Array(await file.arrayBuffer());
  let binary = "";
  for (let i = 0; i < bytes.length; i += 0x8000) binary += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  return btoa(binary);
}

export async function imageBody(file: File) {
  return { fileName: file.name, mimeType: file.type, imageBase64: await toBase64(file) };
}

export type AssistClient = {
  readLabel(file: File): Promise<{ facts: LabelFact[]; model: string }>;
};

export const serviceClient: AssistClient = {
  async readLabel(file) {
    return postJson("/label", await imageBody(file));
  },
};
```
