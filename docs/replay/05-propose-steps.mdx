Now a transcript and a few screen notes become a list of proposed frames.

JSON Schema makes the answer parseable. It does not make it true. A model can produce perfect JSON for a step nobody said. So every proposed step must quote the narration, and plain code checks that quote before the step is shown.

**Source checkpoint:** `01890a7` · **Starting checkpoint:** `41aaa15`

## What changes

- `src/features/replay/proposals.ts`: `STEPS_JSON_SCHEMA`, the prompt, and `parseStepProposals`, the deterministic gate. It rejects non-JSON, empty or long titles, duplicates, and any step whose quote is not in the transcript, with a reason for each.
- `src/features/replay/fixtures/model-output.fixture.ts`: a hand-written string shaped like model output, with one invented step and one duplicate. Labelled as a fixture.
- `qvac-service/qvac-models.ts`: `createQvacDrafter` runs `LLAMA_3_2_1B_INST_Q4_0` with `responseFormat: { type: "json_schema" }` and returns `(await run.final).contentText`.
- `/v1/propose-steps` returns the raw text and the gated result. Unusable output is `422 unusable_output`.
- The panel shows proposals with their quotes as checkboxes, a "held back" list with reasons, and **Add N selected frames**.

To compare with the previous checkpoint: `git diff 41aaa15 01890a7`.

## Build and observe

### 1. Propose from your narration

```bash
git checkout 01890a7
npm run qvac:service
npm run dev
```

Transcribe your recording, optionally read the screenshot, then click **Propose frames**. The first call downloads the LLM.

### 2. Read the quotes

Each proposal shows the words it claims to come from. Open **proposals were held back** to see what the gate refused and why. If your run holds nothing back, that is fine. The gate's behaviour is proven by the fixture test, not by hoping the model fails on camera.

### 3. Choose

Nothing is added yet. Tick the steps you agree with and click **Add N selected frames**. They append to the tape as ordinary frames. Edit them like any other frame.

// The only file that imports @qvac/sdk for the service. Every model loads on first use.
import { mkdirSync, writeFileSync } from "node:fs";
import path from "node:path";

import type { ModelState } from "../src/features/replay/service-contract.ts";
import type { Drafter, Transcriber, Viewer } from "./server.ts";

type Sdk = typeof import("@qvac/sdk");

export function prepareQvacConfig(appDirectory: string) {
  const cacheDirectory = path.join(appDirectory, ".qvac", "models");
  const configPath = path.join(appDirectory, ".qvac", "config.json");
  mkdirSync(cacheDirectory, { recursive: true });
  writeFileSync(configPath, `${JSON.stringify({ cacheDirectory }, null, 2)}\n`, "utf8");
  process.env.QVAC_CONFIG_PATH = configPath;
  return { cacheDirectory, configPath };
}

let sdkPromise: Promise<Sdk> | undefined;
export function sdk() {
  sdkPromise ??= import("@qvac/sdk");
  return sdkPromise;
}

// Close the SDK only if something actually loaded it.
export async function closeSdk() {
  if (!sdkPromise) return;
  try {
    await (await sdkPromise).close();
  } catch {
    // The SDK never loaded or is already closed; nothing is left to release.
  }
}

// One lazy model: load once, report state honestly, unload on shutdown.
export function lazyModel(kind: string, id: string, load: (qvac: Sdk) => Promise<string>) {
  let modelId: string | undefined;
  let state: ModelState["state"] = "not_loaded";
  let loading: Promise<string> | undefined;
  return {
    modelState: (): ModelState => ({ kind, id, state }),
    async ensure() {
      if (modelId) return modelId;
      loading ??= (async () => {
        state = "loading";
        modelId = await load(await sdk());
        state = "ready";
        return modelId;
      })().catch((error: unknown) => {
        state = "error";
        loading = undefined;
        throw error;
      });
      return loading;
    },
    async close() {
      if (modelId) await (await sdk()).unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
    },
  };
}

export function createQvacTranscriber(): Transcriber & { close(): Promise<void> } {
  const model = lazyModel("transcription", "WHISPER_TINY", (q) =>
    q.loadModel({
      modelSrc: q.WHISPER_TINY,
      modelConfig: { language: "en", translate: false, temperature: 0.0 },
    }),
  );
  return {
    modelState: model.modelState,
    async transcribe(wavPath) {
      const modelId = await model.ensure();
      const segments = await (await sdk()).transcribe({ modelId, audioChunk: wavPath, metadata: true });
      return segments.map((s) => ({ text: s.text, startMs: s.startMs, endMs: s.endMs }));
    },
    close: model.close,
  };
}

export function createQvacViewer(): Viewer & { close(): Promise<void> } {
  const model = lazyModel("multimodal", "SMOLVLM2_500M_MULTIMODAL_Q8_0", (q) =>
    q.loadModel({
      modelSrc: q.SMOLVLM2_500M_MULTIMODAL_Q8_0,
      modelConfig: { ctx_size: 1024, projectionModelSrc: q.MMPROJ_SMOLVLM2_500M_MULTIMODAL_Q8_0 },
    }),
  );
  return {
    modelState: model.modelState,
    async describe(imagePath, prompt) {
      const modelId = await model.ensure();
      const run = (await sdk()).completion({
        modelId,
        history: [{ role: "user", content: prompt, attachments: [{ path: imagePath }] }],
        stream: true,
      });
      return (await run.final).contentText;
    },
    close: model.close,
  };
}

export function createQvacDrafter(): Drafter & { close(): Promise<void> } {
  const model = lazyModel("llm", "LLAMA_3_2_1B_INST_Q4_0", (q) =>
    q.loadModel({ modelSrc: q.LLAMA_3_2_1B_INST_Q4_0, modelType: "llm" }),
  );
  return {
    modelState: model.modelState,
    async complete(history, schema) {
      const modelId = await model.ensure();
      const run = (await sdk()).completion({
        modelId,
        history,
        stream: true,
        responseFormat: { type: "json_schema", json_schema: schema },
      });
      return (await run.final).contentText;
    },
    close: model.close,
  };
}

import { createHash } from "node:crypto";
import { mkdtemp, rm, writeFile } from "node:fs/promises";
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";
import { tmpdir } from "node:os";
import path from "node:path";

import { toNarration, type NarrationSegment } from "../src/features/replay/narration.ts";
import { buildStepsHistory, parseStepProposals, STEPS_JSON_SCHEMA, type ChatMessage } from "../src/features/replay/proposals.ts";
import { parseScreenNote, SCREEN_NOTE_PROMPT } from "../src/features/replay/screen-notes.ts";
import {
  parseProposeStepsRequest,
  parseReadScreenshotRequest,
  parseTranscribeRequest,
  type ModelState,
} from "../src/features/replay/service-contract.ts";

// Each model sits behind its own small door. Tests inject fakes; main.ts injects QVAC.
export type Transcriber = {
  modelState(): ModelState;
  transcribe(wavPath: string): Promise<NarrationSegment[]>;
};

// A multimodal model that looks at one image file and answers in plain text.
export type Viewer = {
  modelState(): ModelState;
  describe(imagePath: string, prompt: string): Promise<string>;
};

// Text generation constrained by a JSON schema. It returns raw text; the server validates it.
export type Drafter = {
  modelState(): ModelState;
  complete(history: ChatMessage[], schema: { name: string; schema: Record<string, unknown> }): Promise<string>;
};

export type ServiceModels = { transcriber?: Transcriber; viewer?: Viewer; drafter?: Drafter };

// Write request bytes to a private temp file for the length of one model call.
async function withTempFile<T>(name: string, bytes: Buffer, run: (filePath: string) => Promise<T>) {
  const directory = await mkdtemp(path.join(tmpdir(), "replay-"));
  try {
    const filePath = path.join(directory, name);
    await writeFile(filePath, bytes);
    return await run(filePath);
  } finally {
    await rm(directory, { recursive: true, force: true });
  }
}

const LOOPBACK_ORIGIN = /^http:\/\/(localhost|127\.0\.0\.1|\[::1\])(:\d+)?$/;

// Plain fields, not parameter properties: Node's type stripping cannot run those.
export class RequestError extends Error {
  status: number;
  code: string;
  constructor(status: number, code: string, message: string) {
    super(message);
    this.status = status;
    this.code = code;
  }
}

function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  const headers: Record<string, string> = { "content-type": "application/json" };
  if (origin) {
    headers["access-control-allow-origin"] = origin;
    headers["vary"] = "Origin";
  }
  res.writeHead(status, headers);
  res.end(JSON.stringify(body));
}

async function readJson(req: IncomingMessage, limit: number): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > limit) throw new RequestError(413, "too_large", "Request body is too large");
    chunks.push(chunk as Buffer);
  }
  try {
    return JSON.parse(Buffer.concat(chunks).toString("utf8"));
  } catch {
    throw new RequestError(400, "invalid_json", "Body must be JSON");
  }
}

type Route = (req: IncomingMessage) => Promise<unknown>;

export function createReplayServer(models: ServiceModels) {
  const routes = new Map<string, Route>();

  if (models.transcriber) {
    const transcriber = models.transcriber;
    routes.set("/v1/transcribe", async (req) => {
      const parsed = parseTranscribeRequest(await readJson(req, 50_000_000));
      if (!parsed.ok) throw new RequestError(400, "invalid_request", parsed.message);
      const audio = Buffer.from(parsed.value.audioBase64, "base64");
      const sourceDigest = createHash("sha256").update(audio).digest("hex");
      // The recording lives on disk only for the length of this request.
      return withTempFile("narration.wav", audio, async (wavPath) => {
        const started = Date.now();
        const segments = await transcriber.transcribe(wavPath);
        let narration;
        try {
          narration = toNarration(segments);
        } catch (cause) {
          throw new RequestError(422, "no_speech", (cause as Error).message);
        }
        return { sourceDigest, model: transcriber.modelState().id, transcribeMs: Date.now() - started, narration };
      });
    });
  }

  if (models.viewer) {
    const viewer = models.viewer;
    routes.set("/v1/read-screenshot", async (req) => {
      const parsed = parseReadScreenshotRequest(await readJson(req, 25_000_000));
      if (!parsed.ok) throw new RequestError(400, "invalid_request", parsed.message);
      const image = Buffer.from(parsed.value.imageBase64, "base64");
      const sourceDigest = createHash("sha256").update(image).digest("hex");
      const ext = parsed.value.fileName.toLowerCase().split(".").pop();
      const raw = await withTempFile(`screen.${ext}`, image, (p) => viewer.describe(p, SCREEN_NOTE_PROMPT));
      try {
        return { sourceDigest, model: viewer.modelState().id, note: parseScreenNote(raw) };
      } catch (cause) {
        throw new RequestError(422, "empty_note", (cause as Error).message);
      }
    });
  }

  if (models.drafter) {
    const drafter = models.drafter;
    routes.set("/v1/propose-steps", async (req) => {
      const parsed = parseProposeStepsRequest(await readJson(req, 100_000));
      if (!parsed.ok) throw new RequestError(400, "invalid_request", parsed.message);
      const raw = await drafter.complete(buildStepsHistory(parsed.value), { name: "steps", schema: STEPS_JSON_SCHEMA });
      try {
        return { model: drafter.modelState().id, raw, result: parseStepProposals(raw, parsed.value.transcript) };
      } catch (cause) {
        throw new RequestError(422, "unusable_output", (cause as Error).message);
      }
    });
  }

  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (origin !== undefined && !LOOPBACK_ORIGIN.test(origin)) {
      send(res, 403, { error: { code: "origin_rejected", message: "Only loopback origins may call this service" } });
      return;
    }
    if (req.method === "OPTIONS") {
      res.writeHead(204, {
        "access-control-allow-origin": origin ?? "",
        "access-control-allow-methods": "GET, POST",
        "access-control-allow-headers": "content-type",
      });
      res.end();
      return;
    }
    if (req.method === "GET" && req.url === "/health") {
      const states = Object.values(models).map((m) => m.modelState());
      send(res, 200, { service: "ready", models: states }, origin);
      return;
    }
    const route = req.method === "POST" && req.url ? routes.get(req.url) : undefined;
    if (!route) {
      send(res, 404, { error: { code: "not_found", message: "Unknown route" } }, origin);
      return;
    }
    try {
      send(res, 200, await route(req), origin);
    } catch (cause) {
      if (cause instanceof RequestError) {
        send(res, cause.status, { error: { code: cause.code, message: cause.message } }, origin);
        return;
      }
      const message = cause instanceof Error ? cause.message : String(cause);
      send(res, 500, { error: { code: "model_failed", message } }, origin);
    }
  });
}

// Turning narration into proposed frames.
// The model proposes. This file decides what is even allowed to be shown. A person accepts.

export type ChatMessage = { role: "system" | "user"; content: string };

export const MAX_PROPOSED_STEPS = 12;

export const STEPS_JSON_SCHEMA = {
  type: "object",
  properties: {
    steps: {
      type: "array",
      maxItems: MAX_PROPOSED_STEPS,
      items: {
        type: "object",
        properties: {
          title: { type: "string" },
          detail: { type: "string" },
          quote: { type: "string" },
        },
        required: ["title", "detail", "quote"],
      },
    },
  },
  required: ["steps"],
} as const;

export function buildStepsHistory(input: { transcript: string; screenNotes: string[] }): ChatMessage[] {
  const notes = input.screenNotes.length
    ? `\n\nScreen notes from screenshots (may be incomplete):\n${input.screenNotes.map((n) => `- ${n}`).join("\n")}`
    : "";
  return [
    {
      role: "system",
      content:
        "You turn a spoken demonstration into a short checklist. Each step is one action a person does by hand. " +
        "Use a short imperative title. For every step, copy the exact words from the narration that support it into quote. " +
        "Do not add steps that were not said. Answer only with JSON.",
    },
    { role: "user", content: `Narration:\n${input.transcript}${notes}` },
  ];
}

export type StepProposal = { id: string; title: string; detail: string; quote: string };
export type Rejection = { title: string; reason: string };
export type ProposalResult = { proposals: StepProposal[]; rejected: Rejection[] };

export function normalizeForMatch(text: string) {
  return text.toLowerCase().replace(/[^a-z0-9]+/g, " ").trim();
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

// The deterministic gate. JSON Schema shapes the output; it cannot make it true.
export function parseStepProposals(raw: string, transcript: string): ProposalResult {
  let json: unknown;
  try {
    json = JSON.parse(raw);
  } catch {
    throw new Error("The model did not return JSON");
  }
  if (!isRecord(json) || !Array.isArray(json.steps)) throw new Error("The model's JSON has no steps list");
  const haystack = normalizeForMatch(transcript);
  const proposals: StepProposal[] = [];
  const rejected: Rejection[] = [];
  const seen = new Set<string>();
  for (const item of json.steps.slice(0, MAX_PROPOSED_STEPS)) {
    if (!isRecord(item) || typeof item.title !== "string") {
      rejected.push({ title: "(unreadable)", reason: "Step is not an object with a title" });
      continue;
    }
    const title = item.title.replace(/\s+/g, " ").trim();
    const detail = typeof item.detail === "string" ? item.detail.replace(/\s+/g, " ").trim() : "";
    const quote = typeof item.quote === "string" ? item.quote.trim() : "";
    const key = normalizeForMatch(title);
    if (!title || title.length > 80) {
      rejected.push({ title: title || "(empty)", reason: "Title is empty or longer than 80 characters" });
    } else if (seen.has(key)) {
      rejected.push({ title, reason: "Duplicate of an earlier step" });
    } else if (normalizeForMatch(quote).length < 8 || !haystack.includes(normalizeForMatch(quote))) {
      rejected.push({ title, reason: "Quote does not appear in the narration" });
    } else {
      seen.add(key);
      proposals.push({ id: `p${proposals.length + 1}`, title, detail, quote });
    }
  }
  return { proposals, rejected };
}

                ? "bg-amber-500 text-stone-950"
                : i < run.playhead || run.status === "finished"
                  ? f.outcome === "done"
                    ? "bg-emerald-800"
                    : f.outcome === "failed"
                      ? "bg-red-800"
                      : "bg-stone-700"
                  : "bg-stone-900"
            }`}
          >
            {String(i + 1).padStart(2, "0")}
          </li>
        ))}
      </ol>
      {!ended && step && (
        <div className="rounded border border-stone-700 p-3">
          <p className="text-xs text-stone-400">
            Frame {run.playhead + 1} of {run.frames.length}
            {current.outcome === "failed" && " · marked failed"}
          </p>
          <h2 className="text-lg font-semibold">{step.title}</h2>
          <p className="text-sm text-stone-300">{step.detail}</p>
        </div>
      )}
      {!ended ? (
        <div className="flex flex-wrap gap-2">
          <button type="button" className={btn} disabled={run.status === "paused" || current.outcome === "failed"} onClick={() => onAction({ type: "done" })}>
            Done
          </button>
          <button type="button" className={btn} disabled={run.status === "paused"} onClick={() => onAction({ type: "skip" })}>
            Skip
          </button>
          <button type="button" className={btn} disabled={run.status === "paused" || current.outcome === "failed"} onClick={() => onAction({ type: "fail" })}>
            Mark failed
          </button>
          <button type="button" className={btn} disabled={current.outcome !== "failed" || run.status === "paused"} onClick={() => onAction({ type: "retry" })}>
            Retry
          </button>
          {run.status === "paused" ? (
            <button type="button" className={btn} onClick={() => onAction({ type: "resume" })}>
              Resume
            </button>
          ) : (
            <button type="button" className={btn} onClick={() => onAction({ type: "pause" })}>
              Pause
            </button>
          )}
        </div>
      ) : (
        <div className="space-y-2">
          <p role="status">
            {run.status === "finished" ? "Run finished." : `Run stopped: ${run.stopReason}`}{" "}
            {(() => {
              const s = summarizeRun(run);
              return `${s.done} done, ${s.skipped} skipped, ${s.failed} failed.`;
            })()}
          </p>
          <button type="button" className={btn} onClick={onReplay}>
            Play again
          </button>
        </div>
      )}
      {!ended && (
        <div className="grid gap-2 md:grid-cols-2">
          <form
            className="flex gap-2"
            onSubmit={(e) => {
              e.preventDefault();
              onAction({ type: "annotate", note });
              setNote("");
            }}
          >
            <input aria-label="Frame note" placeholder={current.note || "Note for this frame"} className="flex-1 rounded border border-stone-700 bg-stone-900 px-2" value={note} onChange={(e) => setNote(e.target.value)} />
            <button type="submit" className={btn}>
              Save note
            </button>
          </form>
          <form
            className="flex gap-2"
            onSubmit={(e) => {
              e.preventDefault();
              onAction({ type: "stop", reason });
            }}
          >
            <input aria-label="Stop reason" placeholder="Why stop?" className="flex-1 rounded border border-stone-700 bg-stone-900 px-2" value={reason} onChange={(e) => setReason(e.target.value)} />
            <button type="submit" className={btn}>
              Stop safely
            </button>
          </form>
        </div>
      )}
    </section>
  );
}

                    </li>
                  ))}
              </ul>
            )}
          </li>
        );
      })}
    </ul>
  );
}

export type ClientResult<T> = { ok: true; value: T } | { ok: false; code: string; message: string };

// Every call can fail in three different ways, and the UI says which one happened.
export async function postJson<T>(
  path: string,
  body: unknown,
  parse: (json: unknown) => Parsed<T>,
  fetchImpl: typeof fetch = fetch,
): Promise<ClientResult<T>> {
  let response: Response;
  try {
    response = await fetchImpl(`${SERVICE_URL}${path}`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(body),
    });
  } catch {
    return {
      ok: false,
      code: "service_unreachable",
      message: "The local QVAC service is not running. Start it with npm run qvac:service. Manual editing still works.",
    };
  }
  let json: unknown;
  try {
    json = await response.json();
  } catch {
    return { ok: false, code: "invalid_response", message: "The service answered with something that is not JSON" };
  }
  if (!response.ok) {
    const error = parseServiceError(json);
    return { ok: false, code: error?.code ?? "service_error", message: error?.message ?? `HTTP ${response.status}` };
  }
  const parsed = parse(json);
  return parsed.ok ? parsed : { ok: false, code: "invalid_response", message: parsed.message };
}

export async function fileToBase64(file: Blob) {
  const bytes = new Uint8Array(await file.arrayBuffer());
  let binary = "";
  for (let i = 0; i < bytes.length; i += 0x8000) {
    binary += String.fromCharCode(...bytes.subarray(i, i + 0x8000));
  }
  return btoa(binary);
}

export async function transcribeRecording(
  file: File,
  fetchImpl?: typeof fetch,
): Promise<ClientResult<TranscribeResponse>> {
  const audioBase64 = await fileToBase64(file);
  return postJson("/v1/transcribe", { fileName: file.name, audioBase64 }, parseTranscribeResponse, fetchImpl);
}

// Shapes that cross the loopback boundary between the browser and the QVAC service.
// JSON from another process can be anything, so both sides validate at runtime.
import type { Narration } from "./narration.ts";
import type { ProposalResult } from "./proposals.ts";
import { imageExtension, type ScreenNote } from "./screen-notes.ts";

export const SERVICE_PORT = 4319;
export const SERVICE_URL = `http://127.0.0.1:${SERVICE_PORT}`;

export type Parsed<T> = { ok: true; value: T } | { ok: false; message: string };

export type ModelState = { kind: string; id: string; state: "not_loaded" | "loading" | "ready" | "error" };

export type ServiceError = { error: { code: string; message: string } };

export type TranscribeRequest = { fileName: string; audioBase64: string };
export type TranscribeResponse = {
  sourceDigest: string;
  model: string;
  transcribeMs: number;
  narration: Narration;
};

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === "object" && value !== null && !Array.isArray(value);
}

function nonEmptyString(value: unknown, max: number): value is string {
  return typeof value === "string" && value.trim().length > 0 && value.length <= max;
}

export function parseTranscribeRequest(json: unknown): Parsed<TranscribeRequest> {
  if (!isRecord(json)) return { ok: false, message: "Body must be an object" };
  if (!nonEmptyString(json.fileName, 200)) return { ok: false, message: "fileName is required" };
  if (!/\.wav$/i.test(json.fileName)) return { ok: false, message: "Only .wav recordings are supported" };
  if (!nonEmptyString(json.audioBase64, 40_000_000)) return { ok: false, message: "audioBase64 is required" };
  return { ok: true, value: { fileName: json.fileName, audioBase64: json.audioBase64 } };
}

export function parseTranscribeResponse(json: unknown): Parsed<TranscribeResponse> {
  if (!isRecord(json) || !isRecord(json.narration)) return { ok: false, message: "Missing narration" };
  const n = json.narration;
  if (typeof n.transcript !== "string" || !Array.isArray(n.segments)) {
    return { ok: false, message: "Narration is malformed" };
  }
  const segmentsOk = n.segments.every(
    (s) => isRecord(s) && typeof s.text === "string" && typeof s.startMs === "number" && typeof s.endMs === "number",
  );
  if (!segmentsOk || typeof json.model !== "string" || typeof json.sourceDigest !== "string") {
    return { ok: false, message: "Narration is malformed" };
  }
  return { ok: true, value: json as TranscribeResponse };
}

export function parseServiceError(json: unknown): ServiceError["error"] | null {
  if (isRecord(json) && isRecord(json.error) && typeof json.error.code === "string" && typeof json.error.message === "string") {
    return { code: json.error.code, message: json.error.message };
  }
  return null;
}

export type ReadScreenshotRequest = { fileName: string; imageBase64: string };
export type ReadScreenshotResponse = { sourceDigest: string; model: string; note: ScreenNote };

export function parseReadScreenshotRequest(json: unknown): Parsed<ReadScreenshotRequest> {
  if (!isRecord(json)) return { ok: false, message: "Body must be an object" };
  if (!nonEmptyString(json.fileName, 200)) return { ok: false, message: "fileName is required" };
  if (!imageExtension(json.fileName)) return { ok: false, message: "Only .png and .jpg screenshots are supported" };
  if (!nonEmptyString(json.imageBase64, 20_000_000)) return { ok: false, message: "imageBase64 is required" };
  return { ok: true, value: { fileName: json.fileName, imageBase64: json.imageBase64 } };
}

export function parseReadScreenshotResponse(json: unknown): Parsed<ReadScreenshotResponse> {
  if (!isRecord(json) || !isRecord(json.note) || typeof json.note.text !== "string" || typeof json.model !== "string") {
    return { ok: false, message: "Screen note is malformed" };
  }
  return { ok: true, value: json as ReadScreenshotResponse };
}

export type ProposeStepsRequest = { transcript: string; screenNotes: string[] };
export type ProposeStepsResponse = { model: string; raw: string; result: ProposalResult };

export function parseProposeStepsRequest(json: unknown): Parsed<ProposeStepsRequest> {
  if (!isRecord(json)) return { ok: false, message: "Body must be an object" };
  if (!nonEmptyString(json.transcript, 20_000)) return { ok: false, message: "transcript is required" };
  const notes = json.screenNotes ?? [];
  if (!Array.isArray(notes) || notes.length > 20 || !notes.every((n) => nonEmptyString(n, 1000))) {
    return { ok: false, message: "screenNotes must be up to 20 strings" };
  }
  return { ok: true, value: { transcript: json.transcript, screenNotes: notes as string[] } };
}

export function parseProposeStepsResponse(json: unknown): Parsed<ProposeStepsResponse> {
  if (!isRecord(json) || !isRecord(json.result) || !Array.isArray(json.result.proposals) || !Array.isArray(json.result.rejected)) {
    return { ok: false, message: "Proposal response is malformed" };
  }
  return { ok: true, value: json as ProposeStepsResponse };
}
