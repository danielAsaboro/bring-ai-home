We will put QVAC behind a small loopback service and call it from Pattern, without letting a proposal change anything yet.

Loading a model inside React would tie the model's lifecycle to page reloads and put native code in the browser bundle.
Instead we run a tiny HTTP service on `127.0.0.1:4319`.
The browser sends examples; the service answers with a proposal.
The service never stores patterns.

The app shows the raw proposal in a dashed panel marked **not applied**.
If the service is stopped, the panel says so and the manual workbench keeps working.

**Source checkpoint:** `86e219b` · **Starting checkpoint:** `f06cb75`

| File | Responsibility |
| --- | --- |
| `qvac-service/server.ts` | Loopback HTTP: origin check, size limit, request validation, 503 without a model |

## Build and observe

### 1. Start the service

```bash
git checkout 86e219b
npm run qvac:service
```

In a second terminal: `curl http://127.0.0.1:4319/health`.
It reports whether proposals are available.

### 2. Propose from the app

Run `npm run dev`, open **Contact names**, and press **Propose rules**.
The raw reply appears, marked not applied.
Your weave is unchanged.

### 3. Stop the service

Stop the service and press **Propose rules** again.
The panel says the local service is not running.
Every manual control still works.

### 4. Read the boundary

In `server.ts`, a request from a foreign origin gets 403, an empty example list gets 400, and a missing model gets 503.
The service only forwards `before` and `after`; ids and other fields are dropped.

```javascript
import { fileURLToPath } from "node:url";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const runtime = await import("./qvac-runtime.ts");
const { createPatternServer, SERVICE_PORT } = await import("./server.ts");
const { buildProposalHistory } = await import("./proposal.ts");

let sdk;
let modelId;
let proposer = null;
try {
  await runtime.configureLocalQvac(appDirectory);
  sdk = await import("@qvac/sdk");
  modelId = await runtime.loadProposalModel(sdk);
  proposer = async (examples) => ({
    raw: await runtime.completeText(sdk, modelId, buildProposalHistory(examples)),
    model: runtime.PROPOSAL_MODEL,
  });
} catch (error) {
  process.stderr.write(
    `Proposal model unavailable: ${error instanceof Error ? error.message : String(error)}\n` +
      "The service will answer 503 and Pattern keeps working by hand.\n",
  );
}

const server = createPatternServer({ proposer });
server.listen(SERVICE_PORT, "127.0.0.1", () => {
  process.stderr.write(`Pattern QVAC service on http://127.0.0.1:${SERVICE_PORT}\n`);
});

async function shutdown() {
  server.close();
  if (sdk && modelId) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  if (sdk) await sdk.close().catch(() => undefined);
  process.exit(0);
}
process.on("SIGINT", shutdown);
process.on("SIGTERM", shutdown);
```

```typescript
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";

export const SERVICE_PORT = 4319;
const MAX_BODY_BYTES = 32_768;
const MAX_EXAMPLES = 12;

export type ExamplePair = { before: string; after: string };
export type Proposer = (examples: ExamplePair[]) => Promise<{ raw: string; model: string }>;

export function isAllowedOrigin(origin: string | undefined): boolean {
  if (!origin) return true; // curl and local scripts send no Origin
  try {
    const url = new URL(origin);
    return url.protocol === "http:" && (url.hostname === "localhost" || url.hostname === "127.0.0.1");
  } catch {
    return false;
  }
}

export function parseExamples(value: unknown): ExamplePair[] {
  const examples = (value as { examples?: unknown } | null)?.examples;
  if (
    !Array.isArray(examples) ||
    examples.length === 0 ||
    examples.length > MAX_EXAMPLES ||
    !examples.every((e) => typeof e?.before === "string" && typeof e?.after === "string")
  ) {
    throw new Error(`Request must contain 1 to ${MAX_EXAMPLES} examples with before and after text`);
  }
  return examples.map((e: ExamplePair) => ({ before: e.before, after: e.after }));
}

function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  res.writeHead(status, {
    "content-type": "application/json",
    ...(origin ? { "access-control-allow-origin": origin, vary: "Origin" } : {}),
    "access-control-allow-headers": "content-type",
    "access-control-allow-methods": "GET, POST, OPTIONS",
  });
  res.end(status === 204 ? undefined : JSON.stringify(body));
}

async function readJson(req: IncomingMessage): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > MAX_BODY_BYTES) throw new Error("Request is too large");
    chunks.push(chunk as Buffer);
  }
  return JSON.parse(Buffer.concat(chunks).toString("utf8"));
}

// The service proposes. It never stores patterns. The browser decides.
export function createPatternServer(options: { proposer: Proposer | null }) {
  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (!isAllowedOrigin(origin)) return send(res, 403, { error: "Origin not allowed" });
    if (req.method === "OPTIONS") return send(res, 204, null, origin);
    if (req.method === "GET" && req.url === "/health") {
      return send(res, 200, { ok: true, proposing: options.proposer !== null }, origin);
    }
    if (req.method !== "POST" || req.url !== "/propose") return send(res, 404, { error: "Not found" }, origin);
    let examples: ExamplePair[];
    try {
      examples = parseExamples(await readJson(req));
    } catch (error) {
      return send(res, 400, { error: (error as Error).message }, origin);
    }
    if (!options.proposer) return send(res, 503, { error: "Rule proposals are not available" }, origin);
    try {
      return send(res, 200, await options.proposer(examples), origin);
    } catch (error) {
      return send(res, 502, { error: (error as Error).message }, origin);
    }
  });
}
```

```tsx
"use client";

import { useState } from "react";

import type { Example } from "./patterns";
import { requestProposal, type ProposalResponse } from "./proposal-client";

const button = "rounded border border-stone-400 px-2 py-1 text-sm hover:bg-stone-100 disabled:opacity-40";

export function ProposalPanel({
  examples,
  request = requestProposal,
}: {
  examples: Example[];
  request?: (examples: Example[]) => Promise<ProposalResponse>;
}) {
  const [busy, setBusy] = useState(false);
  const [result, setResult] = useState<ProposalResponse | null>(null);

  return (
    <section aria-label="Proposal" className="space-y-2 rounded border border-dashed border-amber-400 p-3">
      <h3 className="font-semibold">Propose a weave from examples</h3>
      <p className="text-xs text-stone-600">Runs on this machine through the local QVAC service. A proposal never changes your weave by itself.</p>
      <button
        className={button}
        disabled={busy || examples.length === 0}
        onClick={async () => {
          setBusy(true);
          setResult(await request(examples));
          setBusy(false);
        }}
      >
        {busy ? "Proposing…" : "Propose rules"}
      </button>
      {result?.status === "ok" && (
        <div>
          <p className="text-sm">Raw proposal from {result.model} (not applied):</p>
          <pre aria-label="Raw proposal" className="overflow-x-auto bg-stone-100 p-2 text-xs">{result.raw}</pre>
        </div>
      )}
      {result && result.status !== "ok" && <p role="status" className="text-sm text-rose-700">{result.message}</p>}
    </section>
  );
}
```

```typescript
import type { Example } from "./patterns.ts";

export const SERVICE_URL = "http://127.0.0.1:4319";

export type ProposalResponse =
  | { status: "ok"; raw: string; model: string }
  | { status: "unavailable"; message: string }
  | { status: "error"; message: string };

// Transport failures and model failures are reported differently. Neither changes a pattern.
export async function requestProposal(
  examples: readonly Example[],
  fetchImpl: typeof fetch = fetch,
): Promise<ProposalResponse> {
  let res: Response;
  try {
    res = await fetchImpl(`${SERVICE_URL}/propose`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ examples: examples.map(({ before, after }) => ({ before, after })) }),
    });
  } catch {
    return { status: "unavailable", message: "The local QVAC service is not running. Keep weaving by hand." };
  }
  const body = (await res.json().catch(() => ({}))) as { raw?: unknown; model?: unknown; error?: unknown };
  if (res.status === 503) return { status: "unavailable", message: String(body.error ?? "Proposals are not available") };
  if (!res.ok || typeof body.raw !== "string" || typeof body.model !== "string") {
    return { status: "error", message: String(body.error ?? `Service answered ${res.status}`) };
  }
  return { status: "ok", raw: body.raw, model: body.model };
}
```
