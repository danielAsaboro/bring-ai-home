We will embed saved patterns locally with GTE and suggest similar ones, without copying any rules between them.

When a new export arrives, you may already have a weave for something like it.
Local embeddings turn each pattern's name, description, and examples into a vector, and cosine similarity ranks the closest saved patterns.

Similarity only suggests where to look.
Opening a match selects that pattern; it never copies rules.

**Source checkpoint:** `d20c0cf` · **Starting checkpoint:** `e181a61`

## What you change

| File | Responsibility |
| --- | --- |
| `src/features/pattern/similar.ts` | Pattern text, cosine, and ranking with a threshold |
| `qvac-service/qvac-runtime.ts` | Load `GTE_LARGE_FP16` and embed a batch of texts |
| `qvac-service/server.ts` | `POST /embed` with validation and count checking |
| `src/features/pattern/proposal-client.ts` | `requestEmbeddings` validates every vector |
| `src/features/pattern/SimilarPanel.tsx` | Find similar patterns and open a match |

## Build and observe

### 1. Restart the service

```bash
git checkout d20c0cf
npm run qvac:service
```

The first start downloads the embedding model.
`/health` now reports `embedding: true`.

### 2. Find a neighbour

Open **Contact names** and press **Find similar patterns**.
Up to three matches appear with their scores and the model name.
Click one to open it.

### 3. Read the threshold

`rankSimilar` drops the pattern itself and anything under 0.5.
With only three sample patterns, "No saved pattern is close enough" is a valid answer.

import { mkdir, writeFile } from "node:fs/promises";
import path from "node:path";

import type { ChatMessage } from "./proposal.ts";

export const PROPOSAL_MODEL = "LLAMA_3_2_1B_INST_Q4_0";

export function resolveLocalQvacPaths(appDirectory: string) {
  const qvacDirectory = path.join(appDirectory, ".qvac");
  return {
    qvacDirectory,
    cacheDirectory: path.join(qvacDirectory, "models"),
    configPath: path.join(qvacDirectory, "config.json"),
  };
}

// Point the SDK at an app-local cache before @qvac/sdk is imported.
export async function configureLocalQvac(appDirectory: string) {
  const paths = resolveLocalQvacPaths(appDirectory);
  await mkdir(paths.cacheDirectory, { recursive: true });
  await writeFile(paths.configPath, `${JSON.stringify({ cacheDirectory: paths.cacheDirectory }, null, 2)}\n`, "utf8");
  process.env.QVAC_CONFIG_PATH = paths.configPath;
  return paths;
}

type Sdk = typeof import("@qvac/sdk");

export async function loadProposalModel(sdk: Sdk, onProgress?: (percentage: number) => void) {
  return sdk.loadModel({
    modelSrc: sdk.LLAMA_3_2_1B_INST_Q4_0,
    modelType: "llm",
    onProgress: onProgress ? (p) => onProgress(p.percentage) : undefined,
  });
}

export const EMBEDDING_MODEL = "GTE_LARGE_FP16";

export async function loadEmbeddingModel(sdk: Sdk) {
  return sdk.loadModel({ modelSrc: sdk.GTE_LARGE_FP16, modelType: "embeddings" });
}

export async function embedTexts(sdk: Sdk, modelId: string, texts: string[]): Promise<number[][]> {
  const { embedding } = await sdk.embed({ modelId, text: texts });
  return embedding;
}

export async function completeText(sdk: Sdk, modelId: string, history: ChatMessage[]): Promise<string> {
  const run = sdk.completion({ modelId, history, stream: true });
  let raw = "";
  for await (const token of run.tokenStream) raw += token;
  return raw;
}

import { fileURLToPath } from "node:url";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const runtime = await import("./qvac-runtime.ts");
const { createPatternServer, SERVICE_PORT } = await import("./server.ts");
const { buildProposalHistory } = await import("./proposal.ts");

let sdk;
let modelId;
let proposer = null;
let embedModelId;
let embedder = null;
try {
  await runtime.configureLocalQvac(appDirectory);
  sdk = await import("@qvac/sdk");
  modelId = await runtime.loadProposalModel(sdk);
  proposer = async (examples) => ({
    raw: await runtime.completeText(sdk, modelId, buildProposalHistory(examples)),
    model: runtime.PROPOSAL_MODEL,
  });
} catch (error) {
  process.stderr.write(
    `Proposal model unavailable: ${error instanceof Error ? error.message : String(error)}\n` +
      "The service will answer 503 and Pattern keeps working by hand.\n",
  );
}

try {
  sdk ??= await import("@qvac/sdk");
  embedModelId = await runtime.loadEmbeddingModel(sdk);
  embedder = async (texts) => ({
    vectors: await runtime.embedTexts(sdk, embedModelId, texts),
    model: runtime.EMBEDDING_MODEL,
  });
} catch (error) {
  process.stderr.write(`Embedding model unavailable: ${error instanceof Error ? error.message : String(error)}\n`);
}

const server = createPatternServer({ proposer, embedder });
server.listen(SERVICE_PORT, "127.0.0.1", () => {
  process.stderr.write(`Pattern QVAC service on http://127.0.0.1:${SERVICE_PORT}\n`);
});

async function shutdown() {
  server.close();
  if (sdk && embedModelId) await sdk.unloadModel({ modelId: embedModelId, clearStorage: false }).catch(() => undefined);
  if (sdk && modelId) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  if (sdk) await sdk.close().catch(() => undefined);
  process.exit(0);
}
process.on("SIGINT", shutdown);
process.on("SIGTERM", shutdown);

```typescript
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";

import { checkProposal, type ProposalCheck } from "../src/features/pattern/rule-schema.ts";
import { extractProposal } from "./proposal.ts";

export const SERVICE_PORT = 4319;
const MAX_BODY_BYTES = 32_768;
const MAX_EXAMPLES = 12;

export type ExamplePair = { before: string; after: string };
export type Embedder = (texts: string[]) => Promise<{ vectors: number[][]; model: string }>;
export type Proposer = (examples: ExamplePair[]) => Promise<{ raw: string; model: string }>;

export function isAllowedOrigin(origin: string | undefined): boolean {
  if (!origin) return true; // curl and local scripts send no Origin
  try {
    const url = new URL(origin);
    return url.protocol === "http:" && (url.hostname === "localhost" || url.hostname === "127.0.0.1");
  } catch {
    return false;
  }
}

export function parseExamples(value: unknown): ExamplePair[] {
  const examples = (value as { examples?: unknown } | null)?.examples;
  if (
    !Array.isArray(examples) ||
    examples.length === 0 ||
    examples.length > MAX_EXAMPLES ||
    !examples.every((e) => typeof e?.before === "string" && typeof e?.after === "string")
  ) {
    throw new Error(`Request must contain 1 to ${MAX_EXAMPLES} examples with before and after text`);
  }
  return examples.map((e: ExamplePair) => ({ before: e.before, after: e.after }));
}

function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  res.writeHead(status, {
    "content-type": "application/json",
    ...(origin ? { "access-control-allow-origin": origin, vary: "Origin" } : {}),
    "access-control-allow-headers": "content-type",
    "access-control-allow-methods": "GET, POST, OPTIONS",
  });
  res.end(status === 204 ? undefined : JSON.stringify(body));
}

async function readJson(req: IncomingMessage): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > MAX_BODY_BYTES) throw new Error("Request is too large");
    chunks.push(chunk as Buffer);
  }
  return JSON.parse(Buffer.concat(chunks).toString("utf8"));
}

export function gateReply(raw: string): ProposalCheck {
  try {
    return checkProposal(extractProposal(raw));
  } catch (error) {
    return { ok: false, reasons: [(error as Error).message] };
  }
}

// The service proposes. It never stores patterns. The browser decides.
export function parseTexts(value: unknown): string[] {
  const texts = (value as { texts?: unknown } | null)?.texts;
  if (!Array.isArray(texts) || texts.length === 0 || texts.length > 64 || !texts.every((t) => typeof t === "string" && t.length > 0)) {
    throw new Error("Request must contain 1 to 64 non-empty texts");
  }
  return texts as string[];
}

export function createPatternServer(options: { proposer: Proposer | null; embedder?: Embedder | null }) {
  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (!isAllowedOrigin(origin)) return send(res, 403, { error: "Origin not allowed" });
    if (req.method === "OPTIONS") return send(res, 204, null, origin);
    if (req.method === "GET" && req.url === "/health") {
      return send(res, 200, { ok: true, proposing: options.proposer !== null, embedding: Boolean(options.embedder) }, origin);
    }
    if (req.method === "POST" && req.url === "/embed") {
      let texts: string[];
      try {
        texts = parseTexts(await readJson(req));
      } catch (error) {
        return send(res, 400, { error: (error as Error).message }, origin);
      }
      if (!options.embedder) return send(res, 503, { error: "Similarity search is not available" }, origin);
      try {
        const { vectors, model } = await options.embedder(texts);
        if (vectors.length !== texts.length) throw new Error("Embedding count does not match text count");
        return send(res, 200, { vectors, model }, origin);
      } catch (error) {
        return send(res, 502, { error: (error as Error).message }, origin);
      }
    }
    if (req.method !== "POST" || req.url !== "/propose") return send(res, 404, { error: "Not found" }, origin);
    let examples: ExamplePair[];
    try {
      examples = parseExamples(await readJson(req));
    } catch (error) {
      return send(res, 400, { error: (error as Error).message }, origin);
    }
    if (!options.proposer) return send(res, 503, { error: "Rule proposals are not available" }, origin);
    try {
      const { raw, model } = await options.proposer(examples);
      return send(res, 200, { raw, model, check: gateReply(raw) }, origin);
    } catch (error) {
      return send(res, 502, { error: (error as Error).message }, origin);
    }
  });
}
```

"use client";

import { useState } from "react";

import type { Pattern } from "./patterns";
import { requestEmbeddings, type EmbedResponse } from "./proposal-client";
import { patternText, rankSimilar, type Match } from "./similar";

const button = "rounded border border-stone-400 px-2 py-1 text-sm hover:bg-stone-100 disabled:opacity-40";

export function SimilarPanel({
  patterns,
  targetId,
  onOpen,
  embed = requestEmbeddings,
}: {
  patterns: Pattern[];
  targetId: string;
  onOpen: (id: string) => void;
  embed?: (texts: string[]) => Promise<EmbedResponse>;
}) {
  const [state, setState] = useState<{ for: string; matches: Match[]; model: string } | { for: string; message: string } | null>(null);
  const current = state?.for === targetId ? state : null;

  return (
    <section aria-label="Similar patterns" className="space-y-2">
      <h3 className="font-semibold">Similar saved patterns</h3>
      <button
        className={button}
        onClick={async () => {
          const result = await embed(patterns.map(patternText));
          if (result.status !== "ok") return setState({ for: targetId, message: result.message });
          const matches = rankSimilar(patterns, result.vectors, targetId, { limit: 3, minScore: 0.5 });
          setState({ for: targetId, matches, model: result.model });
        }}
      >
        Find similar patterns
      </button>
      {current && "message" in current && <p role="status" className="text-sm text-rose-700">{current.message}</p>}
      {current && "matches" in current && (
        <ul aria-label="Similar matches" className="text-sm">
          {current.matches.length === 0 && <li>No saved pattern is close enough.</li>}
          {current.matches.map((m) => (
            <li key={m.id}>
              <button className="underline" onClick={() => onOpen(m.id)}>{m.name}</button> · {m.score.toFixed(2)} ({current.model})
            </li>
          ))}
        </ul>
      )}
    </section>
  );
}

function RuleFields({ rule, onChange }: { rule: Rule; onChange: (rule: Rule) => void }) {
  switch (rule.kind) {
    case "trim":
      return (
        <label className="text-sm">
          <input
            type="checkbox"
            checked={rule.collapse}
            onChange={(e) => onChange({ ...rule, collapse: e.target.checked })}
          />{" "}
          collapse inner spaces
        </label>
      );
    case "split":
      return (
        <span className="flex flex-wrap gap-2 text-sm">
          <label>
            separator <input aria-label="separator" className={`${input} w-12`} value={rule.separator} onChange={(e) => onChange({ ...rule, separator: e.target.value })} />
          </label>
          <label>
            keep parts{" "}
            <input
              aria-label="keep parts"
              className={`${input} w-20`}
              value={rule.order.map((i) => i + 1).join(",")}
              onChange={(e) =>
                onChange({
                  ...rule,
                  order: e.target.value
                    .split(",")
                    .map((s) => Number.parseInt(s.trim(), 10) - 1)
                    .filter((n) => Number.isInteger(n) && n >= 0),
                })
              }
            />
          </label>
          <label>
            join <input aria-label="join" className={`${input} w-12`} value={rule.join} onChange={(e) => onChange({ ...rule, join: e.target.value })} />
          </label>
        </span>
      );
    case "case":
      return (
        <select aria-label="case" className={input} value={rule.to} onChange={(e) => onChange({ ...rule, to: e.target.value as typeof rule.to })}>
          <option value="upper">upper</option>
          <option value="lower">lower</option>
          <option value="title">title</option>
        </select>
      );
    case "replace":
      return (
        <span className="flex flex-wrap gap-2 text-sm">
          <label>
            find <input aria-label="find" className={`${input} w-24`} value={rule.find} onChange={(e) => onChange({ ...rule, find: e.target.value })} />

          </label>
        </span>
      );
    case "format":
      return (
        <label className="text-sm">
          template <input aria-label="template" className={`${input} w-40`} value={rule.template} onChange={(e) => onChange({ ...rule, template: e.target.value })} />
        </label>
      );
  }
}

export function Workbench({ store }: { store?: PatternStore }) {
  const storeRef = useRef<PatternStore | null>(store ?? null);
  const [patterns, dispatch] = useReducer(reduce, samplePatterns);
  const hydration = useRef<"waiting" | "loaded" | "ready">("waiting");
  const [selectedId, setSelectedId] = useState(samplePatterns[0].id);
  const [newKind, setNewKind] = useState<RuleKind>("replace");
  const [draftBefore, setDraftBefore] = useState("");
  const [draftAfter, setDraftAfter] = useState("");
  const [sample, setSample] = useState("");
  const [note, setNote] = useState("");
  const [isCounter, setIsCounter] = useState(false);

  useEffect(() => {
    storeRef.current ??= createLocalStore(window.localStorage);
    dispatch({ type: "replace-all", patterns: storeRef.current.load() });
    hydration.current = "loaded";

    // Skip the render that still shows the built-in samples, then save every change.
    if (hydration.current === "loaded") hydration.current = "ready";
    else if (hydration.current === "ready") storeRef.current?.save(patterns);
  }, [patterns]);

  const pattern: Pattern = patterns.find((p) => p.id === selectedId) ?? patterns[0];
  const results = useMemo(() => checkExamples(pattern.rules, pattern.examples), [pattern]);
  const passed = results.filter((r) => r.passed).length;
  const counterResults = checkExamples(pattern.rules, pattern.counterexamples ?? []);

  return (

      <aside aria-label="Saved patterns" className="space-y-2">
        <h2 className="font-semibold">Saved patterns</h2>

                aria-current={p.id === id}
                onClick={() => setSelectedId(p.id)}
              >
                {p.name}

          }}
        >
          Reset demo data

          <p className="text-sm text-stone-600">{pattern.description}</p>

          <ol className="space-y-1 font-mono text-xs">

        </section>

import type { Example } from "./patterns.ts";
import { checkProposal, type ProposalCheck } from "./rule-schema.ts";

export const SERVICE_URL = "http://127.0.0.1:4319";

export type ProposalResponse =
  | { status: "ok"; raw: string; model: string; check: ProposalCheck }
  | { status: "unavailable"; message: string }
  | { status: "error"; message: string };

// Transport failures and model failures are reported differently. Neither changes a pattern.
export async function requestProposal(
  examples: readonly Example[],
  fetchImpl: typeof fetch = fetch,
): Promise<ProposalResponse> {
  let res: Response;
  try {
    res = await fetchImpl(`${SERVICE_URL}/propose`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ examples: examples.map(({ before, after }) => ({ before, after })) }),
    });
  } catch {
    return { status: "unavailable", message: "The local QVAC service is not running. Keep weaving by hand." };
  }
  const body = (await res.json().catch(() => ({}))) as {
    raw?: unknown;
    model?: unknown;
    error?: unknown;
    check?: { ok?: unknown; rules?: unknown; reasons?: unknown };
  };
  if (res.status === 503) return { status: "unavailable", message: String(body.error ?? "Proposals are not available") };
  if (!res.ok || typeof body.raw !== "string" || typeof body.model !== "string") {
    return { status: "error", message: String(body.error ?? `Service answered ${res.status}`) };
  }
  // Re-check in the browser: the app never trusts a rule it did not validate itself.
  const check: ProposalCheck =
    body.check?.ok === true
      ? checkProposal(body.check.rules)
      : { ok: false, reasons: Array.isArray(body.check?.reasons) ? body.check.reasons.map(String) : ["Proposal was not checked"] };
  return { status: "ok", raw: body.raw, model: body.model, check };
}

export type EmbedResponse = { status: "ok"; vectors: number[][]; model: string } | { status: "unavailable" | "error"; message: string };

export async function requestEmbeddings(texts: string[], fetchImpl: typeof fetch = fetch): Promise<EmbedResponse> {
  let res: Response;
  try {
    res = await fetchImpl(`${SERVICE_URL}/embed`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ texts }),
    });
  } catch {
    return { status: "unavailable", message: "The local QVAC service is not running." };
  }
  const body = (await res.json().catch(() => ({}))) as { vectors?: unknown; model?: unknown; error?: unknown };
  if (res.status === 503) return { status: "unavailable", message: String(body.error ?? "Similarity is not available") };
  const vectors = body.vectors;
  if (
    !res.ok ||
    typeof body.model !== "string" ||
    !Array.isArray(vectors) ||
    vectors.length !== texts.length ||
    !vectors.every((v) => Array.isArray(v) && v.every((n) => typeof n === "number" && Number.isFinite(n)))
  ) {
    return { status: "error", message: String(body.error ?? `Service answered ${res.status}`) };
  }
  return { status: "ok", vectors: vectors as number[][], model: body.model };
}

import type { Pattern } from "./patterns.ts";

// The text that represents a saved pattern. Built from what a person wrote.
export function patternText(p: Pick<Pattern, "name" | "description" | "examples">): string {
  return [p.name, p.description, ...p.examples.map((e) => `${e.before} -> ${e.after}`)].join("\n");
}

export function cosine(a: readonly number[], b: readonly number[]): number {
  if (a.length !== b.length || a.length === 0) throw new Error("Vectors must have the same non-zero length");
  let dot = 0;
  let na = 0;
  let nb = 0;
  for (let i = 0; i < a.length; i++) {
    dot += a[i] * b[i];
    na += a[i] * a[i];
    nb += b[i] * b[i];
  }
  return na === 0 || nb === 0 ? 0 : dot / Math.sqrt(na * nb);
}

export type Match = { id: string; name: string; score: number };

// Similarity only suggests where to look. It never copies rules between patterns.
export function rankSimilar(
  patterns: readonly Pick<Pattern, "id" | "name">[],
  vectors: readonly number[][],
  targetId: string,
  options: { limit: number; minScore: number },
): Match[] {
  if (patterns.length !== vectors.length) throw new Error("Each pattern needs exactly one vector");
  const target = patterns.findIndex((p) => p.id === targetId);
  if (target === -1) throw new Error(`No pattern ${targetId}`);
  return patterns
    .map((p, i) => ({ id: p.id, name: p.name, score: cosine(vectors[target], vectors[i]) }))
    .filter((m) => m.id !== targetId && m.score >= options.minScore)
    .sort((a, b) => b.score - a.score)
    .slice(0, options.limit);
}
