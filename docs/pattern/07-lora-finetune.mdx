We will train a small LoRA adapter on approved weaves with the SDK's finetune API, then serve proposals from the tuned model.

The pinned SDK, `@qvac/sdk` 0.22.0, exports `finetune()` (see `dist/src/client/api/finetune.d.ts`): it trains LoRA adapter weights on a loaded GGUF model from a JSONL chat dataset, streams loss, and writes the adapter to `outputParametersDir`. The adapter loads again through `modelConfig.lora`.

Training data must be trustworthy. We build it only from weaves a person kept **and** that replay every example and counterexample. The incomplete invoice weave and the empty vendor weave are left out. A tiny fictional set of four more approved weaves completes the dataset.

Not every model or quantization can be fine-tuned. We use `QWEN3_600M_INST_Q4`, the model in the SDK's own fine-tuning example, as the base for the tuned proposer.

## Build and observe

### 1. Inspect the dataset

```bash
git checkout 675f040
npm test -- qvac-service/finetune-data.test.ts
```

Five records pass the gate: Contact names plus the four fixture weaves. Four train, one validates.

### 2. Train

```bash
npm run qvac:finetune
```

Each line shows epoch, step, train or val, and loss. The result lists files written to `.qvac/lora/pattern`. Training needs real memory and time; on a small machine use fewer epochs.

### 3. Serve the tuned model

```bash
PATTERN_LORA_ADAPTER="$PWD/.qvac/lora/pattern/<adapter>.gguf" npm run qvac:service
```

Proposals now report `QWEN3_600M_INST_Q4+LoRA`. They still pass through the same gate and counterexamples. Fine-tuning changes what is proposed, never who decides.

### 4. Be honest about five examples

Four training records cannot teach a model much. This chapter teaches the workflow. Measure the tuned model against the base on your own held-out patterns before you trust it.

```typescript
import { replayGate, type Pattern } from "../src/features/pattern/patterns.ts";
import { buildProposalHistory, type ChatMessage } from "./proposal.ts";

export type TrainingRecord = { messages: ChatMessage[] };

// Only weaves a person kept, and that replay every saved case, become training data.
export function buildTrainingRecords(patterns: readonly Pattern[]): TrainingRecord[] {
  return patterns
    .filter((p) => p.rules.length > 0 && p.examples.length > 0 && replayGate(p.rules, p).ok)
    .map((p) => ({
      messages: [...buildProposalHistory(p.examples), { role: "assistant", content: JSON.stringify(p.rules) }],
    }));
}

export function toJsonl(records: readonly TrainingRecord[]): string {
  return records.map((r) => JSON.stringify(r)).join("\n") + "\n";
}

// Hold the last record out for validation so the run reports an honest eval loss.
export function splitTrainEval(records: readonly TrainingRecord[]) {
  if (records.length < 2) throw new Error("Fine-tuning needs at least two approved patterns");
  return { train: records.slice(0, -1), eval: records.slice(-1) };
}
```

<Accordion title="qvac-service/finetune.mjs">

```javascript
// Train a small LoRA adapter on your approved weaves. Everything stays on this machine.
import { mkdir, readdir, writeFile } from "node:fs/promises";
import path from "node:path";
import { fileURLToPath } from "node:url";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const runtime = await import("./qvac-runtime.ts");
const { buildTrainingRecords, splitTrainEval, toJsonl } = await import("./finetune-data.ts");
const { samplePatterns } = await import("../src/features/pattern/patterns.ts");
const { approvedFixturePatterns } = await import("./test-data/finetune-patterns.ts");

let sdk;
let modelId;
try {
  const { qvacDirectory } = await runtime.configureLocalQvac(appDirectory);
  const dataDir = path.join(qvacDirectory, "finetune");
  const output = path.join(qvacDirectory, "lora", "pattern");
  const checkpoints = path.join(output, "checkpoints");
  await mkdir(checkpoints, { recursive: true });

  const { train, eval: held } = splitTrainEval(buildTrainingRecords([...samplePatterns, ...approvedFixturePatterns]));
  const trainPath = path.join(dataDir, "train.jsonl");
  const evalPath = path.join(dataDir, "eval.jsonl");
  await mkdir(dataDir, { recursive: true });
  await writeFile(trainPath, toJsonl(train), "utf8");
  await writeFile(evalPath, toJsonl(held), "utf8");
  process.stderr.write(`Training on ${train.length} approved weaves, validating on ${held.length}\n`);

  sdk = await import("@qvac/sdk");
  modelId = await runtime.loadTuningBase(sdk);
  const run = runtime.startLoraRun(sdk, modelId, { train: trainPath, eval: evalPath, output, checkpoints });
  for await (const tick of run.progressStream) {
    process.stderr.write(`epoch ${tick.current_epoch + 1} step ${tick.global_steps} ${tick.is_train ? "train" : "val"} loss ${tick.loss?.toFixed(4)}\n`);
  }
  const result = await run.result;
  const adapters = (await readdir(output)).filter((name) => name.endsWith(".gguf"));
  process.stdout.write(`${JSON.stringify({ kind: "lora-run", base: runtime.TUNED_BASE_MODEL, result, output, adapters }, null, 2)}\n`);
} catch (error) {
  process.stderr.write(`LoRA fine-tune failed: ${error instanceof Error ? error.message : String(error)}\n`);
  process.exitCode = 1;
} finally {
  if (sdk && modelId) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  if (sdk) await sdk.close().catch(() => undefined);
}
```

<Accordion title="qvac-service/qvac-runtime.ts">

```typescript
import { mkdir, writeFile } from "node:fs/promises";
import path from "node:path";

import type { ChatMessage } from "./proposal.ts";

export const PROPOSAL_MODEL = "LLAMA_3_2_1B_INST_Q4_0";

export function resolveLocalQvacPaths(appDirectory: string) {
  const qvacDirectory = path.join(appDirectory, ".qvac");
  return {
    qvacDirectory,
    cacheDirectory: path.join(qvacDirectory, "models"),
    configPath: path.join(qvacDirectory, "config.json"),
  };
}

// Point the SDK at an app-local cache before @qvac/sdk is imported.
export async function configureLocalQvac(appDirectory: string) {
  const paths = resolveLocalQvacPaths(appDirectory);
  await mkdir(paths.cacheDirectory, { recursive: true });
  await writeFile(paths.configPath, `${JSON.stringify({ cacheDirectory: paths.cacheDirectory }, null, 2)}\n`, "utf8");
  process.env.QVAC_CONFIG_PATH = paths.configPath;
  return paths;
}

type Sdk = typeof import("@qvac/sdk");

// The LoRA base must be fine-tunable. We use the SDK's own fine-tuning example model.
export const TUNED_BASE_MODEL = "QWEN3_600M_INST_Q4";

export function proposalModelLabel(adapterPath: string | null): string {
  return adapterPath ? `${TUNED_BASE_MODEL}+LoRA` : PROPOSAL_MODEL;
}

export async function loadProposalModel(sdk: Sdk, onProgress?: (percentage: number) => void, adapterPath: string | null = null) {
  if (adapterPath) {
    return sdk.loadModel({
      modelSrc: sdk.QWEN3_600M_INST_Q4,
      modelType: "llm",
      modelConfig: { ctx_size: 1024, lora: adapterPath },
    });
  }
  return sdk.loadModel({
    modelSrc: sdk.LLAMA_3_2_1B_INST_Q4_0,
    modelType: "llm",
    onProgress: onProgress ? (p) => onProgress(p.percentage) : undefined,
  });
}

export const EMBEDDING_MODEL = "GTE_LARGE_FP16";

export async function loadEmbeddingModel(sdk: Sdk) {
  return sdk.loadModel({ modelSrc: sdk.GTE_LARGE_FP16, modelType: "embeddings" });
}

export async function embedTexts(sdk: Sdk, modelId: string, texts: string[]): Promise<number[][]> {
  const { embedding } = await sdk.embed({ modelId, text: texts });
  return embedding;
}

export async function completeText(sdk: Sdk, modelId: string, history: ChatMessage[]): Promise<string> {
  const run = sdk.completion({ modelId, history, stream: true });
  let raw = "";
  for await (const token of run.tokenStream) raw += token;
  return raw;
}

export async function loadTuningBase(sdk: Sdk) {
  return sdk.loadModel({ modelSrc: sdk.QWEN3_600M_INST_Q4, modelType: "llm", modelConfig: { ctx_size: 512 } });
}

export function startLoraRun(
  sdk: Sdk,
  modelId: string,
  dirs: { train: string; eval: string; output: string; checkpoints: string },
) {
  return sdk.finetune({
    modelId,
    options: {
      trainDatasetDir: dirs.train,
      validation: { type: "dataset", path: dirs.eval },
      outputParametersDir: dirs.output,
      checkpointSaveDir: dirs.checkpoints,
      numberOfEpochs: 2,
      learningRate: 1e-4,
      assistantLossOnly: true,
      loraSeed: 7,
    },
  });
}
```

<Accordion title="qvac-service/serve.mjs">

```javascript
import { fileURLToPath } from "node:url";

const appDirectory = fileURLToPath(new URL("../", import.meta.url));
const runtime = await import("./qvac-runtime.ts");
const { createPatternServer, SERVICE_PORT } = await import("./server.ts");
const { buildProposalHistory } = await import("./proposal.ts");

let sdk;
let modelId;
let proposer = null;
let embedModelId;
let embedder = null;
try {
  await runtime.configureLocalQvac(appDirectory);
  sdk = await import("@qvac/sdk");
  // Set PATTERN_LORA_ADAPTER to an adapter written by qvac:finetune to use the tuned model.
  const adapterPath = process.env.PATTERN_LORA_ADAPTER || null;
  modelId = await runtime.loadProposalModel(sdk, undefined, adapterPath);
  const model = runtime.proposalModelLabel(adapterPath);
  proposer = async (examples) => ({
    raw: await runtime.completeText(sdk, modelId, buildProposalHistory(examples)),
    model,
  });
} catch (error) {
  process.stderr.write(
    `Proposal model unavailable: ${error instanceof Error ? error.message : String(error)}\n` +
      "The service will answer 503 and Pattern keeps working by hand.\n",
  );
}

try {
  sdk ??= await import("@qvac/sdk");
  embedModelId = await runtime.loadEmbeddingModel(sdk);
  embedder = async (texts) => ({
    vectors: await runtime.embedTexts(sdk, embedModelId, texts),
    model: runtime.EMBEDDING_MODEL,
  });
} catch (error) {
  process.stderr.write(`Embedding model unavailable: ${error instanceof Error ? error.message : String(error)}\n`);
}

const server = createPatternServer({ proposer, embedder });
server.listen(SERVICE_PORT, "127.0.0.1", () => {
  process.stderr.write(`Pattern QVAC service on http://127.0.0.1:${SERVICE_PORT}\n`);
});

async function shutdown() {
  server.close();
  if (sdk && embedModelId) await sdk.unloadModel({ modelId: embedModelId, clearStorage: false }).catch(() => undefined);
  if (sdk && modelId) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
  if (sdk) await sdk.close().catch(() => undefined);
  process.exit(0);
}
process.on("SIGINT", shutdown);
process.on("SIGTERM", shutdown);
```
