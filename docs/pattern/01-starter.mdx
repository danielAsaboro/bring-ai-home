We will build Pattern, a small reversible transformation workbench, and finish its manual job before any AI arrives.

People clean the same messy exports every week: names typed backwards, invoice codes with stray spaces, vendor names in capitals. Spreadsheet formulas and scripts feel like too much for the job, so it gets done by hand, again.

Pattern is a workbench for that job, built like a weaver's loom. **Examples** are threads: a before value and the after value you want. **Rules** are the weave: ordered, deterministic steps (trim, split, case, replace, format). **Outputs** are swatches you compare against the examples. Every saved weave becomes a **revision** you can restore.

This chapter builds the whole manual product. There is no AI in it, on purpose. Later chapters add local AI one step at a time, and the app must keep working without it.

Open `http://localhost:3000`.

### 2. Finish a weave by hand

Select **Invoice codes**. The heading reads **0 of 3 match**: the weave trims and upper-cases, but `" inv 0042 "` becomes `INV 0042`, not `INV-0042`. Add a **replace** rule with find set to one space and with set to `-`. The heading changes to **3 of 3 match**.

### 3. Watch each step

Type `inv 7` into **Try a sample**. Each rule's before and after is listed, ending in `INV-7`. The sample is never changed in place; the weave produces a new value.

### 4. Save and restore

Save a revision with a note, remove a rule, then press **Restore** on your revision. The weave comes back exactly. **Reset demo data** returns the three fictional patterns.

### 5. Write down the rule for this course

> QVAC may propose rules and similar patterns. The app decides what is a valid, passing weave. Only a person applies one.

// The closed rule vocabulary. Every rule is plain data, readable, and deterministic.
export type Rule =
  | { kind: "trim"; collapse: boolean }
  | { kind: "split"; separator: string; order: number[]; join: string }
  | { kind: "case"; to: "upper" | "lower" | "title" }
  | { kind: "replace"; find: string; with: string }
  | { kind: "format"; template: string };

export type RuleKind = Rule["kind"];

export const RULE_KINDS: RuleKind[] = ["trim", "split", "case", "replace", "format"];

export function defaultRule(kind: RuleKind): Rule {
  switch (kind) {
    case "trim":
      return { kind, collapse: true };
    case "split":
      return { kind, separator: ",", order: [1, 0], join: " " };
    case "case":
      return { kind, to: "title" };
    case "replace":
      return { kind, find: "", with: "" };
    case "format":
      return { kind, template: "{value}" };
  }
}

function titleCase(value: string): string {
  return value.toLowerCase().replace(/(^|[\s-])(\p{L})/gu, (_, lead: string, ch: string) => lead + ch.toUpperCase());
}

export function applyRule(rule: Rule, value: string): string {
  switch (rule.kind) {
    case "trim": {
      const trimmed = value.trim();
      return rule.collapse ? trimmed.replace(/\s+/g, " ") : trimmed;
    }
    case "split": {
      if (rule.separator === "") return value;
      const parts = value.split(rule.separator).map((p) => p.trim());
      return rule.order
        .map((i) => parts[i])
        .filter((p): p is string => p !== undefined && p !== "")
        .join(rule.join);
    }
    case "case":
      if (rule.to === "upper") return value.toUpperCase();
      if (rule.to === "lower") return value.toLowerCase();
      return titleCase(value);
    case "replace":
      return rule.find === "" ? value : value.split(rule.find).join(rule.with);
    case "format":
      return rule.template.split("{value}").join(value);
  }
}

export type WeaveStep = { rule: Rule; before: string; after: string };

// Run an ordered weave. The input string is never modified; every step is kept.
export function runWeave(rules: readonly Rule[], input: string): { output: string; steps: WeaveStep[] } {
  const steps: WeaveStep[] = [];
  let value = input;
  for (const rule of rules) {
    const after = applyRule(rule, value);
    steps.push({ rule, before: value, after });
    value = after;
  }
  return { output: value, steps };
}

const show = (s: string) => JSON.stringify(s);

export function describeRule(rule: Rule): string {
  switch (rule.kind) {
    case "trim":
      return rule.collapse ? "Trim the ends and collapse inner spaces" : "Trim the ends";
    case "split":
      return `Split on ${show(rule.separator)}, keep parts ${rule.order.map((i) => i + 1).join(", ")}, join with ${show(rule.join)}`;
    case "case":
      return `Change to ${rule.to} case`;
    case "replace":
      return `Replace ${show(rule.find)} with ${show(rule.with)}`;
    case "format":
      return `Format as ${show(rule.template)}`;
  }
}

import { samplePatterns, type Pattern } from "./patterns";

// One adapter owns persistence. Nothing else touches localStorage.
export type PatternStore = {
  load(): Pattern[];
  save(patterns: Pattern[]): void;
  reset(): Pattern[];
};

const KEY = "pattern.v1.patterns";

const fresh = (): Pattern[] => structuredClone(samplePatterns);

export function createMemoryStore(initial: Pattern[] = fresh()): PatternStore {
  let patterns = structuredClone(initial);
  return {
    load: () => structuredClone(patterns),
    save: (next) => {
      patterns = structuredClone(next);
    },
    reset: () => {
      patterns = fresh();
      return structuredClone(patterns);
    },
  };
}

export function createLocalStore(storage: Storage): PatternStore {
  return {
    load() {
      try {
        const raw = storage.getItem(KEY);
        if (!raw) return fresh();
        const parsed = JSON.parse(raw) as unknown;
        return Array.isArray(parsed) ? (parsed as Pattern[]) : fresh();
      } catch {
        return fresh();
      }
    },
    save(patterns) {
      try {
        storage.setItem(KEY, JSON.stringify(patterns));
      } catch {
        // Storage can be full or blocked; the session keeps working in memory.
      }
    },
    reset() {
      try {
        storage.removeItem(KEY);
      } catch {
        // ignore
      }
      return fresh();
    },
  };
}

import type { Example, Pattern } from "./patterns";
import type { Rule } from "./rules";

export type Action =
  | { type: "add-rule"; patternId: string; rule: Rule }
  | { type: "update-rule"; patternId: string; index: number; rule: Rule }
  | { type: "move-rule"; patternId: string; index: number; by: -1 | 1 }
  | { type: "remove-rule"; patternId: string; index: number }
  | { type: "add-example"; patternId: string; example: Example }
  | { type: "remove-example"; patternId: string; exampleId: string }
  | { type: "save-revision"; patternId: string; id: string; savedAt: string; note: string }
  | { type: "restore-revision"; patternId: string; revisionId: string }
  | { type: "replace-all"; patterns: Pattern[] };

function edit(patterns: Pattern[], id: string, change: (p: Pattern) => Pattern): Pattern[] {
  return patterns.map((p) => (p.id === id ? change(p) : p));
}

// Every change returns new objects. Saved revisions are never edited in place.
export function reduce(patterns: Pattern[], action: Action): Pattern[] {
  switch (action.type) {
    case "replace-all":
      return action.patterns;
    case "add-rule":
      return edit(patterns, action.patternId, (p) => ({ ...p, rules: [...p.rules, action.rule] }));
    case "update-rule":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: p.rules.map((r, i) => (i === action.index ? action.rule : r)),
      }));
    case "move-rule":
      return edit(patterns, action.patternId, (p) => {
        const to = action.index + action.by;
        if (to < 0 || to >= p.rules.length) return p;
        const rules = [...p.rules];
        [rules[action.index], rules[to]] = [rules[to], rules[action.index]];
        return { ...p, rules };
      });
    case "remove-rule":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: p.rules.filter((_, i) => i !== action.index),
      }));
    case "add-example":
      return edit(patterns, action.patternId, (p) => ({ ...p, examples: [...p.examples, action.example] }));
    case "remove-example":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        examples: p.examples.filter((e) => e.id !== action.exampleId),
      }));
    case "save-revision":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        revisions: [
          ...p.revisions,
          { id: action.id, savedAt: action.savedAt, note: action.note, rules: structuredClone(p.rules) },
        ],
      }));
    case "restore-revision":
      return edit(patterns, action.patternId, (p) => {
        const revision = p.revisions.find((r) => r.id === action.revisionId);
        return revision ? { ...p, rules: structuredClone(revision.rules) } : p;
      });
  }
}
