We will keep held-out counterexamples away from the model and refuse any proposal that does not replay every saved case.

Three examples are easy to overfit. A weave that turns `" inv 0042 "` into `INV-0042` by stripping `INV ` and formatting `INV-{value}` matches every example, then turns an already clean `INV-0042` into `INV-INV-0042`.

Counterexamples are saved cases the model never sees. The app replays the proposal against examples and counterexamples together. If any case fails, the panel lists it (input, actual, expected) and **Apply as new revision** is disabled. A person can still fix the weave by hand.

"use client";

import { useState } from "react";

import type { Example, GateResult } from "./patterns";
import { requestProposal, type ProposalResponse } from "./proposal-client";
import { describeRule, type Rule } from "./rules";

const button = "rounded border border-stone-400 px-2 py-1 text-sm hover:bg-stone-100 disabled:opacity-40";

export function ProposalPanel({
  examples,
  onApply,
  gate,
  request = requestProposal,
}: {
  examples: Example[];
  gate: (rules: Rule[]) => GateResult;
  onApply: (rules: Rule[], model: string) => void;
  request?: (examples: Example[]) => Promise<ProposalResponse>;
}) {
  const [busy, setBusy] = useState(false);
  const [result, setResult] = useState<ProposalResponse | null>(null);

  return (
    <section aria-label="Proposal" className="space-y-2 rounded border border-dashed border-amber-400 p-3">
      <h3 className="font-semibold">Propose a weave from examples</h3>
      <p className="text-xs text-stone-600">Runs on this machine through the local QVAC service. A proposal never changes your weave until you apply it, and applying it saves your current weave first.</p>
      <button
        className={button}
        disabled={busy || examples.length === 0}
        onClick={async () => {
          setBusy(true);
          setResult(await request(examples));
          setBusy(false);
        }}
      >
        {busy ? "Proposing…" : "Propose rules"}
      </button>
      {result?.status === "ok" && result.check.ok && (
        <div className="space-y-2">
          <p className="text-sm">Proposed by {result.model}:</p>
          <ol aria-label="Proposed rules" className="list-decimal pl-5 text-sm">
            {result.check.rules.map((rule, i) => (
              <li key={i}>{describeRule(rule)}</li>
            ))}
          </ol>
          {(() => {
            const verdict = gate(result.check.rules);
            return verdict.ok ? (
              <p className="text-sm text-emerald-700">Replays every example and counterexample.</p>
            ) : (
              <div role="alert" className="text-sm text-rose-700">
                <p>This proposal fails {verdict.failures.length} saved case(s). Fix the weave by hand or ask again.</p>
                <ul className="list-disc pl-5 font-mono text-xs">
                  {verdict.failures.map((f) => (
                    <li key={f.example.id}>
                      {JSON.stringify(f.example.before)} → {JSON.stringify(f.actual)}, expected {JSON.stringify(f.example.after)}
                    </li>
                  ))}
                </ul>
              </div>
            );
          })()}
          <div className="flex gap-2">
            <button
              className={button}
              disabled={!gate(result.check.rules).ok}
              onClick={() => {
                if (result.check.ok) onApply(result.check.rules, result.model);
                setResult(null);
              }}
            >
              Apply as new revision
            </button>
            <button className={button} onClick={() => setResult(null)}>Discard</button>
          </div>
        </div>
      )}
      {result?.status === "ok" && !result.check.ok && (
        <div role="status" className="text-sm text-rose-700">
          <p>The proposal was not readable as Pattern rules. Nothing was applied.</p>
          <ul className="list-disc pl-5">
            {result.check.reasons.map((r) => (
              <li key={r}>{r}</li>
            ))}
          </ul>
          <details>
            <summary>Raw reply</summary>
            <pre className="overflow-x-auto bg-stone-100 p-2 text-xs">{result.raw}</pre>
          </details>
        </div>
      )}
      {result && result.status !== "ok" && <p role="status" className="text-sm text-rose-700">{result.message}</p>}
    </section>
  );
}

import { runWeave, type Rule } from "./rules.ts";

export type Example = { id: string; before: string; after: string };

export type Revision = { id: string; savedAt: string; note: string; rules: Rule[] };

export type Pattern = {
  id: string;
  name: string;
  description: string;
  examples: Example[];
  /** Held-out cases that are never sent to the model. Optional so older saved data still loads. */
  counterexamples?: Example[];
  rules: Rule[];
  revisions: Revision[];
};

export type ExampleResult = { example: Example; actual: string; passed: boolean };

export function checkExamples(rules: readonly Rule[], examples: readonly Example[]): ExampleResult[] {
  return examples.map((example) => {
    const actual = runWeave(rules, example.before).output;
    return { example, actual, passed: actual === example.after };
  });
}

export type GateResult = { ok: boolean; failures: ExampleResult[] };

// The app decides: a weave must reproduce every example and every counterexample.
export function replayGate(rules: readonly Rule[], pattern: Pick<Pattern, "examples" | "counterexamples">): GateResult {
  const all = [...pattern.examples, ...(pattern.counterexamples ?? [])];
  const failures = checkExamples(rules, all).filter((r) => !r.passed);
  return { ok: failures.length === 0, failures };
}

import type { Example, Pattern } from "./patterns";
import type { Rule } from "./rules";

export type Action =
  | { type: "add-rule"; patternId: string; rule: Rule }
  | { type: "update-rule"; patternId: string; index: number; rule: Rule }
  | { type: "move-rule"; patternId: string; index: number; by: -1 | 1 }
  | { type: "remove-rule"; patternId: string; index: number }
  | { type: "add-example"; patternId: string; example: Example; counter?: boolean }
  | { type: "remove-example"; patternId: string; exampleId: string }
  | { type: "save-revision"; patternId: string; id: string; savedAt: string; note: string }
  | { type: "restore-revision"; patternId: string; revisionId: string }
  | { type: "replace-all"; patterns: Pattern[] }
  | { type: "apply-proposal"; patternId: string; rules: Rule[]; model: string; ids: [string, string]; savedAt: string };

function edit(patterns: Pattern[], id: string, change: (p: Pattern) => Pattern): Pattern[] {
  return patterns.map((p) => (p.id === id ? change(p) : p));
}

// Every change returns new objects. Saved revisions are never edited in place.
export function reduce(patterns: Pattern[], action: Action): Pattern[] {
  switch (action.type) {
    case "replace-all":
      return action.patterns;
    case "add-rule":
      return edit(patterns, action.patternId, (p) => ({ ...p, rules: [...p.rules, action.rule] }));
    case "update-rule":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: p.rules.map((r, i) => (i === action.index ? action.rule : r)),
      }));
    case "move-rule":
      return edit(patterns, action.patternId, (p) => {
        const to = action.index + action.by;
        if (to < 0 || to >= p.rules.length) return p;
        const rules = [...p.rules];
        [rules[action.index], rules[to]] = [rules[to], rules[action.index]];
        return { ...p, rules };
      });
    case "remove-rule":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: p.rules.filter((_, i) => i !== action.index),
      }));
    case "add-example":
      return edit(patterns, action.patternId, (p) =>
        action.counter
          ? { ...p, counterexamples: [...(p.counterexamples ?? []), action.example] }
          : { ...p, examples: [...p.examples, action.example] },
      );
    case "remove-example":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        examples: p.examples.filter((e) => e.id !== action.exampleId),
        counterexamples: (p.counterexamples ?? []).filter((e) => e.id !== action.exampleId),
      }));
    case "save-revision":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        revisions: [
          ...p.revisions,
          { id: action.id, savedAt: action.savedAt, note: action.note, rules: structuredClone(p.rules) },
        ],
      }));
    case "apply-proposal":
      // A person confirmed it. Keep the old weave first, so the change can be undone.
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: structuredClone(action.rules),
        revisions: [
          ...p.revisions,
          { id: action.ids[0], savedAt: action.savedAt, note: "Before proposal", rules: structuredClone(p.rules) },
          { id: action.ids[1], savedAt: action.savedAt, note: `Applied proposal from ${action.model}`, rules: structuredClone(action.rules) },
        ],
      }));
    case "restore-revision":
      return edit(patterns, action.patternId, (p) => {
        const revision = p.revisions.find((r) => r.id === action.revisionId);
        return revision ? { ...p, rules: structuredClone(revision.rules) } : p;
      });
  }
}
