We will gate every proposal through the closed rule vocabulary, show it in plain language, and apply it only as a revision a person can undo.

A raw reply can contain anything: a regex, a made-up `script` rule, a template with no `{value}`, or broken JSON. Showing that to a person is not enough. The app must decide whether a proposal is even a Pattern weave.

The gate is `checkProposal`. It accepts only the five known kinds with their exact fields and short literal text, at most six rules, and rejects the whole proposal with readable reasons otherwise. The service runs the gate, and the browser runs it again, because the app never trusts a rule it did not check itself.

**Readable** means each rule is shown as a sentence. **Reversible** means applying saves your current weave as **Before proposal** first, so one click on Restore undoes it.

```typescript
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";

import { checkProposal, type ProposalCheck } from "../src/features/pattern/rule-schema.ts";
import { extractProposal } from "./proposal.ts";

export const SERVICE_PORT = 4319;
const MAX_BODY_BYTES = 32_768;
const MAX_EXAMPLES = 12;

export type ExamplePair = { before: string; after: string };
export type Proposer = (examples: ExamplePair[]) => Promise<{ raw: string; model: string }>;

export function isAllowedOrigin(origin: string | undefined): boolean {
  if (!origin) return true; // curl and local scripts send no Origin
  try {
    const url = new URL(origin);
    return url.protocol === "http:" && (url.hostname === "localhost" || url.hostname === "127.0.0.1");
  } catch {
    return false;
  }
}

export function parseExamples(value: unknown): ExamplePair[] {
  const examples = (value as { examples?: unknown } | null)?.examples;
  if (
    !Array.isArray(examples) ||
    examples.length === 0 ||
    examples.length > MAX_EXAMPLES ||
    !examples.every((e) => typeof e?.before === "string" && typeof e?.after === "string")
  ) {
    throw new Error(`Request must contain 1 to ${MAX_EXAMPLES} examples with before and after text`);
  }
  return examples.map((e: ExamplePair) => ({ before: e.before, after: e.after }));
}

function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  res.writeHead(status, {
    "content-type": "application/json",
    ...(origin ? { "access-control-allow-origin": origin, vary: "Origin" } : {}),
    "access-control-allow-headers": "content-type",
    "access-control-allow-methods": "GET, POST, OPTIONS",
  });
  res.end(status === 204 ? undefined : JSON.stringify(body));
}

async function readJson(req: IncomingMessage): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > MAX_BODY_BYTES) throw new Error("Request is too large");
    chunks.push(chunk as Buffer);
  }
  return JSON.parse(Buffer.concat(chunks).toString("utf8"));
}

export function gateReply(raw: string): ProposalCheck {
  try {
    return checkProposal(extractProposal(raw));
  } catch (error) {
    return { ok: false, reasons: [(error as Error).message] };
  }
}

// The service proposes. It never stores patterns. The browser decides.
export function createPatternServer(options: { proposer: Proposer | null }) {
  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (!isAllowedOrigin(origin)) return send(res, 403, { error: "Origin not allowed" });
    if (req.method === "OPTIONS") return send(res, 204, null, origin);
    if (req.method === "GET" && req.url === "/health") {
      return send(res, 200, { ok: true, proposing: options.proposer !== null }, origin);
    }
    if (req.method !== "POST" || req.url !== "/propose") return send(res, 404, { error: "Not found" }, origin);
    let examples: ExamplePair[];
    try {
      examples = parseExamples(await readJson(req));
    } catch (error) {
      return send(res, 400, { error: (error as Error).message }, origin);
    }
    if (!options.proposer) return send(res, 503, { error: "Rule proposals are not available" }, origin);
    try {
      const { raw, model } = await options.proposer(examples);
      return send(res, 200, { raw, model, check: gateReply(raw) }, origin);
    } catch (error) {
      return send(res, 502, { error: (error as Error).message }, origin);
    }
  });
}
```

```tsx
const button = "rounded border border-stone-400 px-2 py-1 text-sm hover:bg-stone-100 disabled:opacity-40";

export function ProposalPanel({
  examples,
  onApply,
  request = requestProposal,
}: {
  examples: Example[];
  onApply: (rules: Rule[], model: string) => void;
  request?: (examples: Example[]) => Promise<ProposalResponse>;
}) {
  const [busy, setBusy] = useState(false);

  return (

      <h3 className="font-semibold">Propose a weave from examples</h3>

        disabled={busy || examples.length === 0}
        onClick={async () => {
          setBusy(true);
          setResult(await request(examples));
          setBusy(false);
        }}
      >
        {busy ? "Proposing…" : "Propose rules"}
      </button>
      {result?.status === "ok" && result.check.ok && (
        <div className="space-y-2">
          <p className="text-sm">Proposed by {result.model}:</p>
          <ol aria-label="Proposed rules" className="list-decimal pl-5 text-sm">
            {result.check.rules.map((rule, i) => (
              <li key={i}>{describeRule(rule)}</li>
            ))}
          </ol>
          <div className="flex gap-2">
            <button
              className={button}
              onClick={() => {
                if (result.check.ok) onApply(result.check.rules, result.model);
                setResult(null);
              }}
            >
              Apply as new revision
            </button>
            <button className={button} onClick={() => setResult(null)}>Discard</button>
          </div>
        </div>
      )}
      {result?.status === "ok" && !result.check.ok && (

              <li key={r}>{r}</li>

          </ul>
          <details>
            <summary>Raw reply</summary>
            <pre className="overflow-x-auto bg-stone-100 p-2 text-xs">{result.raw}</pre>
          </details>
        </div>
      )}
      {result && result.status !== "ok" && <p role="status" className="text-sm text-rose-700">{result.message}</p>}
    </section>
  );
}
```

```typescript
import type { Example } from "./patterns.ts";
import { checkProposal, type ProposalCheck } from "./rule-schema.ts";

export const SERVICE_URL = "http://127.0.0.1:4319";

export type ProposalResponse =
  | { status: "ok"; raw: string; model: string; check: ProposalCheck }
  | { status: "unavailable"; message: string }
  | { status: "error"; message: string };

// Transport failures and model failures are reported differently. Neither changes a pattern.
export async function requestProposal(
  examples: readonly Example[],
  fetchImpl: typeof fetch = fetch,
): Promise<ProposalResponse> {
  let res: Response;
  try {
    res = await fetchImpl(`${SERVICE_URL}/propose`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ examples: examples.map(({ before, after }) => ({ before, after })) }),
    });
  } catch {
    return { status: "unavailable", message: "The local QVAC service is not running. Keep weaving by hand." };
  }
  const body = (await res.json().catch(() => ({}))) as {
    raw?: unknown;
    model?: unknown;
    error?: unknown;
    check?: { ok?: unknown; rules?: unknown; reasons?: unknown };
  };
  if (res.status === 503) return { status: "unavailable", message: String(body.error ?? "Proposals are not available") };
  if (!res.ok || typeof body.raw !== "string" || typeof body.model !== "string") {
    return { status: "error", message: String(body.error ?? `Service answered ${res.status}`) };
  }
  // Re-check in the browser: the app never trusts a rule it did not validate itself.
  const check: ProposalCheck =
    body.check?.ok === true
      ? checkProposal(body.check.rules)
      : { ok: false, reasons: Array.isArray(body.check?.reasons) ? body.check.reasons.map(String) : ["Proposal was not checked"] };
  return { status: "ok", raw: body.raw, model: body.model, check };
}
```

```typescript
import type { Rule } from "./rules.ts";

export const LIMITS = { maxRules: 6, maxText: 40, maxParts: 6 } as const;

export type ProposalCheck =
  | { ok: true; rules: Rule[] }
  | { ok: false; reasons: string[] };

const KEYS: Record<Rule["kind"], string[]> = {
  trim: ["kind", "collapse"],
  split: ["kind", "separator", "order", "join"],
  case: ["kind", "to"],
  replace: ["kind", "find", "with"],
  format: ["kind", "template"],
};

function readableText(value: unknown, allowEmpty: boolean): value is string {
  return (
    typeof value === "string" &&
    value.length <= LIMITS.maxText &&
    (allowEmpty || value.length > 0) &&
    !/[\u0000-\u001f\u007f]/.test(value)
  );
}

function checkRule(value: unknown): Rule | string {
  if (!value || typeof value !== "object" || Array.isArray(value)) return "is not an object";
  const v = value as Record<string, unknown>;
  const kind = v.kind as Rule["kind"];
  if (!(kind in KEYS)) return `has unknown kind ${JSON.stringify(v.kind)}`;
  const extra = Object.keys(v).filter((k) => !KEYS[kind].includes(k));
  if (extra.length > 0) return `has unexpected fields ${extra.join(", ")}`;
  switch (kind) {
    case "trim":
      return typeof v.collapse === "boolean" ? { kind, collapse: v.collapse } : "needs collapse true or false";
    case "split": {
      const order = v.order;
      if (!readableText(v.separator, false) || !readableText(v.join, true)) return "needs a short separator and join";
      if (
        !Array.isArray(order) ||
        order.length === 0 ||
        order.length > LIMITS.maxParts ||
        !order.every((n) => Number.isInteger(n) && n >= 0 && n < LIMITS.maxParts)
      ) {
        return `needs 1 to ${LIMITS.maxParts} part numbers`;
      }
      return { kind, separator: v.separator as string, order: order as number[], join: v.join as string };
    }
    case "case":
      return v.to === "upper" || v.to === "lower" || v.to === "title" ? { kind, to: v.to } : "needs to: upper, lower, or title";
    case "replace":
      return readableText(v.find, false) && readableText(v.with, true)
        ? { kind, find: v.find as string, with: v.with as string }
        : "needs short literal find and with text";
    case "format":
      return readableText(v.template, false) && (v.template as string).split("{value}").length === 2
        ? { kind, template: v.template as string }
        : "needs a short template with exactly one {value}";
  }
}

// The model proposes. This gate decides whether a proposal is even readable.
// Anything outside the closed vocabulary is rejected whole, with reasons a person can read.
export function checkProposal(candidate: unknown): ProposalCheck {
  if (!Array.isArray(candidate)) return { ok: false, reasons: ["Proposal is not a list of rules"] };
  if (candidate.length === 0) return { ok: false, reasons: ["Proposal has no rules"] };
  if (candidate.length > LIMITS.maxRules) return { ok: false, reasons: [`Proposal has more than ${LIMITS.maxRules} rules`] };
  const rules: Rule[] = [];
  const reasons: string[] = [];
  candidate.forEach((item, i) => {
    const result = checkRule(item);
    if (typeof result === "string") reasons.push(`Rule ${i + 1} ${result}`);
    else rules.push(result);
  });
  return reasons.length > 0 ? { ok: false, reasons } : { ok: true, rules };
}
```

```typescript
import type { Example, Pattern } from "./patterns";
import type { Rule } from "./rules";

export type Action =
  | { type: "add-rule"; patternId: string; rule: Rule }
  | { type: "update-rule"; patternId: string; index: number; rule: Rule }
  | { type: "move-rule"; patternId: string; index: number; by: -1 | 1 }
  | { type: "remove-rule"; patternId: string; index: number }
  | { type: "add-example"; patternId: string; example: Example }
  | { type: "remove-example"; patternId: string; exampleId: string }
  | { type: "save-revision"; patternId: string; id: string; savedAt: string; note: string }
  | { type: "restore-revision"; patternId: string; revisionId: string }
  | { type: "replace-all"; patterns: Pattern[] }
  | { type: "apply-proposal"; patternId: string; rules: Rule[]; model: string; ids: [string, string]; savedAt: string };

function edit(patterns: Pattern[], id: string, change: (p: Pattern) => Pattern): Pattern[] {
  return patterns.map((p) => (p.id === id ? change(p) : p));
}

// Every change returns new objects. Saved revisions are never edited in place.
export function reduce(patterns: Pattern[], action: Action): Pattern[] {
  switch (action.type) {
    case "replace-all":
      return action.patterns;
    case "add-rule":
      return edit(patterns, action.patternId, (p) => ({ ...p, rules: [...p.rules, action.rule] }));
    case "update-rule":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: p.rules.map((r, i) => (i === action.index ? action.rule : r)),
      }));
    case "move-rule":
      return edit(patterns, action.patternId, (p) => {
        const to = action.index + action.by;
        if (to < 0 || to >= p.rules.length) return p;
        const rules = [...p.rules];
        [rules[action.index], rules[to]] = [rules[to], rules[action.index]];
        return { ...p, rules };
      });
    case "remove-rule":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: p.rules.filter((_, i) => i !== action.index),
      }));
    case "add-example":
      return edit(patterns, action.patternId, (p) => ({ ...p, examples: [...p.examples, action.example] }));
    case "remove-example":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        examples: p.examples.filter((e) => e.id !== action.exampleId),
      }));
    case "save-revision":
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        revisions: [
          ...p.revisions,
          { id: action.id, savedAt: action.savedAt, note: action.note, rules: structuredClone(p.rules) },
        ],
      }));
    case "apply-proposal":
      // A person confirmed it. Keep the old weave first, so the change can be undone.
      return edit(patterns, action.patternId, (p) => ({
        ...p,
        rules: structuredClone(action.rules),
        revisions: [
          ...p.revisions,
          { id: action.ids[0], savedAt: action.savedAt, note: "Before proposal", rules: structuredClone(p.rules) },
          { id: action.ids[1], savedAt: action.savedAt, note: `Applied proposal from ${action.model}`, rules: structuredClone(action.rules) },
        ],
      }));
    case "restore-revision":
      return edit(patterns, action.patternId, (p) => {
        const revision = p.revisions.find((r) => r.id === action.revisionId);
        return revision ? { ...p, rules: structuredClone(revision.rules) } : p;
      });
  }
}
```
