We will run each active watcher over the stored frames. Ordinary prefilters run first: source, hours, and motion. Only frames that pass are sent to the local vision model. Only a clear YES becomes a hit, and every hit waits for a person to confirm or dismiss it.

## Build and observe

### 1. Read the decision

Open `watch-eval.ts`. `passesPrefilter` is ordinary code: it needs no model. `evaluateWatcher` asks the injected `check` for each remaining frame. An `unclear` or `no` is not a hit. A service that is unavailable stops the run with a message; it does not turn into "no alerts".

### 2. Run the watchers

```bash
git checkout 5c3b3ce
npm run qvac:service
npm run dev
```

Accept the parcel watcher from chapter 04, then click **Check watchers on stored frames**. The status line reports how many frames were checked and how many hits the model proposed. Each hit shows the frame, the model name, and the model's reason.

Click **Confirm** on a hit that really shows a parcel. Click **Dismiss** on anything else. Run the check again: your decisions stay, because `mergeHits` keeps them by watcher and frame.

### 3. When the service is down

Stop the service and run the check. The status line says it stopped and why. No hit is invented and the event tape still works.

```typescript
import type { LookoutEvent } from "./events";
import type { Rule } from "./rules";
import { sampleSources } from "./sources";
import type { WatchHit } from "./watch-eval";
import type { Watcher } from "./watcher";

export type LookoutState = {
  version: 1;
  armed: Record<string, boolean>;
  rules: Rule[];
  tape: LookoutEvent[];
  watchers: Watcher[];
  hits: WatchHit[];
};

export const STORAGE_KEY = "lookout:v1";

export function defaultState(): LookoutState {
  return {
    version: 1,
    armed: Object.fromEntries(sampleSources.map((s) => [s.id, false])),
    rules: [
      { id: "rule-motion", sourceId: "any", kind: "motion", minMotion: 0.3 },
      { id: "rule-change", sourceId: "front-gate", kind: "change", minChange: 0.15 },
    ],
    tape: [],
    watchers: [],
    hits: [],
  };
}

type KeyValue = Pick<Storage, "getItem" | "setItem" | "removeItem">;

// The only place Lookout touches browser storage.
export function createStore(storage: KeyValue | undefined) {
  return {
    load(): LookoutState {
      try {
        const raw = storage?.getItem(STORAGE_KEY);
        if (!raw) return defaultState();
        const parsed = JSON.parse(raw) as Partial<LookoutState>;
        if (parsed.version !== 1 || !parsed.armed || !Array.isArray(parsed.rules) || !Array.isArray(parsed.tape)) {
          return defaultState();
        }
        return { ...defaultState(), ...parsed } as LookoutState;
      } catch {
        return defaultState();
      }
    },
    save(state: LookoutState) {
      try {
        storage?.setItem(STORAGE_KEY, JSON.stringify(state));
      } catch {
        // Storage can be full or blocked. The app keeps working in memory.
      }
    },
    reset(): LookoutState {
      try {
        storage?.removeItem(STORAGE_KEY);
      } catch {
        // ignore
      }
      return defaultState();
    },
  };
}
```

```typescript
// Fixture verdicts keyed by frame. These are hand-written, not model output.
function fixtureCheck(answers: Record<string, "yes" | "no" | "unclear">) {
  return async (frame: { id: string }): Promise<CheckOutcome> => ({
    ok: true,
    value: { frameId: frame.id, frameDigest: "d", model: "fixture", raw: "", ms: 1, verdict: { answer: answers[frame.id] ?? "no", reason: "" } },
  });
}

describe("evaluateWatcher", () => {
  it("only turns a clear yes into a pending hit", async () => {
    const run = await evaluateWatcher(watcher, gate, fixtureCheck({ "front-gate-2": "unclear", "front-gate-3": "yes" }));
    expect(run.checked).toBe(4);
    expect(run.hits.map((h) => [h.frameId, h.status])).toEqual([["front-gate-3", "pending"]]);
  });

  it("skips frames the ordinary prefilters exclude", async () => {
    const run = await evaluateWatcher({ ...watcher, minMotion: 0.3 }, gate, fixtureCheck({}));
    expect(run).toMatchObject({ checked: 2, skipped: 2 });
  });

  it("stops at the first unavailable answer and reports it", async () => {
    const down = async (): Promise<CheckOutcome> => ({ ok: false, kind: "unavailable", message: "not running" });
    const run = await evaluateWatcher(watcher, gate, down);
    expect(run).toMatchObject({ checked: 0, hits: [], errors: ["front-gate-1: not running"] });
  });

  it("keeps human decisions across re-runs", async () => {
    const first = await evaluateWatcher(watcher, gate, fixtureCheck({ "front-gate-3": "yes" }));
    const decided = decideHit(first.hits, "w1:front-gate-3", "dismissed");
    const again = mergeHits(decided, first.hits);
    expect(again).toHaveLength(1);
    expect(again[0].status).toBe("dismissed");
  });
```

```typescript
import type { CheckOutcome } from "./check-client";
import { inWindow } from "./rules.ts";
import type { Frame } from "./sources";
import type { Verdict } from "./verdict";
import type { Watcher } from "./watcher";

export type WatchHit = {
  id: string;
  watcherId: string;
  frameId: string;
  verdict: Verdict;
  model: string;
  status: "pending" | "confirmed" | "dismissed";
};

export type WatchRun = { hits: WatchHit[]; checked: number; skipped: number; errors: string[] };

// Ordinary prefilters run first. They are cheap, deterministic, and keep
// frames away from the model when the watcher cannot apply.
export function passesPrefilter(watcher: Watcher, frame: Frame): boolean {
  return (
    frame.sourceId === watcher.sourceId &&
    inWindow(frame.at, watcher.fromHour, watcher.toHour) &&
    frame.motion >= watcher.minMotion
  );
}

// The model proposes a verdict per frame. The app decides: only a clear "yes"
// becomes a hit, and every hit waits for a person to confirm or dismiss it.
export async function evaluateWatcher(
  watcher: Watcher,
  frames: Frame[],
  check: (frame: Frame, condition: string) => Promise<CheckOutcome>,
): Promise<WatchRun> {
  const run: WatchRun = { hits: [], checked: 0, skipped: 0, errors: [] };
  for (const frame of frames) {
    if (!passesPrefilter(watcher, frame)) {
      run.skipped += 1;
      continue;
    }
    const outcome = await check(frame, watcher.condition);
    if (!outcome.ok) {
      run.errors.push(`${frame.id}: ${outcome.message}`);
      // A service that is down will not come back mid-run; stop asking.
      if (outcome.kind === "unavailable") break;
      continue;
    }
    run.checked += 1;
    if (outcome.value.verdict.answer === "yes") {
      run.hits.push({
        id: `${watcher.id}:${frame.id}`,
        watcherId: watcher.id,
        frameId: frame.id,
        verdict: outcome.value.verdict,
        model: outcome.value.model,
        status: "pending",
      });
    }
  }
  return run;
}

// Re-running keeps earlier human decisions for the same watcher and frame.
export function mergeHits(previous: WatchHit[], next: WatchHit[]): WatchHit[] {
  const decided = new Map(previous.filter((h) => h.status !== "pending").map((h) => [h.id, h]));
  const merged = next.map((h) => decided.get(h.id) ?? h);
  const kept = previous.filter((h) => h.status !== "pending" && !merged.some((m) => m.id === h.id));
  return [...merged, ...kept];
}

export function decideHit(hits: WatchHit[], id: string, status: "confirmed" | "dismissed"): WatchHit[] {
  return hits.map((h) => (h.id === id ? { ...h, status } : h));
}
```
