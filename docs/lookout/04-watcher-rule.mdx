A local language model proposes it as JSON.
The app validates it against fixed lists.
A person reads every field and accepts it, or discards it.

## Build and observe

### 1. Read the authority line

Open `watcher.ts`.
The JSON Schema asks for `sourceId`, `condition`, `fromHour`, `toHour`, and `minMotion`.
A schema constrains shape.
It does not make the answer right.
So `validateWatcherProposal` checks again: the source must be one of the three real IDs, hours are whole numbers 0 to 23, motion is 0 to 1, and the condition is short and non-empty.

### 2. Read the proposer

Open `qvac-proposer.ts`.
It calls `completion` with `responseFormat: { type: "json_schema", json_schema: { name: "watcher", schema } }` and reads `(await run.final).contentText`.

### 3. Propose a watcher

```bash
git checkout f15f98d
npm run qvac:service
npm run dev
```

Type **Tell me when a parcel is left by the gate** and click **Propose watcher**.
A yellow box shows each field and the model that proposed it.
Nothing is active yet.
Click **Accept watcher** to activate it, or **Discard**.

If the model invents a source, the service answers `422 invalid_proposal` and returns the raw reply, so you can see what it said.
The UI shows "No watcher proposed" and the reply.
Stop the service and try again: the UI says the local service is not running.

### 4. A runtime import detail

`watcher.ts` imports `./verdict.ts` with the extension.
The Node service loads these files directly, without a bundler, and Node needs the extension.
Next accepts it because of `allowImportingTsExtensions`.

import type { WatcherProposer } from "./server.ts";
import { WATCHER_JSON_SCHEMA } from "../src/features/lookout/watcher.ts";

type Sdk = typeof import("@qvac/sdk");

export const LANGUAGE_MODEL = "LLAMA_3_2_1B_INST_Q4_0";

// A small local LLM that proposes a watcher as JSON. Loaded on first use.
export function createQvacProposer(): WatcherProposer & { close(): Promise<void> } {
  let sdk: Sdk | undefined;
  let loading: Promise<string> | undefined;

  function ensureModel() {
    loading ??= (async () => {
      sdk = await import("@qvac/sdk");
      return sdk.loadModel({ modelSrc: sdk.LLAMA_3_2_1B_INST_Q4_0, modelType: "llm" });
    })().catch((error: unknown) => {
      loading = undefined;
      throw error;
    });
    return loading;
  }

  return {
    model: LANGUAGE_MODEL,
    async complete(history) {
      const modelId = await ensureModel();
      const run = sdk!.completion({
        modelId,
        history,
        stream: true,
        responseFormat: { type: "json_schema", json_schema: { name: "watcher", schema: WATCHER_JSON_SCHEMA } },
      });
      return (await run.final).contentText;
    },
    async close() {
      const modelId = await loading?.catch(() => undefined);
      if (sdk && modelId) await sdk.unloadModel({ modelId, clearStorage: false }).catch(() => undefined);
    },
  };
}

async function shutdown() {
  server.close();
  await proposer?.close();
  await checker?.close();
  process.exit(0);

import { createHash } from "node:crypto";
import { createServer, type IncomingMessage, type ServerResponse } from "node:http";

import { parseCheckFrameRequest } from "../src/features/lookout/check-contract.ts";
import { sampleSources } from "../src/features/lookout/sources.ts";
import { buildConditionPrompt, parseVerdict } from "../src/features/lookout/verdict.ts";
import { buildWatcherHistory, parseWatcherJson } from "../src/features/lookout/watcher.ts";

export const SERVICE_PORT = 4319;
const MAX_BODY_BYTES = 2_100_000;

// The only thing the server knows about inference. QVAC lives behind it.
export type FrameChecker = {
  model: string;
  check(png: Buffer, prompt: string): Promise<string>;
};

export type WatcherProposer = {
  model: string;
  complete(history: ReturnType<typeof buildWatcherHistory>): Promise<string>;
};

export function isAllowedOrigin(origin: string | undefined): boolean {
  if (!origin) return true; // curl and local scripts send no Origin
  try {
    const url = new URL(origin);
    return url.protocol === "http:" && ["localhost", "127.0.0.1", "[::1]"].includes(url.hostname);
  } catch {
    return false;
  }
}

function send(res: ServerResponse, status: number, body: unknown, origin?: string) {
  res.writeHead(status, {
    "content-type": "application/json",
    ...(origin ? { "access-control-allow-origin": origin, vary: "Origin" } : {}),
    "access-control-allow-headers": "content-type",
    "access-control-allow-methods": "GET, POST, OPTIONS",
  });
  res.end(JSON.stringify(body));
}

function fail(res: ServerResponse, status: number, code: string, message: string, origin?: string) {
  send(res, status, { error: { code, message } }, origin);
}

async function readJson(req: IncomingMessage): Promise<unknown> {
  let size = 0;
  const chunks: Buffer[] = [];
  for await (const chunk of req) {
    size += (chunk as Buffer).length;
    if (size > MAX_BODY_BYTES) throw new Error("too_large");
    chunks.push(chunk as Buffer);
  }
  return JSON.parse(Buffer.concat(chunks).toString("utf8"));
}

export function createLookoutServer({
  checker,
  proposer = null,
}: {
  checker: FrameChecker | null;
  proposer?: WatcherProposer | null;
}) {
  return createServer(async (req, res) => {
    const origin = req.headers.origin;
    if (!isAllowedOrigin(origin)) return fail(res, 403, "origin_denied", "Only local pages may call Lookout's service");
    if (req.method === "OPTIONS") return send(res, 204, null, origin);

    if (req.method === "GET" && req.url === "/health") {
      return send(
        res,
        200,
        {
          service: "ready",
          vision: checker ? { model: checker.model, state: "configured" } : { state: "not_configured" },
          language: proposer ? { model: proposer.model, state: "configured" } : { state: "not_configured" },
        },
        origin,
      );
    }

    if (req.method === "POST" && req.url === "/v1/check-frame") {
      let body: unknown;
      try {
        body = await readJson(req);
      } catch (error) {
        const tooLarge = error instanceof Error && error.message === "too_large";
        return fail(res, tooLarge ? 413 : 400, tooLarge ? "too_large" : "bad_json", tooLarge ? "Frame is too large" : "Body must be JSON", origin);
      }
      const parsed = parseCheckFrameRequest(body);
      if (!parsed.ok) return fail(res, 400, "bad_request", parsed.message, origin);
      let prompt: string;
      try {
        prompt = buildConditionPrompt(parsed.value.condition);
      } catch (error) {
        return fail(res, 400, "bad_condition", (error as Error).message, origin);
      }
      if (!checker) return fail(res, 503, "vision_unavailable", "The local vision model is not configured", origin);

      const png = Buffer.from(parsed.value.pngBase64, "base64");
      const frameDigest = createHash("sha256").update(png).digest("hex");
      const startedAt = Date.now();
      try {
        const raw = await checker.check(png, prompt);
        return send(
          res,
          200,
          { frameId: parsed.value.frameId, frameDigest, model: checker.model, raw, verdict: parseVerdict(raw), ms: Date.now() - startedAt },
          origin,
        );
      } catch (error) {
        return fail(res, 503, "vision_failed", error instanceof Error ? error.message : "Vision check failed", origin);
      }
    }

    if (req.method === "POST" && req.url === "/v1/propose-watcher") {
      let body: unknown;
      try {
        body = await readJson(req);
      } catch {
        return fail(res, 400, "bad_json", "Body must be JSON", origin);
      }
      const sentence = (body as { sentence?: unknown } | null)?.sentence;
      if (typeof sentence !== "string" || !sentence.trim() || sentence.length > 300) {
        return fail(res, 400, "bad_request", "sentence is required (300 characters or fewer)", origin);
      }
      if (!proposer) return fail(res, 503, "language_unavailable", "The local language model is not configured", origin);
      let raw: string;
      try {
        raw = await proposer.complete(buildWatcherHistory(sentence, sampleSources));
      } catch (error) {
        return fail(res, 503, "language_failed", error instanceof Error ? error.message : "Proposal failed", origin);
      }
      const parsed = parseWatcherJson(raw, sampleSources.map((s) => s.id));
      // A rejected proposal still returns the raw text, so a person can see why.
      if (!parsed.ok) return send(res, 422, { error: { code: "invalid_proposal", message: parsed.message }, raw }, origin);
      return send(res, 200, { proposal: parsed.value, raw, model: proposer.model }, origin);
    }

    return fail(res, 404, "not_found", "Unknown route", origin);
  });
}

"use client";

import { useEffect, useMemo, useState } from "react";

import { acknowledge, mergeTape, replayFrames } from "./events";
import { ANY_SOURCE, describeRule, type Rule } from "./rules";
import { findFrame, sampleSources } from "./sources";
import { createStore, defaultState, type LookoutState } from "./store";
import type { WatcherProposal } from "./watcher";
import { proposeWatcher, type ProposeOutcome } from "./watcher-client";

function browserStorage() {
  try {
    return typeof window === "undefined" ? undefined : window.localStorage;
  } catch {
    return undefined;
  }
}

function sourceName(id: string) {
  return id === ANY_SOURCE ? "Any source" : (sampleSources.find((s) => s.id === id)?.name ?? id);
}

export function ControlRoom() {
  const store = useMemo(() => createStore(browserStorage()), []);
  const [state, setState] = useState<LookoutState>(defaultState);
  const [loaded, setLoaded] = useState(false);
  const [draftKind, setDraftKind] = useState<Rule["kind"]>("motion");
  const [draftSource, setDraftSource] = useState<string>(ANY_SOURCE);
  const [draftValue, setDraftValue] = useState("40");
  const [draftTo, setDraftTo] = useState("6");
  const [sentence, setSentence] = useState("");
  const [proposing, setProposing] = useState(false);
  const [proposal, setProposal] = useState<ProposeOutcome | null>(null);

  useEffect(() => {
    let active = true;
    Promise.resolve().then(() => {
      if (!active) return;
      setState(store.load());
      setLoaded(true);
    });
    return () => {
      active = false;
    };
  }, [store]);

  function update(next: LookoutState) {
    setState(next);
    store.save(next);
  }

  const armedCount = Object.values(state.armed).filter(Boolean).length;
  const unacked = state.tape.filter((e) => !e.acknowledged).length;

  function addRule() {
    const id = `rule-${Date.now()}`;
    const n = Number(draftValue);
    let rule: Rule;
    if (draftKind === "time") {
      rule = { id, sourceId: draftSource, kind: "time", fromHour: n % 24, toHour: Number(draftTo) % 24 };
    } else if (draftKind === "motion") {
      rule = { id, sourceId: draftSource, kind: "motion", minMotion: Math.min(Math.max(n, 0), 100) / 100 };
    } else {
      rule = { id, sourceId: draftSource, kind: "change", minChange: Math.min(Math.max(n, 0), 100) / 100 };
    }
    update({ ...state, rules: [...state.rules, rule] });
  }

  async function propose() {
    setProposing(true);
    setProposal(await proposeWatcher(sentence));
    setProposing(false);
  }

  function activate(p: WatcherProposal) {
    update({ ...state, watchers: [...state.watchers, { ...p, id: `watcher-${Date.now()}` }] });
    setProposal(null);
    setSentence("");
  }

  return (
    <div className="mx-auto max-w-5xl p-4 font-sans text-slate-900">
      <header className="mb-4 flex flex-wrap items-center justify-between gap-2">
        <h1 className="text-2xl font-bold">Lookout</h1>
        <button
          className="rounded border px-3 py-1 text-sm"
          onClick={() => update(store.reset())}
        >
          Reset demo data
        </button>
      </header>

      <p
        role="status"
        aria-label="Armed state"
        className={`mb-6 rounded p-4 text-xl font-bold ${armedCount > 0 ? "bg-red-600 text-white" : "bg-slate-200 text-slate-700"}`}
      >
        {armedCount > 0
          ? `ARMED · ${armedCount} of ${sampleSources.length} sources watched`
          : "DISARMED · no source is watched"}
      </p>

      <section aria-labelledby="tape-heading" className="mb-8">
        <div className="mb-2 flex flex-wrap items-center gap-3">
          <h2 id="tape-heading" className="text-lg font-semibold">
            Event tape
          </h2>
          <span className="text-sm text-slate-600">{unacked} to review</span>
          <button
            className="rounded bg-slate-900 px-3 py-1 text-sm text-white"
            onClick={() =>
              update({ ...state, tape: mergeTape(state.tape, replayFrames(sampleSources, state.armed, state.rules)) })
            }
          >
            Replay stored frames
          </button>
        </div>
        {state.tape.length === 0 ? (
          <p className="text-slate-600">
            {loaded ? "No events. Arm a source and replay its stored frames." : "Loading…"}
          </p>
        ) : (
          <ol aria-label="Events" className="space-y-2">
            {state.tape.map((event) => {
              const frame = findFrame(sampleSources, event.frameId);
              return (
                <li
                  key={event.id}
                  className={`flex flex-wrap items-center gap-3 rounded border p-2 ${event.acknowledged ? "opacity-60" : "border-red-400"}`}
                >
                  {frame ? (
                    // eslint-disable-next-line @next/next/no-img-element
                    <img src={frame.image} alt={frame.caption} width={128} height={80} className="rounded" />
                  ) : null}
                  <div className="min-w-0 flex-1">
                    <p className="font-medium">
                      {sourceName(event.sourceId)} · {event.at.replace("T", " ")}
                    </p>
                    <p className="text-sm text-slate-600">Matched: {event.reasons.join(", ")}</p>
                  </div>
                  {event.acknowledged ? (
                    <span className="text-sm">Acknowledged</span>
                  ) : (
                    <button
                      className="rounded border px-2 py-1 text-sm"
                      onClick={() => update({ ...state, tape: acknowledge(state.tape, event.id) })}
                    >
                      Acknowledge {sourceName(event.sourceId)} {event.at.slice(11)}
                    </button>
                  )}
                </li>
              );
            })}
          </ol>
        )}
      </section>

      <section aria-labelledby="sources-heading" className="mb-8">
        <h2 id="sources-heading" className="mb-2 text-lg font-semibold">
          Sources
        </h2>
        <p className="mb-2 text-sm text-slate-600">
          Stored still frames from sample sources. Lookout does not connect to cameras.
        </p>
        <ul className="grid gap-3 sm:grid-cols-3">
          {sampleSources.map((source) => {
            const armed = Boolean(state.armed[source.id]);
            const latest = source.frames[source.frames.length - 1];
            return (

import type { LookoutEvent } from "./events";
import type { Rule } from "./rules";
import { sampleSources } from "./sources";
import type { Watcher } from "./watcher";

export type LookoutState = {
  version: 1;
  armed: Record<string, boolean>;
  rules: Rule[];
  tape: LookoutEvent[];
  watchers: Watcher[];
};

export const STORAGE_KEY = "lookout:v1";

export function defaultState(): LookoutState {
  return {
    version: 1,
    armed: Object.fromEntries(sampleSources.map((s) => [s.id, false])),
    rules: [
      { id: "rule-motion", sourceId: "any", kind: "motion", minMotion: 0.3 },
      { id: "rule-change", sourceId: "front-gate", kind: "change", minChange: 0.15 },
    ],
    tape: [],
    watchers: [],
  };
}

type KeyValue = Pick<Storage, "getItem" | "setItem" | "removeItem">;

// The only place Lookout touches browser storage.
export function createStore(storage: KeyValue | undefined) {
  return {
    load(): LookoutState {
      try {
        const raw = storage?.getItem(STORAGE_KEY);
        if (!raw) return defaultState();
        const parsed = JSON.parse(raw) as Partial<LookoutState>;
        if (parsed.version !== 1 || !parsed.armed || !Array.isArray(parsed.rules) || !Array.isArray(parsed.tape)) {
          return defaultState();
        }
        return { ...defaultState(), ...parsed } as LookoutState;
      } catch {
        return defaultState();
      }
    },
    save(state: LookoutState) {
      try {
        storage?.setItem(STORAGE_KEY, JSON.stringify(state));
      } catch {
        // Storage can be full or blocked. The app keeps working in memory.
      }
    },
    reset(): LookoutState {
      try {
        storage?.removeItem(STORAGE_KEY);
      } catch {
        // ignore
      }
      return defaultState();
    },
  };
}

import { SERVICE_URL } from "./check-contract";
import { sampleSources } from "./sources";
import { validateWatcherProposal, type WatcherProposal } from "./watcher";

export type ProposeOutcome =
  | { ok: true; proposal: WatcherProposal; raw: string; model: string }
  | { ok: false; message: string; raw?: string };

export async function proposeWatcher(
  sentence: string,
  fetchImpl: typeof fetch = fetch,
  serviceUrl = SERVICE_URL,
): Promise<ProposeOutcome> {
  let response: Response;
  try {
    response = await fetchImpl(`${serviceUrl}/v1/propose-watcher`, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ sentence }),
    });
  } catch {
    return { ok: false, message: "The local QVAC service is not running" };
  }
  const body = (await response.json().catch(() => null)) as
    | { proposal?: unknown; raw?: string; model?: string; error?: { message?: string } }
    | null;
  if (!response.ok || !body) {
    return { ok: false, message: body?.error?.message ?? `Service answered ${response.status}`, raw: body?.raw };
  }
  // Validate again on this side of the boundary.
  const checked = validateWatcherProposal(body.proposal, sampleSources.map((s) => s.id));
  if (!checked.ok) return { ok: false, message: checked.message, raw: body.raw };
  return { ok: true, proposal: checked.value, raw: body.raw ?? "", model: body.model ?? "unknown" };
}

// A watcher is an inspectable rule: ordinary prefilters (source, hours, motion)
// plus one plain-language condition that a vision model checks later.
// A language model may propose one. Only a person may activate it.
import { MAX_CONDITION_LENGTH } from "./verdict.ts";

export type Watcher = {
  id: string;
  sourceId: string;
  condition: string;
  fromHour: number;
  toHour: number;
  minMotion: number;
};

export type WatcherProposal = Omit<Watcher, "id">;

export const WATCHER_JSON_SCHEMA = {
  type: "object",
  additionalProperties: false,
  required: ["sourceId", "condition", "fromHour", "toHour", "minMotion"],
  properties: {
    sourceId: { type: "string" },
    condition: { type: "string" },
    fromHour: { type: "integer", minimum: 0, maximum: 23 },
    toHour: { type: "integer", minimum: 0, maximum: 23 },
    minMotion: { type: "number", minimum: 0, maximum: 1 },
  },
} as const;

export function buildWatcherHistory(sentence: string, sources: { id: string; name: string }[]) {
  const list = sources.map((s) => `${s.id} (${s.name})`).join(", ");
  return [
    {
      role: "system" as const,
      content:
        "You turn a request into a watcher rule as JSON. " +
        `sourceId must be one of: ${list}. ` +
        "condition is a short visible situation to look for in one still frame. " +
        "fromHour and toHour are whole hours 0-23; use 0 and 0 for all day. " +
        "minMotion is 0 to 1; use 0 when the thing may be still.",
    },
    { role: "user" as const, content: sentence },
  ];
}

type Result<T> = { ok: true; value: T } | { ok: false; message: string };

function isHour(v: unknown): v is number {
  return typeof v === "number" && Number.isInteger(v) && v >= 0 && v <= 23;
}

// The app decides. Shape, the fixed source list, and ranges are checked here,
// whatever the JSON Schema promised.
export function validateWatcherProposal(value: unknown, sourceIds: readonly string[]): Result<WatcherProposal> {
  if (typeof value !== "object" || value === null) return { ok: false, message: "Proposal is not an object" };
  const v = value as Record<string, unknown>;
  if (typeof v.sourceId !== "string" || !sourceIds.includes(v.sourceId)) {
    return { ok: false, message: `Unknown source "${String(v.sourceId)}"` };
  }
  const condition = typeof v.condition === "string" ? v.condition.replace(/\s+/g, " ").trim() : "";
  if (!condition || condition.length > MAX_CONDITION_LENGTH) return { ok: false, message: "Condition is empty or too long" };
  if (!isHour(v.fromHour) || !isHour(v.toHour)) return { ok: false, message: "Hours must be whole numbers from 0 to 23" };
  if (typeof v.minMotion !== "number" || !(v.minMotion >= 0 && v.minMotion <= 1)) {
    return { ok: false, message: "minMotion must be between 0 and 1" };
  }
  return {
    ok: true,
    value: { sourceId: v.sourceId, condition, fromHour: v.fromHour, toHour: v.toHour, minMotion: v.minMotion },
  };
}

export function parseWatcherJson(raw: string, sourceIds: readonly string[]): Result<WatcherProposal> {
  try {
    return validateWatcherProposal(JSON.parse(raw), sourceIds);
  } catch {
    return { ok: false, message: "Model reply was not JSON" };
  }
}
