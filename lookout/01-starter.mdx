We will build Lookout without AI. At the end of this chapter you can arm a source, replay its stored frames through ordinary rules, review the event tape, and acknowledge events. Every later chapter starts from this commit.

Lookout is an event-first control room for visual sources you own. Events lead. Feeds support them. Armed state is impossible to miss.

Lookout does not connect to cameras. Its sources are three fictional places, each with four drawn still frames. Every frame, time, and place is invented.

Use Node 22.22 or newer. Open `http://localhost:3000`.

### 2. See the disarmed state

The banner at the top reads **DISARMED · no source is watched**. The event tape says there are no events. Nothing is watched until you say so.

### 3. Arm a source

Click **Arm Front gate**. The banner turns red and reads **ARMED · 1 of 3 sources watched**. The Front gate card gets a red border. You cannot miss it, and that is the point.

### 4. Replay stored frames

Click **Replay stored frames**. Lookout runs the four Front gate frames through two default rules: motion at least 30% on any source, and scene change at least 15% on the Front gate.

Three events appear, newest first: the cat at 22:15, the parcel at 10:44, and the courier at 10:42. The empty gate at 07:10 raises nothing.

Look at the cat. It is a real event by these rules and a false alarm by any human standard. Plain motion and change rules cannot tell a cat from a parcel. Keep that in mind for chapter 02.

### 5. Acknowledge and reload

Click **Acknowledge Front gate 10:44**. The counter drops from 3 to 2. Reload the page. The acknowledgement is still there, because `store.ts` saved it.

Open `store.ts`. It is the only file that touches browser storage. If storage is blocked or the saved JSON is broken, Lookout falls back to defaults and keeps working. **Reset demo data** clears it.

### 6. Write down the rule for this course

> A model may propose verdicts and watcher rules. The app decides what it will believe. Only a person activates a watcher or confirms an alert.

We will test this rule in every chapter.

import { evaluateRules, type Rule } from "./rules";
import { allFrames, type Source } from "./sources";

export type LookoutEvent = {
  id: string;
  frameId: string;
  sourceId: string;
  at: string;
  reasons: string[];
  acknowledged: boolean;
};

// Replays every stored frame through the ordinary rules. Disarmed sources are
// skipped entirely: they never raise events.
export function replayFrames(
  sources: Source[],
  armed: Record<string, boolean>,
  rules: Rule[],
): LookoutEvent[] {
  const events: LookoutEvent[] = [];
  for (const frame of allFrames(sources)) {
    if (!armed[frame.sourceId]) continue;
    const reasons = evaluateRules(rules, frame);
    if (!reasons) continue;
    events.push({
      id: `evt-${frame.id}`,
      frameId: frame.id,
      sourceId: frame.sourceId,
      at: frame.at,
      reasons,
      acknowledged: false,
    });
  }
  // The tape shows newest first.
  return events.reverse();
}

// Keeps acknowledgements across replays of the same frames.
export function mergeTape(previous: LookoutEvent[], next: LookoutEvent[]): LookoutEvent[] {
  const acked = new Set(previous.filter((e) => e.acknowledged).map((e) => e.id));
  return next.map((e) => (acked.has(e.id) ? { ...e, acknowledged: true } : e));
}

export function acknowledge(tape: LookoutEvent[], eventId: string): LookoutEvent[] {
  return tape.map((e) => (e.id === eventId ? { ...e, acknowledged: true } : e));
}

import type { Frame } from "./sources";

// Ordinary, deterministic rules. No model is involved.
export type Rule =
  | { id: string; sourceId: string; kind: "motion"; minMotion: number }
  | { id: string; sourceId: string; kind: "change"; minChange: number }
  | { id: string; sourceId: string; kind: "time"; fromHour: number; toHour: number };

export const ANY_SOURCE = "any";

function hourOf(at: string): number {
  return Number(at.slice(11, 13));
}

// True when the hour falls inside [fromHour, toHour). Windows may wrap midnight.
export function inWindow(at: string, fromHour: number, toHour: number): boolean {
  const h = hourOf(at);
  if (fromHour === toHour) return true;
  return fromHour < toHour ? h >= fromHour && h < toHour : h >= fromHour || h < toHour;
}

export function describeRule(rule: Rule): string {
  switch (rule.kind) {
    case "motion":
      return `motion ≥ ${Math.round(rule.minMotion * 100)}%`;
    case "change":
      return `scene change ≥ ${Math.round(rule.minChange * 100)}%`;
    case "time":
      return `between ${String(rule.fromHour).padStart(2, "0")}:00 and ${String(rule.toHour).padStart(2, "0")}:00`;
  }
}

function applies(rule: Rule, frame: Frame): boolean {
  return rule.sourceId === ANY_SOURCE || rule.sourceId === frame.sourceId;
}

function matches(rule: Rule, frame: Frame): boolean {
  switch (rule.kind) {
    case "motion":
      return frame.motion >= rule.minMotion;
    case "change":
      return frame.change >= rule.minChange;
    case "time":
      return inWindow(frame.at, rule.fromHour, rule.toHour);
  }
}

// A frame raises an event when any applicable motion or change rule matches and
// every applicable time rule matches. Time rules only narrow; they never trigger alone.
export function evaluateRules(rules: Rule[], frame: Frame): string[] | null {
  const applicable = rules.filter((r) => applies(r, frame));
  const windows = applicable.filter((r) => r.kind === "time");
  const fired = applicable.filter((r) => r.kind !== "time" && matches(r, frame));
  if (fired.length === 0) return null;
  if (!windows.every((r) => matches(r, frame))) return null;
  return [...windows, ...fired].map(describeRule);
}

  for (const source of sources) {

import type { LookoutEvent } from "./events";
import type { Rule } from "./rules";
import { sampleSources } from "./sources";

export type LookoutState = {
  version: 1;
  armed: Record<string, boolean>;
  rules: Rule[];
  tape: LookoutEvent[];
};

export const STORAGE_KEY = "lookout:v1";

export function defaultState(): LookoutState {
  return {
    version: 1,
    armed: Object.fromEntries(sampleSources.map((s) => [s.id, false])),
    rules: [
      { id: "rule-motion", sourceId: "any", kind: "motion", minMotion: 0.3 },
      { id: "rule-change", sourceId: "front-gate", kind: "change", minChange: 0.15 },
    ],
    tape: [],
  };
}

type KeyValue = Pick<Storage, "getItem" | "setItem" | "removeItem">;

// The only place Lookout touches browser storage.
export function createStore(storage: KeyValue | undefined) {
  return {
    load(): LookoutState {
      try {
        const raw = storage?.getItem(STORAGE_KEY);
        if (!raw) return defaultState();
        const parsed = JSON.parse(raw) as Partial<LookoutState>;
        if (parsed.version !== 1 || !parsed.armed || !Array.isArray(parsed.rules) || !Array.isArray(parsed.tape)) {
          return defaultState();
        }
        return { ...defaultState(), ...parsed } as LookoutState;
      } catch {
        return defaultState();
      }
    },
    save(state: LookoutState) {
      try {
        storage?.setItem(STORAGE_KEY, JSON.stringify(state));
      } catch {
        // Storage can be full or blocked. The app keeps working in memory.
      }
    },
    reset(): LookoutState {
      try {
        storage?.removeItem(STORAGE_KEY);
      } catch {
        // ignore
      }
      return defaultState();
    },
  };
}
